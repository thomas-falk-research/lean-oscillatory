/-- PLANTED copy of `TriCrossoverUpperStatement`: crossover scale t^{1/3} ↦ t^{1/4}. -/
def P_TriCrossoverUpperStatement : Prop :=
  ∀ a₀ : ℝ, |a₀| = triExceptional → ∃ r > 0, ∃ C : ℝ, ∀ (α t : ℝ) (n : ℤ × ℤ), |α - a₀| ≤ r → 1 ≤ t →
    ‖triKernel α t n‖ ≤ C * t ^ (-(2 : ℝ) / 3) * crossEnvelope (|α - a₀| * t ^ ((1 : ℝ) / 4))

/-- Witness: the draft implies the planted copy (`t^{1/4} ≤ t^{1/3}` for `t ≥ 1`, the envelope is antitone). -/
theorem witness_A_h (h : TriCrossoverUpperStatement) : P_TriCrossoverUpperStatement := fun a₀ ha => by
  obtain ⟨r, hr, C, hC⟩ := h a₀ ha
  refine ⟨r, hr, C, fun α t n hα ht => ?_⟩
  have h1 := hC α t n hα ht
  have hC0 := a5u_const_nonneg (by linarith) h1
  have hmono : crossEnvelope (|α - a₀| * t ^ ((1 : ℝ) / 3)) ≤ crossEnvelope (|α - a₀| * t ^ ((1 : ℝ) / 4)) :=
    a5u_env_antitone (mul_le_mul_of_nonneg_left (Real.rpow_le_rpow_of_exponent_le ht (by norm_num))
      (abs_nonneg _))
  exact h1.trans (mul_le_mul_of_nonneg_left hmono (mul_nonneg hC0 (Real.rpow_nonneg (by linarith) _)))

#print axioms witness_A_h

end OscInt.API.TriangularCrossoverSanity
