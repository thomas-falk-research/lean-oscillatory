import OscInt.Front.TriangularCrossoverSanity

namespace OscInt.API.TriangularCrossoverSanity

open OscInt.Statements OscInt.API.TriangularDecaySanity OscInt.API.TriangularAnisoSanity

/-- PLANTED copy of `crossEnvelope`: envelope exponent −1/4 ↦ −1/3. -/
noncomputable def P_crossEnvelope (s : ℝ) : ℝ := if s ≤ 1 then 1 else s ^ (-(1 : ℝ) / 3)

/-- PLANTED copy of `TriCrossoverUpperStatement`: envelope exponent −1/4 ↦ −1/3. -/
def P_TriCrossoverUpperStatement : Prop :=
  ∀ a₀ : ℝ, |a₀| = triExceptional → ∃ r > 0, ∃ C : ℝ, ∀ (α t : ℝ) (n : ℤ × ℤ), |α - a₀| ≤ r → 1 ≤ t →
    ‖triKernel α t n‖ ≤ C * t ^ (-(2 : ℝ) / 3) * P_crossEnvelope (|α - a₀| * t ^ ((1 : ℝ) / 3))

/-- Witness (relative to K2-D): at `α = a* + δ`, `0 < δ ≤ min(r, a*)/2`, `triRate α = 3/4`, but for `t > δ^{−3}` the
planted envelope gives `C δ^{−1/3} t^{−7/9}`, a faster rate. -/
theorem witness_A_a_rel (hD : TriDecayLowerStatement) : ¬ P_TriCrossoverUpperStatement := by
  intro h
  have hp := triExceptional_pos
  obtain ⟨r, hr, C, hC⟩ := h triExceptional a5u_abs_exceptional
  obtain ⟨δ, hδ, hδr, hδa⟩ : ∃ δ : ℝ, 0 < δ ∧ δ ≤ r ∧ δ < triExceptional :=
    ⟨min r triExceptional / 2, half_pos (lt_min hr hp),
      (half_le_self (lt_min hr hp).le).trans (min_le_left _ _),
      (half_lt_self (lt_min hr hp)).trans_le (min_le_right _ _)⟩
  refine a5u_rate_clash hD (triExceptional + δ) (7 / 9) (C * δ ^ (-(1 : ℝ) / 3)) (δ⁻¹ ^ 3 + 1)
    (by rw [a5u_triRate_window a5u_abs_exceptional hδ hδa]; norm_num) fun t n ht => ?_
  have h3 : 0 ≤ δ⁻¹ ^ 3 := by positivity
  have ht1 : 1 ≤ t := by linarith
  have ht0 : 0 < t := by linarith
  have hs : 1 < δ * t ^ ((1 : ℝ) / 3) := a5u_scale_gt_one hδ (by linarith)
  have h1 := hC (triExceptional + δ) t n (by rw [add_sub_cancel_left, abs_of_pos hδ]; exact hδr) ht1
  have e : P_crossEnvelope (δ * t ^ ((1 : ℝ) / 3)) = (δ * t ^ ((1 : ℝ) / 3)) ^ (-(1 : ℝ) / 3) := by
    simp only [P_crossEnvelope, not_le.mpr hs, ↓reduceIte]
  rw [add_sub_cancel_left, abs_of_pos hδ, e] at h1
  have e2 : t ^ (-(2 : ℝ) / 3) * (δ * t ^ ((1 : ℝ) / 3)) ^ (-(1 : ℝ) / 3) =
      δ ^ (-(1 : ℝ) / 3) * t ^ (-(7 / 9 : ℝ)) := by
    rw [Real.mul_rpow hδ.le (Real.rpow_nonneg ht0.le _), ← Real.rpow_mul ht0.le, mul_left_comm,
      ← Real.rpow_add ht0]
    norm_num
  calc ‖triKernel (triExceptional + δ) t n‖ ≤
        C * t ^ (-(2 : ℝ) / 3) * (δ * t ^ ((1 : ℝ) / 3)) ^ (-(1 : ℝ) / 3) := h1
    _ = C * δ ^ (-(1 : ℝ) / 3) * t ^ (-(7 / 9 : ℝ)) := by rw [mul_assoc, e2, mul_assoc]

/-- Unconditional, from the proved K2-D (`OscInt.API.triDecayLower`). -/
theorem witness_A_a : ¬ P_TriCrossoverUpperStatement :=
  witness_A_a_rel OscInt.API.triDecayLower

#print axioms witness_A_a_rel

#print axioms witness_A_a

end OscInt.API.TriangularCrossoverSanity
