import OscInt.Front.TriangularCrossoverSanity

namespace OscInt.API.TriangularCrossoverSanity

open OscInt.Statements OscInt.API.TriangularDecaySanity OscInt.API.TriangularAnisoSanity

/-- PLANTED copy of `TriCrossoverUpperStatement`: envelope argument |α − a₀| ↦ |α| (distance to 0, not to a₀). -/
def P_TriCrossoverUpperStatement : Prop :=
  ∀ a₀ : ℝ, |a₀| = triExceptional → ∃ r > 0, ∃ C : ℝ, ∀ (α t : ℝ) (n : ℤ × ℤ), |α - a₀| ≤ r → 1 ≤ t →
    ‖triKernel α t n‖ ≤ C * t ^ (-(2 : ℝ) / 3) * crossEnvelope (|α| * t ^ ((1 : ℝ) / 3))

/-- Witness (relative to K2-D): at the centre `α = a₀ = a*` the planted envelope argument is `a* t^{1/3}`, so the
plant claims `C a*^{−1/4} t^{−3/4}` where K2-D gives `c t^{−2/3}`. -/
theorem witness_A_j_rel (hD : TriDecayLowerStatement) : ¬ P_TriCrossoverUpperStatement := by
  intro h
  have hp := triExceptional_pos
  obtain ⟨r, hr, C, hC⟩ := h triExceptional a5u_abs_exceptional
  have hw : |triExceptional - triExceptional| ≤ r := by rw [sub_self, abs_zero]; exact hr.le
  have hC0 : 0 ≤ C := a5u_const_nonneg one_pos (hC _ 1 (0, 0) hw le_rfl)
  refine a5u_rate_clash hD triExceptional (3 / 4) (C * triExceptional ^ (-(1 : ℝ) / 4)) 1
    (by rw [triRate_exceptional]; norm_num) fun t n ht => ?_
  have h1 := hC _ t n hw ht
  rw [a5u_abs_exceptional, mul_assoc] at h1
  calc ‖triKernel triExceptional t n‖ ≤ C * (t ^ (-(2 : ℝ) / 3) * crossEnvelope (triExceptional * t ^ ((1 : ℝ) / 3))) := h1
    _ ≤ C * (triExceptional ^ (-(1 : ℝ) / 4) * t ^ (-(3 : ℝ) / 4)) :=
        mul_le_mul_of_nonneg_left (a5u_envelope_le hp (by linarith)) hC0
    _ = C * triExceptional ^ (-(1 : ℝ) / 4) * t ^ (-(3 / 4 : ℝ)) := by
        rw [show (-(3 : ℝ) / 4) = -(3 / 4) by ring]
        ring

/-- Unconditional, from the proved K2-D (`OscInt.API.triDecayLower`). -/
theorem witness_A_j : ¬ P_TriCrossoverUpperStatement :=
  witness_A_j_rel OscInt.API.triDecayLower

#print axioms witness_A_j_rel

#print axioms witness_A_j

end OscInt.API.TriangularCrossoverSanity
