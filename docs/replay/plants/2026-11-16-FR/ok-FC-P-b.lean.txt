/-- The kernel of `i u′ + Δu = 0` on the line `ℤ`: `G(t, n) = (2π)^{−1} e^{−2it} I(2t, n)`. -/
noncomputable def lineKernel (t : ℝ) (n : ℤ) : ℂ :=
  ((2 * Real.pi : ℝ) : ℂ)⁻¹ * Complex.exp (↑(-(2 * t)) * Complex.I) * latticeKernel1 (2 * t) (n : ℝ)

/-- The kernel on the ring of `N` sites: `R_N(t, r) = N^{−1} Σ_{j<N} e^{i(2t(cos k_j − 1) + r k_j)}`, `k_j = 2πj/N`. -/
noncomputable def ringKernel (N : ℕ) (t : ℝ) (r : ℤ) : ℂ :=
  ((N : ℝ) : ℂ)⁻¹ * ∑ j ∈ Finset.range N,
    Complex.exp (↑(2 * t * (Real.cos (2 * Real.pi * (j : ℝ) / (N : ℝ)) - 1) +
      (r : ℝ) * (2 * Real.pi * (j : ℝ) / (N : ℝ))) * Complex.I)

/-- The kernel on the open chain of the sites `1, …, M`: the odd reflection of the ring of `2(M + 1)` sites,
`C_M(t; x, y) = R_{2(M+1)}(t, x − y) − R_{2(M+1)}(t, x + y)`. -/
noncomputable def chainKernel (M : ℕ) (t : ℝ) (x y : ℤ) : ℂ :=
  ringKernel (2 * (M + 1)) t (x - y) - ringKernel (2 * (M + 1)) t (x + y)

/-- **FR-K** The ring kernel solves `i ∂_t R + ΔR = 0` in the site variable, equals the unit datum of the ring at
`t = 0` (1 on `Nℤ`, 0 elsewhere) and has period `N`. -/
def RingKernelStatement : Prop :=
  ∀ (N : ℕ) (t : ℝ) (r : ℤ), 0 < N →
    HasDerivAt (fun s : ℝ => ringKernel N s r)
      (Complex.I * (ringKernel N t (r + 1) + ringKernel N t (r - 1) - 2 * ringKernel N t r)) t ∧
    ringKernel N 0 r = (if (N : ℤ) ∣ r then 1 else 0) ∧
    ringKernel N t (r + (N : ℤ)) = ringKernel N t r

theorem statement_pin_RingKernel : RingKernelStatement = RingKernelStatement := rfl

/-- **FR-U** Every solution of period `N` of `i u′ + Δu = 0` is the kernel sum of its datum over one period:
`u(t, x) = Σ_{y<N} R_N(t, x − y) u(0, y)`. -/
def RingUniqueStatement : Prop :=
  ∀ (N : ℕ) (u : ℝ → ℤ → ℂ), 0 < N → (∀ (t : ℝ) (x : ℤ), u t (x + (N : ℤ)) = u t x) →
    (∀ (t : ℝ) (x : ℤ), HasDerivAt (fun s : ℝ => u s x) (Complex.I * (u t (x + 1) + u t (x - 1) - 2 * u t x)) t) →
    ∀ (t : ℝ) (x : ℤ), u t x = ∑ y ∈ Finset.range N, ringKernel N t (x - (y : ℤ)) * u 0 (y : ℤ)

theorem statement_pin_RingUnique : RingUniqueStatement = RingUniqueStatement := rfl

/-- **FR-S** Method of images: the ring kernel is the line kernel summed over the coset `r + Nℤ`. -/
def RingCosetSumStatement : Prop :=
  ∀ (N : ℕ) (t : ℝ) (r : ℤ), 0 < N →
    HasSum (fun m : ℤ => lineKernel t (r + m * (N : ℤ))) (ringKernel N t r)

theorem statement_pin_RingCosetSum : RingCosetSumStatement = RingCosetSumStatement := rfl

/-- **FR-P** Finite-size error of the ring: `|R_N(t, r) − G(t, r)| ≤ 2 e^{2|t| sinh η} cosh(rη) / (e^{ηN} − 1)` for every
`η > 0`. -/
def RingFiniteSizeStatement : Prop :=
  ∀ (N : ℕ) (t η : ℝ) (r : ℤ), 0 < N → 0 < η →
    ‖ringKernel N t r - lineKernel t r‖ ≤
      2 * Real.exp (2 * |t| * Real.sinh η) * Real.cosh ((r : ℝ) * η) / (Real.exp (η * (N : ℝ)) - 1)

theorem statement_pin_RingFiniteSize : RingFiniteSizeStatement = RingFiniteSizeStatement := rfl

/-- **FC-K** The chain kernel solves the equation in the site variable, vanishes at the walls `x = 0` and
`x = M + 1`, and is the unit datum at `t = 0` on the sites `1, …, M`. -/
def ChainKernelStatement : Prop :=
  ∀ (M : ℕ) (t : ℝ) (x y : ℤ), 1 ≤ y → y ≤ (M : ℤ) →
    HasDerivAt (fun s : ℝ => chainKernel M s x y)
      (Complex.I * (chainKernel M t (x + 1) y + chainKernel M t (x - 1) y - 2 * chainKernel M t x y)) t ∧
    chainKernel M t 0 y = 0 ∧ chainKernel M t ((M : ℤ) + 1) y = 0 ∧
    (1 ≤ x → x ≤ (M : ℤ) → chainKernel M 0 x y = (if x = y then 1 else 0))

theorem statement_pin_ChainKernel : ChainKernelStatement = ChainKernelStatement := rfl

/-- **FC-U** Every solution of `i u′ + Δu = 0` on the open chain `1, …, M` with `u = 0` at the walls `0` and `M + 1`
is the kernel sum of its datum: `u(t, x) = Σ_{y=1}^{M} C_M(t; x, y) u(0, y)`. -/
def ChainUniqueStatement : Prop :=
  ∀ (M : ℕ) (u : ℝ → ℤ → ℂ), (∀ t : ℝ, u t 0 = 0 ∧ u t ((M : ℤ) + 1) = 0) →
    (∀ (t : ℝ) (x : ℤ), 1 ≤ x → x ≤ (M : ℤ) →
      HasDerivAt (fun s : ℝ => u s x) (Complex.I * (u t (x + 1) + u t (x - 1) - 2 * u t x)) t) →
    ∀ (t : ℝ) (x : ℤ), 1 ≤ x → x ≤ (M : ℤ) →
      u t x = ∑ y ∈ Finset.Icc (1 : ℤ) (M : ℤ), chainKernel M t x y * u 0 y

theorem statement_pin_ChainUnique : ChainUniqueStatement = ChainUniqueStatement := rfl

/-- **FC-P** Finite-size error of the open chain: for sites `1 ≤ x, y ≤ M` and every `η > 0`,
`|C_M(t; x, y) − G(t, x − y)| ≤ e^{2|t| sinh η} (e^{−(x+y)η} + 2 (cosh((x−y)η) + cosh((x+y)η)) / (e^{2(M+1)η} − 1))`. -/
def ChainFiniteSizeStatement : Prop :=
  ∀ (M : ℕ) (t η : ℝ) (x y : ℤ), 0 < η → 1 ≤ x → x ≤ (M : ℤ) → 1 ≤ y → y ≤ (M : ℤ) →
    ‖chainKernel M t x y - lineKernel t (x - y)‖ ≤
      Real.exp (2 * |t| * Real.sinh η) *
        (Real.exp (-(((x + y : ℤ) : ℝ) * η)) +
          2 * (Real.cosh (((x - y : ℤ) : ℝ) * η) + Real.cosh (((x + y : ℤ) : ℝ) * η)) /
            (Real.exp (2 * ((M : ℝ) + 1) * η) - 1))

theorem statement_pin_ChainFiniteSize : ChainFiniteSizeStatement = ChainFiniteSizeStatement := rfl

end OscInt.Statements

/-- One mode of a ring kernel: with `cos θ = c` and `e^{iθ} = z`, the summand `e^{i(2t(cos θ − 1) + rθ)}` is
`zʳ e^{2it(c − 1)}`. A fact about the exponential; no kernel is mentioned. -/
theorem exp_mode (t θ c : ℝ) (r : ℤ) (z : ℂ) (hc : Real.cos θ = c) (hz : Complex.exp (↑θ * Complex.I) = z) :
    Complex.exp (↑(2 * t * (Real.cos θ - 1) + (r : ℝ) * θ) * Complex.I) =
      z ^ r * Complex.exp (↑(2 * t * (c - 1)) * Complex.I) := by
  rw [← hz, ← Complex.exp_int_mul, ← Complex.exp_add, hc]
  congr 1; push_cast; ring

/-- `e^{iπ/2} = i`. -/
theorem exp_half_pi : Complex.exp (↑(Real.pi / 2) * Complex.I) = Complex.I := by
  rw [Complex.exp_mul_I]; simp

/-- `e^{3iπ/2} = −i`. -/
theorem exp_three_half_pi : Complex.exp (↑(3 * Real.pi / 2) * Complex.I) = -Complex.I := by
  rw [show (↑(3 * Real.pi / 2) : ℂ) * Complex.I = ↑Real.pi * Complex.I + ↑(Real.pi / 2) * Complex.I by
    push_cast; ring, Complex.exp_add, Complex.exp_pi_mul_I, exp_half_pi]
  ring

/-- `cos(3π/2) = 0`. -/
theorem cos_three_half_pi : Real.cos (3 * Real.pi / 2) = 0 := by
  rw [show 3 * Real.pi / 2 = Real.pi / 2 + Real.pi by ring, Real.cos_add_pi, Real.cos_pi_div_two, neg_zero]

/-- **Datum, the line at time 0.** `G(0, n) = δ_{n,0}`: the integral of `e^{ink}` over `[−π, π]` divided by `2π`. -/
theorem lineKernel_zero (n : ℤ) : lineKernel 0 n = if n = 0 then 1 else 0 := by
  have hpi : ((2 * Real.pi : ℝ) : ℂ) ≠ 0 := by exact_mod_cast (by positivity : (2 * Real.pi : ℝ) ≠ 0)
  rw [lineKernel]
  split_ifs with hn
  · subst hn
    simp only [mul_zero, Int.cast_zero, latticeKernel1_zero_zero, neg_zero, Complex.ofReal_zero, zero_mul,
      Complex.exp_zero, mul_one]
    exact inv_mul_cancel₀ hpi
  · have h0 := integral_exp_int_mul n hn
    have h1 : latticeKernel1 (2 * 0) (n : ℝ) = 0 := by simpa [latticeKernel1] using h0
    rw [h1, mul_zero]

/-- **Datum, ring of 4 sites.** `R₄(t, r) = (1 + (iʳ + (−i)ʳ) e^{−2it} + (−1)ʳ e^{−4it}) / 4`: the modes
`k = 0, π/2, π, 3π/2` with `cos k = 1, 0, −1, 0`. -/
theorem ringKernel_four (t : ℝ) (r : ℤ) :
    ringKernel 4 t r = (1 + (Complex.I ^ r + (-Complex.I) ^ r) * Complex.exp (↑(-(2 * t)) * Complex.I) +
      (-1) ^ r * Complex.exp (↑(-(4 * t)) * Complex.I)) / 4 := by
  have hθ1 : 2 * Real.pi * ((1 : ℕ) : ℝ) / ((4 : ℕ) : ℝ) = Real.pi / 2 := by push_cast; ring
  have hθ2 : 2 * Real.pi * ((2 : ℕ) : ℝ) / ((4 : ℕ) : ℝ) = Real.pi := by push_cast; ring
  have hθ3 : 2 * Real.pi * ((3 : ℕ) : ℝ) / ((4 : ℕ) : ℝ) = 3 * Real.pi / 2 := by push_cast; ring
  have h1 := exp_mode t (Real.pi / 2) 0 r Complex.I Real.cos_pi_div_two exp_half_pi
  have h2 := exp_mode t Real.pi (-1) r (-1) Real.cos_pi Complex.exp_pi_mul_I
  have h3 := exp_mode t (3 * Real.pi / 2) 0 r (-Complex.I) cos_three_half_pi exp_three_half_pi
  rw [show 2 * t * ((0 : ℝ) - 1) = -(2 * t) by ring] at h1 h3
  rw [show 2 * t * ((-1 : ℝ) - 1) = -(4 * t) by ring] at h2
  rw [ringKernel, Finset.sum_range_succ, Finset.sum_range_succ, Finset.sum_range_succ, Finset.sum_range_one,
    hθ1, hθ2, hθ3, h1, h2, h3]
  simp
  ring

/-- `R₄(t, 0) = (1 + 2 e^{−2it} + e^{−4it}) / 4`. -/
theorem ringKernel_four_zero (t : ℝ) :
    ringKernel 4 t 0 = (1 + 2 * Complex.exp (↑(-(2 * t)) * Complex.I) + Complex.exp (↑(-(4 * t)) * Complex.I)) / 4 := by
  rw [ringKernel_four]; simp; ring

/-- `R₄(t, 2) = (1 − 2 e^{−2it} + e^{−4it}) / 4`. -/
theorem ringKernel_four_two (t : ℝ) :
    ringKernel 4 t 2 = (1 - 2 * Complex.exp (↑(-(2 * t)) * Complex.I) + Complex.exp (↑(-(4 * t)) * Complex.I)) / 4 := by
  rw [ringKernel_four]; norm_num [zpow_ofNat]; ring

/-- **Datum, chain of 1 site.** `C₁(t; 1, 1) = R₄(t, 0) − R₄(t, 2) = e^{−2it}`: the one site of the chain has the
eigenvalue `−2` of the Dirichlet Laplacian. -/
theorem chainKernel_one (t : ℝ) : chainKernel 1 t 1 1 = Complex.exp (↑(-(2 * t)) * Complex.I) := by
  have h : chainKernel 1 t 1 1 = ringKernel 4 t 0 - ringKernel 4 t 2 := by
    rw [chainKernel]; norm_num
  rw [h, ringKernel_four_zero, ringKernel_four_two]; ring

def P_ChainFiniteSizeStatement : Prop :=
  ∀ (M : ℕ) (t η : ℝ) (x y : ℤ), 0 < η → 1 ≤ x → x ≤ (M : ℤ) → 1 ≤ y → y ≤ (M : ℤ) →
    ‖chainKernel M t x y - lineKernel t (x - y)‖ ≤
      Real.exp (2 * |t| * Real.sinh η) *
        (Real.exp (-(((x + y : ℤ) : ℝ) * η)) +
          2 * (Real.cosh (((x - y : ℤ) : ℝ) * η) + Real.cosh (((x + y : ℤ) : ℝ) * η)) /
            (Real.exp (2 * ((M : ℝ) + 1) * η) - 1))

theorem P_chainP_one (h : P_ChainFiniteSizeStatement) (t η : ℝ) (hη : 0 < η) :
    ‖Complex.exp (↑(-(2 * t)) * Complex.I) - lineKernel t 0‖ ≤
      Real.exp (2 * |t| * Real.sinh η) *
        (Real.exp (-(2 * η)) + 2 * (1 + Real.cosh (2 * η)) / (Real.exp (4 * η) - 1)) := by
  have h1 := h 1 t η 1 1 hη le_rfl (by norm_num) le_rfl (by norm_num)
  rw [chainKernel_one, show (1 : ℤ) - 1 = 0 by norm_num, show (1 : ℤ) + 1 = 2 by norm_num] at h1
  rw [show 2 * (((1 : ℕ) : ℝ) + 1) * η = 4 * η by push_cast; ring] at h1
  simpa using h1

theorem P_chainP_one_zero (h : P_ChainFiniteSizeStatement) (η : ℝ) (hη : 0 < η) :
    0 ≤ Real.exp (-(2 * η)) + 2 * (1 + Real.cosh (2 * η)) / (Real.exp (4 * η) - 1) := by
  have h1 := P_chainP_one h 0 η hη
  rw [lineKernel_zero, ite_eq_left rfl] at h1
  simpa using h1

end OscInt.API.FiniteRingSanity
