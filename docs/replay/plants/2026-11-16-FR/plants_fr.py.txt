#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-FR statements against copies of their sanity theorems.

As the batch-PA generator (docs/replay/plants/2026-11-16-PA/plants_pa.py.txt). The batch is not yet a module of the
library, so every generated file is self-contained: imports, the statement text copied verbatim from
FiniteRing-statements.lean, the unconditional data lemmas of FiniteRingSanity.lean that the file needs (copied
verbatim, selected by name closure), then, under the prefix P_, a copy of one statement (and of chainKernel with its
data lemmas where the mutation is in that definition) and the sanity theorems that go through the statement constant.
Exactly one textual mutation is applied to one copied def. The unplanted copy (ok-) must compile. Witness files
(docs/lessons.md, rule 9): a compiled proof of the negation of the planted statement from unconditional data lemmas.
No file proves, or uses a proof of, a statement of the batch.
Verdicts: DETECTED (witness compiled), NOT DETECTED (no witness), TRUE WEAKENING (implied by the draft statement).
Usage: plants_fr.py REPO SANITY_DIR OUTDIR SCRATCHDIR [plant ids]
"""
import os, re, subprocess, sys

REPO, SDIR, OUT, SCR = sys.argv[1:5]
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)

i0, i1 = STMT_TEXT.index('namespace OscInt.Statements'), STMT_TEXT.index('end OscInt.Statements')
STMT_TEXT = STMT_TEXT[i0:i1] + 'end OscInt.Statements\n'
STMT = STMT_TEXT.split('\n')
SAN_ALL = open(os.path.join(SDIR, 'FiniteRingSanity.lean')).read()
SAN_BODY = SAN_ALL[SAN_ALL.index('namespace OscInt.API.FiniteRingSanity'):]
SAN = SAN_BODY.split('\n')
K, U, S, P = 'RingKernelStatement', 'RingUniqueStatement', 'RingCosetSumStatement', 'RingFiniteSizeStatement'
CK, CU, CP = 'ChainKernelStatement', 'ChainUniqueStatement', 'ChainFiniteSizeStatement'
THMS = {K: ['ringK_two_deriv', 'ringK_unit', 'ringK_period'], U: ['ringU_mode', 'ringU_const'],
        S: ['ringS_origin', 'ringS_two'], P: ['ringP_one', 'ringP_time', 'ringP_neg_one'],
        CK: ['chainK_one', 'chainK_unit', 'chainK_two_offdiag'], CU: ['chainU_one'], CP: ['chainP_one', 'chainP_one_zero']}
CHAIN = 'chainKernel'
CHAIN_DATA = ['chainKernel_one', 'chainKernel_one_wall_zero', 'chainKernel_one_wall_two']
DECL = re.compile(r'^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9\']+)')


def block(lines, name):
    """The declaration `name` without its docstring, up to the next blank line."""
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    """All declarations of the sanity file with docstrings, in file order: name -> (index, text)."""
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()
THROUGH = {t for ts in THMS.values() for t in ts}


def closure(text):
    """Unconditional data declarations of the sanity file used by `text`, transitively, in file order."""
    need, todo = set(), [text]
    while todo:
        t = todo.pop()
        for w in set(re.findall(r"[A-Za-z_][A-Za-z_0-9']*", t)):
            if w in DATA and w not in need and w not in THROUGH:
                need.add(w)
                todo.append(DATA[w][1])
    return [DATA[n][1] for n in sorted(need, key=lambda n: DATA[n][0])]


HEADER = """import OscInt.Statements.PeriodicAnalytic
import OscInt.Front.PeriodicAnalyticSanity

"""
OPEN = """
namespace OscInt.API.FiniteRingSanity

open OscInt.Statements OscInt.API.GlobalSanity OscInt.API.LatticeExplicitSanity OscInt.API.LatticeCubeRootSanity
"""


def make(stmt, mut, planted, witness=None):
    target, old, new, count = mut
    in_def = target == CHAIN
    defs = [target] if target == stmt else [target, stmt]
    parts = []
    for d in defs:
        t = block(STMT, d)
        if planted and d == target:
            assert t.count(old) == count, (d, old, t.count(old))
            t = t.replace(old, new)
        parts.append(t)
    renamed = list(defs)
    if in_def:
        parts += [block(SAN, th) for th in CHAIN_DATA] if witness is None else []
        renamed += CHAIN_DATA if witness is None else []
    if witness is None:
        parts += [block(SAN, th) for th in THMS[stmt]]
        renamed += THMS[stmt]
    else:
        parts.append(f'theorem witness_not_{stmt} : ¬ {stmt} :=\n  {witness}')
    text = '\n\n'.join(parts)
    for n in sorted(renamed, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    data = closure(text)
    return (HEADER + STMT_TEXT + OPEN + '\n' + '\n\n'.join(data + [text]) + '\n\nend OscInt.API.FiniteRingSanity\n')


def run(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs

        line += '\n    first error: ' + errs[0].split('/')[-1][:160]
    by_pid.setdefault(pid, {})[tag.split('-')[0]] = (verdict, line)
lines = []
for pid, stmt, mut, witness, note in PLANTS:
    if pid not in by_pid:
        continue
    r = by_pid[pid]
    lines += [r[k][1] for k in ('ok', 'plant', 'witness') if k in r]
    if witness:
        v = ('DETECTED (witness compiled: ¬ planted statement)' if r['witness'][0] == 'accepted'
             else 'WITNESS FAILED  <-- MISMATCH')
    elif pid in TRUE_WEAKENING:
        v = 'TRUE WEAKENING: ' + TRUE_WEAKENING[pid]
    else:
        v = 'NOT DETECTED: ' + NOT_DETECTED[pid]
    lines.append(f'  {pid} [{note}]: {v}')
open(os.path.join(OUT, 'SUMMARY.txt' if not ONLY else 'SUMMARY-partial.txt'), 'w').write('\n'.join(lines) + '\n')
print('\n'.join(lines))
