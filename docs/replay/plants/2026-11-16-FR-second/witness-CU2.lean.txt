def P_ChainUniqueStatement : Prop :=
  ∀ (M : ℕ) (u : ℝ → ℤ → ℂ), (∀ t : ℝ, u t 0 = 0 ∧ u t ((M : ℤ) + 1) = 0) →
    ∀ (t : ℝ) (x : ℤ), 1 ≤ x → x ≤ (M : ℤ) →
      u t x = ∑ y ∈ Finset.Icc (1 : ℤ) (M : ℤ), chainKernel M t x y * u 0 y

theorem witness_not_ChainUniqueStatement : ¬ P_ChainUniqueStatement :=
  by
  intro h
  have h1 := h 1 (fun s x => if x = 1 then (s : ℂ) else 0) (fun t => by norm_num) 1 1 le_rfl (by norm_num)
  rw [Nat.cast_one, Finset.Icc_self, Finset.sum_singleton] at h1
  simp at h1

end OscInt.API.FiniteRingSanity2
