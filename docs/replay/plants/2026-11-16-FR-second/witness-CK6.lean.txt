/-- `R₄(t, −2) = (1 − 2 e^{−2it} + e^{−4it}) / 4`. -/
theorem ringKernel_four_neg_two (t : ℝ) :
    ringKernel 4 t (-2) =
      (1 - 2 * Complex.exp (↑(-(2 * t)) * Complex.I) + Complex.exp (↑(-(4 * t)) * Complex.I)) / 4 := by
  rw [ringKernel_four]; norm_num [zpow_neg, zpow_ofNat]; ring

/-- **Data for the planted control U1**: `R₄(0, 0) = 1` and `R₄(0, −1) = R₄(0, −2) = 0`. -/
theorem ringKernel_four_time_zero_neg :
    ringKernel 4 0 0 = 1 ∧ ringKernel 4 0 (-1) = 0 ∧ ringKernel 4 0 (-2) = 0 := by
  refine ⟨?_, ?_, ?_⟩
  · rw [ringKernel_four_zero]; norm_num
  · rw [ringKernel_four_neg_one]; simp
  · rw [ringKernel_four_neg_two]; norm_num

def P_ChainKernelStatement : Prop :=
  ∀ (M : ℕ) (t : ℝ) (x y : ℤ), 1 ≤ y → y ≤ (M : ℤ) →
    HasDerivAt (fun s : ℝ => chainKernel M s x y)
      (Complex.I * (chainKernel M t (x + 1) y + chainKernel M t (x - 1) y - 2 * chainKernel M t x y)) t ∧
    chainKernel M t 0 y = 0 ∧ chainKernel M t ((M : ℤ) + 1) y = 0 ∧
    (x ≤ (M : ℤ) → chainKernel M 0 x y = (if x = y then 1 else 0))

theorem witness_not_ChainKernelStatement : ¬ P_ChainKernelStatement :=
  by
  intro h
  have h1 := (h 1 0 (-1) 1 le_rfl (by norm_num)).2.2.2 (by norm_num)
  have e : chainKernel 1 0 (-1) 1 = ringKernel 4 0 (-2) - ringKernel 4 0 0 := by
    rw [chainKernel]; norm_num
  rw [e, ringKernel_four_time_zero_neg.2.2, ringKernel_four_time_zero_neg.1, ite_eq_right (by norm_num)] at h1
  norm_num at h1

end OscInt.API.FiniteRingSanity2
