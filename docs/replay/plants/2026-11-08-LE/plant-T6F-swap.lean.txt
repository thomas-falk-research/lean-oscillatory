import OscInt.Front.LatticeExplicitSanity

namespace OscInt.API.LatticeExplicitSanity

open OscInt.Statements OscInt.API.GlobalSanity

def P_LatticeKernelFactorStatement : Prop :=
  ∀ (t : ℝ) (x : ℤ × ℤ), latticeKernel t x = latticeKernel1 t (x.1 : ℝ) * latticeKernel1 t (x.1 : ℝ)

theorem P_latticeKernelFactor_generic (h : P_LatticeKernelFactorStatement) :
    latticeKernel 3 (1, -2) = latticeKernel1 3 1 * latticeKernel1 3 (-2) := by
  simpa using h 3 (1, -2)

end OscInt.API.LatticeExplicitSanity
