def P_LatticeKernel1ExplicitStatement : Prop :=
  ∀ t ξ : ℝ, 1 < t →
    ‖latticeKernel1 t ξ‖ ≤
      24 * (2 : ℝ) ^ ((1 : ℝ) / 4) * t ^ (-(1 : ℝ) / 2) + 36 * (2 : ℝ) ^ ((1 : ℝ) / 6) * t ^ (-(1 : ℝ) / 3)

theorem P_latticeKernel1Explicit_eighth (h : P_LatticeKernel1ExplicitStatement) :

      72 * (2 : ℝ) ^ ((1 : ℝ) / 6) := by
  have := h (1 / 8) (1 / 2) (by norm_num)
  rw [eighth_rpow_neg_third] at this
  linarith
