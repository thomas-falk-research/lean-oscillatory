import OscInt.Front.TriangularCrossoverSanity

namespace OscInt.API.TriangularCrossoverSanity

open OscInt.Statements OscInt.API.TriangularDecaySanity OscInt.API.TriangularAnisoSanity

/-- PLANTED copy of `TriCrossoverUpperStatement`: `∃ C` moved after `∀ α` (C may depend on α). -/
def P_TriCrossoverUpperStatement : Prop :=
  ∀ a₀ : ℝ, |a₀| = triExceptional → ∃ r > 0, ∀ α : ℝ, |α - a₀| ≤ r → ∃ C : ℝ, ∀ (t : ℝ) (n : ℤ × ℤ), 1 ≤ t →
    ‖triKernel α t n‖ ≤ C * t ^ (-(2 : ℝ) / 3) * crossEnvelope (|α - a₀| * t ^ ((1 : ℝ) / 3))

/-- Witness: the draft implies the planted copy (the one constant serves every `α`). -/
theorem witness_A_2 (h : TriCrossoverUpperStatement) : P_TriCrossoverUpperStatement := fun a₀ ha => by
  obtain ⟨r, hr, C, hC⟩ := h a₀ ha
  exact ⟨r, hr, fun α hα => ⟨C, fun t n ht => hC α t n hα ht⟩⟩

/-- The planted copy from K2-U alone, window radius `a*/2`: at `α = a₀` (`triRate = 2/3`, envelope `1`) the K2-U
constant; at `α ≠ a₀` (`triRate α = 3/4`, `ε = |α − a₀| ≤ a*/2 < 1`) the K2-U constant `C_α`, since
`t^{−3/4} ≤ min(t^{−2/3}, ε^{−1/4} t^{−3/4}) = t^{−2/3} E(ε t^{1/3})` for `t ≥ 1`. -/
theorem witness_A_2_triU (hU : TriDecayUpperStatement) : P_TriCrossoverUpperStatement := fun a₀ ha => by
  have hp := triExceptional_pos
  have hlt := triExceptional_lt_half
  refine ⟨triExceptional / 2, half_pos hp, fun α hα => ?_⟩
  obtain ⟨C, hC⟩ := hU α
  refine ⟨C, fun t n ht => ?_⟩
  have ht0 : 0 < t := by linarith
  have h1 := hC t n ht
  have hC0 : 0 ≤ C := by
    have h2 := hC 1 n le_rfl
    rw [Real.one_rpow, mul_one] at h2
    exact (norm_nonneg _).trans h2
  by_cases he : α = a₀
  · subst he
    rw [sub_self, abs_zero, zero_mul, a5u_env_zero, mul_one]
    rwa [a5u_triRate_center ha, show -(2 / 3 : ℝ) = -(2 : ℝ) / 3 by ring] at h1
  · have hε : 0 < |α - a₀| := abs_pos.mpr (sub_ne_zero.mpr he)
    have hε1 : |α - a₀| ≤ 1 := by linarith
    have hrate : triRate α = 3 / 4 := by
      rcases a5u_exceptional_cases ha with rfl | rfl
      · have h3 := abs_le.mp hα
        refine triRate_generic α (by linarith [h3.1]) ?_
        rw [abs_of_pos (by linarith [h3.1])]
        intro e
        exact he e
      · have h3 := abs_le.mp hα
        refine triRate_generic α (by linarith [h3.2]) ?_
        rw [abs_of_neg (by linarith [h3.2])]
        intro e
        exact he (by linarith)
    rw [hrate, show -(3 / 4 : ℝ) = -(3 : ℝ) / 4 by ring] at h1
    rw [mul_assoc, a5u_envelope_min hε ht0]
    refine h1.trans (mul_le_mul_of_nonneg_left (le_min ?_ ?_) hC0)
    · exact Real.rpow_le_rpow_of_exponent_le ht (by norm_num)
    · have h4 : 1 ≤ |α - a₀| ^ (-(1 : ℝ) / 4) :=
        Real.one_le_rpow_of_pos_of_le_one_of_nonpos hε hε1 (by norm_num)
      calc t ^ (-(3 : ℝ) / 4) = 1 * t ^ (-(3 : ℝ) / 4) := (one_mul _).symm
        _ ≤ |α - a₀| ^ (-(1 : ℝ) / 4) * t ^ (-(3 : ℝ) / 4) :=
          mul_le_mul_of_nonneg_right h4 (Real.rpow_nonneg ht0.le _)

/-- The planted copy from the proved K2-U (`OscInt.API.triDecayUpper`). -/
theorem witness_A_2_proved : P_TriCrossoverUpperStatement := witness_A_2_triU OscInt.API.triDecayUpper

#print axioms witness_A_2

#print axioms witness_A_2_proved

end OscInt.API.TriangularCrossoverSanity
