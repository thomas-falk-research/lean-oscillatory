import OscInt.Front.LatticeGroupSanity

namespace OscInt.API.LatticeGroupSanity

open OscInt.Statements OscInt.API.HypercubicSanity

def P_LatticeSchrodingerDispersiveStatement : Prop :=
  ∀ (d : ℕ) (f : LatticeL2 d) (u : ℝ → LatticeL2 d), IsLatticeSchrodingerSolution d f u →
    Summable (fun y : Fin d → ℤ => ‖f y‖) →
    ∀ (t : ℝ) (x : Fin d → ℤ),
      ‖u t x‖ ≤ min 1 ((8 * (1 + Real.sqrt 7) / 3 / (2 * Real.pi)) ^ d * (2 * |t|) ^ (-(d : ℝ) / 3)) *
        ∑' y : Fin d → ℤ, ‖f y‖

theorem witness_not_LatticeSchrodingerDispersiveStatement : LatticeSchrodingerExistsStatement → ¬ P_LatticeSchrodingerDispersiveStatement :=
  by
  intro hE h
  obtain ⟨u, hu⟩ := hE 1 (delta0 1)
  exact dispersive_false_at_zero u hu (h 1 _ u hu (delta0_summable 1) 0 0)

end OscInt.API.LatticeGroupSanity
