#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-LOC2 statement against copies of its sanity theorems.

As the batch-K2 generator (docs/replay/plants/2026-11-28-K2/plants_k2.py.txt). The draft statement module
(OscInt/Statements/TorusLocalizationC2.lean, not frozen, already built in the repository) is read from a copy
(stmt/TorusLocalizationC2.lean); the sanity layer (src/OscInt/Front/TorusLocalizationC2Sanity.lean, namespace
OscInt.API.TorusLocalizationC2Sanity) is compiled into a scratch library placed first on LEAN_PATH, so every
generated file imports OscInt.Front.TorusLocalizationC2Sanity and contains, under the prefix P_: a copy of the
statement constant (and, for a plant in a definition, copies of that definition and of every definition between it
and the statement), with exactly one textual mutation in the planted copy (none in the ok- copy), followed by copies
of every sanity declaration whose text names a copied constant (renamed P_..., so that they go through the copy).
Witness files contain the planted copy and compiled theorems of one kind:
  not   : ¬ planted statement, from unconditional data                              -> DETECTED
  relp  : true statement → ¬ planted statement, and ¬ planted statement from the     -> DETECTED (via the proved
          proved true statement (OscInt.API.*, batch K2)                               statement)
  weak  : draft statement → planted statement                                     -> TRUE WEAKENING
  str   : planted statement → draft statement (a strengthening), no refutation     -> NOT DETECTED
  char  : the planted definition differs from the draft's on a datum, no refutation -> NOT DETECTED
The witness of L-a also contains, before the sanity namespace, the auxiliary namespace OscInt.FaWitnessAux copied
verbatim from the lead's compiled K2 witness (gen/witness-F-a-compiled.lean.txt = the repository's
docs/replay/plants/2026-11-28-K2-F-a-witness/witness-F-a-compiled.lean.txt), from the line after its import to
`end OscInt.FaWitnessAux`; its theorem `refute` is used at `D = locConstC2 1`.
No file proves, or uses a proof of, the batch-LOC2 statement. A rejected script alone is never counted as a
detection. Compiler output is recorded with the scratch directory removed from the paths.
Usage: plants_loc2.py WORK OUTDIR SCRATCHDIR
"""
import os, re, subprocess, sys, textwrap, time

WORK, OUT, SCR = [os.path.abspath(a) for a in sys.argv[1:4]]
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT = open(os.path.join(WORK, 'stmt/TorusLocalizationC2.lean')).read().split('\n')
SAN_ALL = open(os.path.join(WORK, 'src/OscInt/Front/TorusLocalizationC2Sanity.lean')).read()
NS = 'OscInt.API.TorusLocalizationC2Sanity'
SAN = SAN_ALL[SAN_ALL.index('namespace ' + NS):].split('\n')
FA_SRC = open(os.path.join(WORK, 'gen/witness-F-a-compiled.lean.txt')).read()
FA_AUX = FA_SRC[FA_SRC.index('\n') + 1:FA_SRC.index('end OscInt.FaWitnessAux') + len('end OscInt.FaWitnessAux')]
ST, L2, LAP, LC = 'TorusLocalizationC2Statement', 'boxL2', 'boxLaplacian', 'locConstC2'
DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([^\s:({\[]+)")
WORD = re.compile(r"[^\W\d][\w']*")


def block(lines, name):
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()


def words(t):
    ls = t.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    return set(WORD.findall('\n'.join(ls[k:])))


def through(names):
    """Sanity declarations (in file order) whose text names one of `names` or an earlier selected declaration."""
    sel, known = [], set(names)
    for n, (_, t) in sorted(DATA.items(), key=lambda kv: kv[1][0]):
        if words(t) & known:
            sel.append(n)
            known.add(n)
    return sel


assert through([ST]), ST

HEAD = f"""namespace {NS}

open OscInt.Statements OscInt.API.Poly OscInt.API.CuspSanity OscInt.API.TriangularDecaySanity MeasureTheory
open scoped ContDiff
"""
CLOSE = f'\n\nend {NS}\n'


def wrap(s):
    return '\n'.join(textwrap.wrap('/-- ' + s + ' -/', 118))


def rename(text, names):
    ls = text.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    doc, decl = '\n'.join(ls[:k]), '\n'.join(ls[k:])
    for n in sorted(names, key=len, reverse=True):
        decl = re.sub(r"(?<![\w.'])" + re.escape(n) + r"(?![\w'])", 'P_' + n, decl)
    return (doc + '\n' if doc else '') + decl


def make(copies, mut, planted, note, witness=None, pre=''):
    old, new, count = mut
    blocks = [block(STMT, c) for c in copies]
    joined = '\n\n'.join(blocks)
    if planted:
        assert joined.count(old) == count, (copies, old, joined.count(old))
        joined = joined.replace(old, new)
    blocks = joined.split('\n\n')
    assert len(blocks) == len(copies)
    parts = []
    for c, b in zip(copies, blocks):
        doc = f'PLANTED copy of `{c}`: {note}.' if planted else f'Unplanted copy of `{c}`.'
        parts.append(wrap(doc) + '\n' + b)
    names = list(copies)
    if witness is None:
        th = through(copies)
        parts += [DATA[n][1] for n in th]
        names += th
    text = '\n\n'.join(rename(p, names) for p in parts)
    if witness is not None:
        text += '\n\n' + witness.strip('\n')
        for n in re.findall(r'^theorem ([A-Za-z_0-9]+)', witness, re.M):
            text += f'\n\n#print axioms {n}'
    pre = (pre.strip('\n') + '\n\n') if (pre and witness is not None) else ''
    return 'import OscInt.Front.TorusLocalizationC2Sanity\n\n' + pre + HEAD + '\n' + text + CLOSE


def compile_text(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    t0 = time.time()
    r = subprocess.run([os.path.join(WORK, 'lean.sh'), src], capture_output=True, text=True, timeout=1800)
    out = (r.stdout + r.stderr).replace(SCR + os.sep, '').replace(WORK + os.sep, '')
    out = re.sub(r'elapsed [0-9.]+ s rc=\d+\n?', '', out)
    return r.returncode, out, time.time() - t0

VERDICT = {
    'not': 'DETECTED (witness compiled: ¬ planted statement)',
    'relp': 'DETECTED via the proved batch-K2 statement (witness compiled: true statement → ¬ planted statement, '
            'and ¬ planted statement from OscInt.API.*)',
    'weak': 'TRUE WEAKENING (compiled: draft statement → planted statement); no datum can refute it',
    'str': 'NOT DETECTED (compiled: planted statement → draft statement, a strengthening; no refutation)',
    'char': 'NOT DETECTED (compiled characterisation: the planted definition differs on a datum; no refutation)',
}

only = set(os.environ.get('ONLY', '').split(',')) - {''}
log = []
t_start = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
for pid, copies, mut, note, kind, wit, pre in PLANTS:
    if only and pid not in only:
        continue
    for tag, planted, w in (('ok-' + pid, False, None), ('plant-' + pid, True, None), ('witness-' + pid, True, wit)):
        text = make(copies, mut, planted, note, w, pre)
        rc, out, dt = compile_text(tag, text)
        want = 'recorded' if tag.startswith('plant-') else 'accepted'
        got = 'accepted' if rc == 0 else 'rejected'
        if w is not None and rc == 0:
            ax = re.findall(r"depends on axioms: \[([^\]]*)\]", out)
            nth = len(re.findall(r'^theorem ', w, re.M))
            if len(ax) != nth or any(a.replace(' ', '') != 'propext,Classical.choice,Quot.sound' for a in ax):
                got = 'rejected (axioms)'
        open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
        open(os.path.join(OUT, tag + '.out'), 'w').write(f'exit code {rc}\n' + out)
        line = f'{tag}: {got} (want {want}) [{dt:.1f} s]'
        if rc != 0:
            first = next((l for l in out.split('\n') if 'error' in l), '')
            line += '\n    first error: ' + first[:110]
        log.append(line)
        print(line, flush=True)
        if tag.startswith('witness-'):
            v = VERDICT[kind] if got == 'accepted' else 'NOT DETECTED (witness did not compile)'
            log.append(f'  {pid} [{note}]: {v}')
            print(log[-1], flush=True)
t_end = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
log.append(f'generator run {t_start} to {t_end} UTC')
open(os.path.join(OUT, 'SUMMARY.txt' if not only else 'SUMMARY-partial.txt'), 'w').write('\n'.join(log) + '\n')
