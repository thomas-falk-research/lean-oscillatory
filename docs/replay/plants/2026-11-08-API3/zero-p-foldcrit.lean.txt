open OscInt.Statements OscInt.API.ConvertAux OscInt.API.FamilySanity
def PFoldCritStatement : Prop :=
  ∀ (Λ : Type) [TopologicalSpace Λ] [CompactSpace Λ] (Ψ : Λ → ℝ × ℝ → ℝ) (K K' : Set (ℝ × ℝ)),
    FamilyCk 4 Ψ → IsCompact K → IsCompact K' →
    (∀ σ, ∀ p ∈ K', ∀ k ∈ K, IsCritical (Ψ σ) 0 k → FoldOrNondeg (Ψ σ) k) →
    ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (σ : Λ) (p : ℝ × ℝ) (m : ℝ),
      AmpBound 2 K A M → p ∈ K' → 1 ≤ m →
      ‖oscInt2 (fun k => m * (Ψ σ k + lin p k)) A‖ ≤ C * M * m ^ (-(5 : ℝ) / 6)


theorem PfoldCrit_fold_bite (h : PFoldCritStatement) : ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M m : ℝ),
    AmpBound 2 (Metric.closedBall 0 2) A M → 1 ≤ m →
    ‖oscInt2 (fun k => m * ((k.1 ^ 3 + k.2 ^ 3) + lin (0, -3) k)) A‖ ≤ C * M * m ^ (-(5 : ℝ) / 6) := by
  obtain ⟨C, hC⟩ := h Unit (fun _ => fun k : ℝ × ℝ => k.1 ^ 3 + k.2 ^ 3) (Metric.closedBall 0 2)
    {((0 : ℝ), (-3 : ℝ))} (sanity_familyCk_const 4 _ (by fun_prop)) (isCompact_closedBall _ _)
    isCompact_singleton (fun _ => cubes_fold_at_crit)
  exact ⟨C, fun A M m hA hm => hC A M () (0, -3) m hA (Set.mem_singleton _) hm⟩
