/-- **Small curvature, change of variable `x = 10u`.**
`∫_{−21}^{21} e^{i(1/100)x²/2} dx = 10 ∫_{−21/10}^{21/10} e^{iu²/2} du`. -/
theorem smallCurv_integral_eq :
    ∫ x in (-21 : ℝ)..21, Complex.exp (((1 / 100 * x ^ 2 / 2 : ℝ) : ℂ) * Complex.I) =
      (10 : ℝ) • ∫ u in (-(21 / 10 : ℝ))..(21 / 10), Complex.exp (((u ^ 2 / 2 : ℝ) : ℂ) * Complex.I) := by
  have h := intervalIntegral.smul_integral_comp_mul_left (a := (-21 : ℝ)) (b := 21)
    (fun u : ℝ => Complex.exp (((u ^ 2 / 2 : ℝ) : ℂ) * Complex.I)) (1 / 10)
  have e1 : (1 / 10 : ℝ) * (-21) = -(21 / 10) := by norm_num
  have e2 : (1 / 10 : ℝ) * 21 = 21 / 10 := by norm_num
  rw [e1, e2] at h
  rw [← h, smul_smul]
  have hf : ∀ x : ℝ, ((1 / 10 * x) ^ 2 / 2 : ℝ) = 1 / 100 * x ^ 2 / 2 := fun x => by ring
  simp only [hf]
  norm_num

/-- **A1. Certified value at small curvature.** `33 ≤ |∫_{−21}^{21} e^{i(1/100)x²/2} dx|` (true value `33.6141…`;
ten times `quad_norm_lower`). -/
theorem smallCurv_norm_lower :
    330 / 10 ≤ ‖∫ x in (-21 : ℝ)..21, Complex.exp (((1 / 100 * x ^ 2 / 2 : ℝ) : ℂ) * Complex.I)‖ := by
  rw [smallCurv_integral_eq, norm_smul, Real.norm_eq_abs, abs_of_pos (by norm_num : (0 : ℝ) < 10)]
  linarith [quad_norm_lower]

/-- `(x^r)^n = x^{rn}` for `x ≥ 0` and a natural number `n`. -/
theorem rpow_pow_nat {x : ℝ} (hx : 0 ≤ x) (r : ℝ) (n : ℕ) : (x ^ r) ^ n = x ^ (r * (n : ℝ)) := by
  rw [← Real.rpow_natCast, ← Real.rpow_mul hx]

/-- `1/4 ≤ (1/100)^{1/4}` (it is `0.31622…`; its fourth power is `1/100 ≥ 1/256`). -/
theorem rpow_hundredth_quarter_ge : 1 / 4 ≤ (1 / 100 : ℝ) ^ ((1 : ℝ) / 4) := by
  have h4 : ((1 / 100 : ℝ) ^ ((1 : ℝ) / 4)) ^ 4 = 1 / 100 := by
    rw [rpow_pow_nat (by norm_num), show ((1 : ℝ) / 4) * ((4 : ℕ) : ℝ) = 1 by norm_num, Real.rpow_one]
  by_contra hlt
  rw [not_le] at hlt
  have h5 := pow_le_pow_left₀ (Real.rpow_nonneg (by norm_num : (0 : ℝ) ≤ 1 / 100) ((1 : ℝ) / 4)) hlt.le 4
  rw [h4] at h5
  norm_num at h5

/-- (Witness file only: it refutes a planted statement and proves nothing of the batch.) `A(w) ≤ 2B` when
`2w ≤ B²`, `B ≥ 0`: the trivial bound `A(w) ≤ 2 √(2w)` from `|sin| ≤ 1`. -/
theorem vdc2Profile_le_of_le (w B : ℝ) (hB : 0 ≤ B) (hw : 2 * w ≤ B ^ 2) : vdc2Profile w ≤ 2 * B := by
  have h := intervalIntegral.norm_integral_le_of_norm_le_const (a := 0) (b := Real.sqrt (2 * w)) (C := 1)
    (f := fun u : ℝ => Real.sin (w - u ^ 2 / 2)) (fun u _ => by simpa using Real.abs_sin_le_one _)
  rw [sub_zero, abs_of_nonneg (Real.sqrt_nonneg _), one_mul, Real.norm_eq_abs] at h
  have h1 := le_abs_self (∫ u in (0 : ℝ)..Real.sqrt (2 * w), Real.sin (w - u ^ 2 / 2))
  have h2 : Real.sqrt (2 * w) ≤ B := (Real.sqrt_le_left hB).2 hw
  unfold vdc2Profile
  linarith

/-- (Witness file only.) `A(w) ≤ 26/5` for `w ≤ π` (`2π < 6.3 ≤ (13/5)²`). -/
theorem vdc2Profile_le_pi (w : ℝ) (hw : w ≤ Real.pi) : vdc2Profile w ≤ 26 / 5 := by

def P_SecondDerivSharpStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b lam : ℝ), a ≤ b → 0 < lam → HasDerivOn f f' a b → DerivIncreasesAtRate f' a b lam →
    ∃ w ∈ Icc (0 : ℝ) Real.pi,
      ‖∫ x in a..b, Complex.exp (((f x : ℝ) : ℂ) * Complex.I)‖ ≤ vdc2Profile w / lam ^ ((1 : ℝ) / 4)

theorem witness_not_SecondDerivSharpStatement : ¬ P_SecondDerivSharpStatement := by
  intro h
  obtain ⟨w, hw, hle⟩ := h (fun x => 1 / 100 * x ^ 2 / 2) (fun x => 1 / 100 * x) (-21) 21 (1 / 100) (by norm_num)
    (by norm_num) (scaled_hasDerivOn _ _ _) (scaled_increases _ _ _)
  have hz := rpow_hundredth_quarter_ge
  rw [le_div_iff₀ (by linarith)] at hle
  have h3 := mul_le_mul_of_nonneg_right smallCurv_norm_lower
    (by linarith : (0 : ℝ) ≤ (1 / 100 : ℝ) ^ ((1 : ℝ) / 4))
