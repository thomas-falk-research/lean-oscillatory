"""Second run of planted copies of the batch-VC2 statements (mutations found by the blind pre-proof review).

As the first run (docs/replay/plants/2026-11-16-VC2/plants_vc2.py.txt) and the second runs of batches FR and HW. The
batch is now a frozen module (OscInt/Statements/SecondDerivSharp.lean) and the first sanity layer is compiled
(OscInt/Front/SecondDerivSharpSanity.lean); the second layer (SecondDerivSharpSanity2.lean) is not a module of the
library yet. Every generated file therefore imports OscInt.Front.SecondDerivSharpSanity only and contains, in the
namespace of the second layer: the unconditional data lemmas of the second layer that the file needs (copied verbatim
with their docstrings, selected by name closure), then, under the prefix P_, a copy of ONE statement constant of the
statement module (for the plant M1: of DerivIncreasesAtRate, of VC2-S, which uses it, and of the three rate lemmas of
the two layers) and copies of the scripts that go through it: every sanity theorem of both layers that takes the
statement constant as its hypothesis. Exactly one textual mutation is applied to one copied definition. The unplanted
copy (ok-) must compile. Files with identical text (the ok- copies of one statement) are compiled once.
Witness files (docs/lessons.md, rule 9): a compiled proof of the negation of the planted statement from unconditional
data lemmas of the two layers. Four witnesses (E2, E4, H1, H2) also use the trivial bound A(w) <= 2 sqrt(2w), which
holds for every w: as in the first run it is proved in those witness files only (WITNESS_ONLY below), where it refutes
a planted statement; it is not part of a sanity layer. No file proves, or uses a proof of, a statement of the batch.
datum-M1: a compiled proof, on the planted definition of M1, that the quartic datum of the second layer does not
satisfy it and that the parabola does (the rejection of the copied rate lemmas alone would not tell these apart).
Verdicts: DETECTED (witness compiled), NOT DETECTED (no witness; "script rejected" alone is recorded and is not a
detection), TRUE WEAKENING (implied by the frozen statement).
Usage: plants_vc2b.py REPO SANITY2_FILE OUTDIR SCRATCHDIR [plant ids]
"""

DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9']+)")
FIRST_LAYER = {m.group(2) for m in map(DECL.match, SAN1) if m}


def block(lines, name):
    """The declaration `name` without its docstring, up to the next blank line; None if absent."""
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    return None


def script(name):
    t = block(SAN2, name)
    return t if t is not None else block(SAN1, name)


def data_blocks():
    """All declarations of the second layer with docstrings, in file order: name -> (index, text)."""
    res, cur = {}, []
    for l in SAN2[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()
THROUGH = {t for ts in THMS.values() for t in ts}
for n in THROUGH | set(RATE_DATA):
    assert (n in DATA) != (n in FIRST_LAYER), n
for n, (_, t) in DATA.items():  # a declaration of the second layer names a statement constant iff it is a script
    assert (n in THROUGH) == any(s in t for s in (S, A, N, K)), n


def closure(text, renamed):
    """Unconditional declarations of the second layer used by `text`, transitively, in file order."""
    need, todo = set(), [text]
    while todo:
        t = todo.pop()
        for w in set(re.findall(r"[A-Za-z_][A-Za-z_0-9']*", t)):
            if w in DATA and w not in need and w not in renamed and w not in THROUGH:
                need.add(w)
                todo.append(DATA[w][1])
    return [DATA[n][1] for n in sorted(need, key=lambda n: DATA[n][0])]

def make(stmt, mut, planted, witness=None, wonly=False):
    """stmt: the statement constant; mut = (definition to mutate, old, new, count); witness: None, a proof term of
    `¬ P_<stmt>`, or ('raw', text) for the datum file of M1 (theorems about the planted definition)."""
    target, old, new, count = mut
    in_def = target == RATE
    raw = isinstance(witness, tuple)
    defs = [target] if (target == stmt or raw) else [target, stmt]
    parts = []
    for d in defs:
        t = block(STMT, d)
        assert t is not None, d
        if planted and d == target:
            assert t.count(old) == count, (d, old, t.count(old))
            t = t.replace(old, new)
        parts.append(t)
    renamed = list(defs)
    if raw:
        parts.append(witness[1])
    else:
        if in_def:
            parts += [script(th) for th in RATE_DATA]
            renamed += RATE_DATA
        if witness is None:
            parts += [script(th) for th in THMS[stmt]]
            renamed += THMS[stmt]
        else:
            parts.append(f'theorem witness_not_{stmt} : ¬ {stmt} := {witness}')
    text = '\n\n'.join(parts)
    for n in sorted(renamed, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r"(?![\w'])", 'P_' + n, text)
    extra = [WITNESS_ONLY] if wonly else []
    data = closure(text + '\n'.join(extra), set(renamed))
    return HEADER + '\n\n'.join(data + extra + [text]) + FOOTER


CACHE = {}


def run(tag, text):
    open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
    if text in CACHE:  # identical text (an ok- copy of the same statement): compiled once, output empty
        out, rc = CACHE[text]
    else:
        src = os.path.join(SCR, tag + '.lean')
        open(src, 'w').write(text)
        r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                           cwd=REPO, capture_output=True, text=True)
        out, rc = r.stdout + r.stderr, r.returncode
        if rc == 0 and out == '':
            CACHE[text] = (out, rc)
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    # the copied declarations in which an error is reported, in file order
    lines, failing = text.split('\n'), []
    for e in errs:
        m = re.search(r'\.lean:(\d+):\d+: error', e)
        if not m:
            continue
        i = int(m.group(1)) - 1
        while i >= 0 and not DECL.match(lines[i]):
            i -= 1
        name = DECL.match(lines[i]).group(2) if i >= 0 else '?'
        if name not in failing:
            failing.append(name)
    return ('accepted' if rc == 0 and not errs else 'rejected'), errs, failing
