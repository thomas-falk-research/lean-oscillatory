/-- **Large curvature, change of variable `x = u/10`.**
`∫_{−21/100}^{21/100} e^{i·100x²/2} dx = (1/10) ∫_{−21/10}^{21/10} e^{iu²/2} du`. -/
theorem largeCurv_integral_eq :
    ∫ x in (-(21 / 100 : ℝ))..(21 / 100), Complex.exp (((100 * x ^ 2 / 2 : ℝ) : ℂ) * Complex.I) =
      (1 / 10 : ℝ) • ∫ u in (-(21 / 10 : ℝ))..(21 / 10), Complex.exp (((u ^ 2 / 2 : ℝ) : ℂ) * Complex.I) := by
  have h := intervalIntegral.smul_integral_comp_mul_left (a := (-(21 / 100) : ℝ)) (b := 21 / 100)
    (fun u : ℝ => Complex.exp (((u ^ 2 / 2 : ℝ) : ℂ) * Complex.I)) 10
  have e1 : (10 : ℝ) * (-(21 / 100)) = -(21 / 10) := by norm_num
  have e2 : (10 : ℝ) * (21 / 100) = 21 / 10 := by norm_num
  rw [e1, e2] at h
  rw [← h, smul_smul]
  have hf : ∀ x : ℝ, ((10 * x) ^ 2 / 2 : ℝ) = 100 * x ^ 2 / 2 := fun x => by ring
  simp only [hf]
  norm_num

/-- **A2. Certified value at large curvature, at the extremal width.**
`33/100 ≤ |∫_{−21/100}^{21/100} e^{i·100x²/2} dx|` (true value `0.336141…`; a tenth of `quad_norm_lower`). -/
theorem largeCurv_norm_lower :
    33 / 100 ≤
      ‖∫ x in (-(21 / 100 : ℝ))..(21 / 100), Complex.exp (((100 * x ^ 2 / 2 : ℝ) : ℂ) * Complex.I)‖ := by
  rw [largeCurv_integral_eq, norm_smul, Real.norm_eq_abs, abs_of_pos (by norm_num : (0 : ℝ) < 1 / 10)]
  linarith [quad_norm_lower]

/-- `(x^r)^n = x^{rn}` for `x ≥ 0` and a natural number `n`. -/
theorem rpow_pow_nat {x : ℝ} (hx : 0 ≤ x) (r : ℝ) (n : ℕ) : (x ^ r) ^ n = x ^ (r * (n : ℝ)) := by
  rw [← Real.rpow_natCast, ← Real.rpow_mul hx]

/-- `520/33 < 100^{3/5}` (it is `15.8489…`; its fifth power is `10⁶ > (520/33)⁵ = 971509.4…`). -/
theorem rpow_hundred_three_fifths_gt : 520 / 33 < (100 : ℝ) ^ ((3 : ℝ) / 5) := by
  have h5 : ((100 : ℝ) ^ ((3 : ℝ) / 5)) ^ 5 = 1000000 := by
    rw [rpow_pow_nat (by norm_num), show ((3 : ℝ) / 5) * ((5 : ℕ) : ℝ) = ((3 : ℕ) : ℝ) by norm_num,
      Real.rpow_natCast]
    norm_num
  by_contra hle
  rw [not_lt] at hle
  have h6 := pow_le_pow_left₀ (Real.rpow_nonneg (by norm_num : (0 : ℝ) ≤ 100) ((3 : ℝ) / 5)) hle 5
  rw [h5] at h6
  norm_num at h6

/-- (Witness file only: it refutes a planted statement and proves nothing of the batch.) `A(w) ≤ 2B` when
`2w ≤ B²`, `B ≥ 0`: the trivial bound `A(w) ≤ 2 √(2w)` from `|sin| ≤ 1`. -/
theorem vdc2Profile_le_of_le (w B : ℝ) (hB : 0 ≤ B) (hw : 2 * w ≤ B ^ 2) : vdc2Profile w ≤ 2 * B := by
  have h := intervalIntegral.norm_integral_le_of_norm_le_const (a := 0) (b := Real.sqrt (2 * w)) (C := 1)
    (f := fun u : ℝ => Real.sin (w - u ^ 2 / 2)) (fun u _ => by simpa using Real.abs_sin_le_one _)
  rw [sub_zero, abs_of_nonneg (Real.sqrt_nonneg _), one_mul, Real.norm_eq_abs] at h
  have h1 := le_abs_self (∫ u in (0 : ℝ)..Real.sqrt (2 * w), Real.sin (w - u ^ 2 / 2))
  have h2 : Real.sqrt (2 * w) ≤ B := (Real.sqrt_le_left hB).2 hw
  unfold vdc2Profile
  linarith

/-- (Witness file only.) `A(w) ≤ 26/5` for `w ≤ π` (`2π < 6.3 ≤ (13/5)²`). -/
theorem vdc2Profile_le_pi (w : ℝ) (hw : w ≤ Real.pi) : vdc2Profile w ≤ 26 / 5 := by

  linarith

def P_SecondDerivSharpStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b lam : ℝ), a ≤ b → 0 < lam → HasDerivOn f f' a b → DerivIncreasesAtRate f' a b lam →
    ∃ w ∈ Icc (0 : ℝ) Real.pi,
      ‖∫ x in a..b, Complex.exp (((f x : ℝ) : ℂ) * Complex.I)‖ ≤ vdc2Profile w / lam ^ ((3 : ℝ) / 5)

theorem witness_not_SecondDerivSharpStatement : ¬ P_SecondDerivSharpStatement := by
  intro h
  obtain ⟨w, hw, hle⟩ := h (fun x => 100 * x ^ 2 / 2) (fun x => 100 * x) (-(21 / 100)) (21 / 100) 100 (by norm_num)
    (by norm_num) (scaled_hasDerivOn _ _ _) (scaled_increases _ _ _)
  have hz := rpow_hundred_three_fifths_gt
  rw [le_div_iff₀ (by linarith)] at hle
  have h3 := mul_le_mul_of_nonneg_right largeCurv_norm_lower
    (by linarith : (0 : ℝ) ≤ (100 : ℝ) ^ ((3 : ℝ) / 5))
  linarith [vdc2Profile_le_pi w hw.2]
