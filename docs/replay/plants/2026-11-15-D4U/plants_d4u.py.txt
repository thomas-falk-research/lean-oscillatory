#!/usr/bin/env python3
"""Generate and compile planted copies of the batch-D4U statements against copies of their sanity theorems.

As the batch-LS generator: for each plant, copy the named def (from OscInt/Statements/D4Uniform.lean) and sanity
theorems (from OscInt/Front/D4UniformSanity.lean) under the prefix P_, apply exactly one textual mutation to the copied
def, compile with `lake env lean -DautoImplicit=false`. The unplanted copy must compile. Witness files (lessons 44):
a theorem about the planted def, proved from the proved S4.2 (`OscInt.d4Limit`) and S4.3 (`OscInt.d4ConstantPos`)
or from the definitions, showing that the planted def is false (or, for a planted hypothesis, that it holds where the
sanity layer's bite says the original fails); it must compile. A plant without a witness that the sanity layer
rejects is recorded as "rejected (recorded)": the sanity layer sees the change, but no semantic evidence is given.
Usage: plants_d4u.py REPO OUTDIR [plant ids]
"""
import os, re, subprocess, sys

REPO, OUT = sys.argv[1], sys.argv[2]
os.makedirs(OUT, exist_ok=True)
STMT = open(os.path.join(REPO, 'OscInt/Statements/D4Uniform.lean')).read().split('\n')
SAN = open(os.path.join(REPO, 'OscInt/Front/D4UniformSanity.lean')).read().split('\n')
TU = 'D4LimitUniformStatement'
TO = 'D4LimitOffsetStatement'
ATB = 'AmpTotallyBounded'
DP = 'D4LimitPhase'


def block(lines, kind, name):
    pat = re.compile(r'^(noncomputable )?' + kind + r' ' + re.escape(name) + r'\b')
    for i, l in enumerate(lines):
        if pat.match(l):
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {kind} {name}')


LIM = '(d4Constant (cubicCoeffs φ) : ℂ) * A 0'

# a contradiction with the proved limit, for the bump of radius r ≤ min δ δ2 (the planted statement's δ, S4.2's δ2)
TRUE_LIMIT = (
    '  obtain ⟨δ2, hδ2, hT⟩ := W.true_limit\n'
    '  have hr : 0 < min (min δ δ2) (1 / 2) := lt_min (lt_min hδ hδ2) (by norm_num)\n'
    '  have hlim := hT _ hr ((min_le_left _ _).trans (min_le_right _ _))\n'
    '  obtain ⟨M, hM⟩ := bumpC_ampBound _ hr\n'
    '  have hM2 : AmpBound 2 (Metric.closedBall 0 δ) (bumpC _ hr) M := W.ampBound_ball hM\n'
    '    ((min_le_left _ _).trans (min_le_left _ _))\n')

W_U_ZERO = (f'¬ P_{TU}', 'by\n  intro h\n  obtain ⟨δ, hδ, hU⟩ := h _ phase_generic\n' + TRUE_LIMIT +
            '  refine W.c_ne (tendsto_nhds_unique hlim ?_)\n  rw [Metric.tendsto_atTop]\n  intro ε hε\n'
            '  obtain ⟨T, hT2⟩ := hU _ (W.singleton_tb hM2) ε hε\n'
            '  exact ⟨T, fun t ht => by simpa [dist_eq_norm] using hT2 t ht _ rfl⟩')
W_U_POINT = (f'¬ P_{TU}', 'by\n  intro h\n  obtain ⟨δ, hδ, hU⟩ := h _ phase_generic\n' + TRUE_LIMIT +
             '  have h10 := W.bump_one_zero hr (min_le_right _ _)\n'
             '  refine W.c_ne (tendsto_nhds_unique hlim ?_)\n  rw [Metric.tendsto_atTop]\n  intro ε hε\n'
             '  obtain ⟨T, hT2⟩ := hU _ (W.singleton_tb hM2) ε hε\n'
             '  refine ⟨T, fun t ht => ?_⟩\n  have := hT2 t ht _ rfl\n'
             '  rw [h10, mul_zero, sub_zero] at this\n  simpa [dist_eq_norm] using this')
W_O_ZERO = (f'¬ P_{TO}', 'by\n  intro h\n  obtain ⟨δ, hδ, hO⟩ := h _ phase_generic\n' + TRUE_LIMIT +
            '  refine W.c_ne (tendsto_nhds_unique hlim ?_)\n  rw [Metric.tendsto_atTop]\n  intro ε hε\n'
            '  obtain ⟨T, hT2⟩ := hO _ M hM2 0 ε hε\n'
            '  refine ⟨T, fun t ht => ?_⟩\n  have := hT2 t ht 0 (by simp)\n'
            '  simpa [dist_eq_norm, W.lin_zero] using this')
W_O_RATE = (f'¬ P_{TO}', 'by\n  intro h\n  obtain ⟨δ, hδ, hO⟩ := h _ phase_generic\n' + TRUE_LIMIT +
            '  have hhalf : Tendsto (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) *\n'
            '      oscInt2 (fun κ => t * genCubic 1 2 3 5 κ) (bumpC _ hr)) atTop (𝓝 (d4Constant (1, 2, 3, 5) : ℂ)) := by\n'
            '    rw [Metric.tendsto_atTop]\n    intro ε hε\n    obtain ⟨T, hT2⟩ := hO _ M hM2 0 ε hε\n'
            '    refine ⟨T, fun t ht => ?_⟩\n    have := hT2 t ht 0 (by simp)\n'
            '    simpa [dist_eq_norm, W.lin_zero, coeffs_generic, bumpC_zero] using this\n'
            '  exact W.c_ne (W.rate_contradiction hlim hhalf)')
W_ATB = (f'¬ P_{ATB} (Metric.closedBall 0 1) (scaleFamily (bumpC 1 one_pos))',
         'by\n  rintro ⟨-, h⟩\n  obtain ⟨F, hF, hSF, -⟩ := h 1 one_pos\n'
         '  exact W.scale_infinite _ (bumpC_zero 1 one_pos) (hF.subset hSF)')
W_DP = (f'P_{DP} (genCubic 1 0 0 0)',
        'by\n  obtain ⟨h1, h2, h3, h4, -⟩ := OscInt.API.Sanity.Aux.hyps_genCubic 1 0 0 0\n'
        '  exact ⟨h1, h2, h3, h4, trivial⟩')

PLANTS = [
    # (id, def, sanity theorems, (old, new), want, witness)
    ('U-limit-zero', TU, ['uniform_sanity'], (LIM, '(0 : ℂ)'), None, W_U_ZERO),
    ('U-limit-point', TU, ['uniform_sanity'], (LIM, '(d4Constant (cubicCoeffs φ) : ℂ) * A (1, 0)'), None, W_U_POINT),
    ('U-not-uniform', TU, ['uniform_sanity'],
     ('∃ T : ℝ, ∀ t : ℝ, T ≤ t → ∀ A ∈ S,', '∀ A ∈ S, ∃ T : ℝ, ∀ t : ℝ, T ≤ t →'), None, None),
    ('ATB-finite', ATB, ['scaleFamily_totallyBounded'], ('B.Finite ∧ B ⊆ S ∧', 'B.Finite ∧ S ⊆ B ∧'), None, W_ATB),
    ('O-limit-zero', TO, ['offset_sanity'], (LIM, '(0 : ℂ)'), None, W_O_ZERO),
    ('O-rate', TO, ['offset_sanity'], ('t ^ ((2 : ℝ) / 3)', 't ^ ((1 : ℝ) / 2)'), None, W_O_RATE),
    ('O-moving', TO, ['offset_sanity'], ('t * φ κ + lin y κ', 't * φ κ + t * lin y κ'), None, None),
    ('O-unbounded', TO, ['offset_sanity'], ('∀ y : ℝ × ℝ, ‖y‖ ≤ R →', '∀ y : ℝ × ℝ,'), None, None),
    ('P-no-disc', DP, ['phase_bite'], ('disc (cubicCoeffs φ) ≠ 0', 'True'), None, W_DP),
]

HEADER = """import OscInt.Front.D4UniformSanity
import OscInt.D4.LimitFinal
import OscInt.D4.ConstantPosFinal

namespace OscInt.API.D4UniformSanity

open Filter Topology OscInt.Statements OscInt.D4

theorem W.c_ne : (d4Constant (1, 2, 3, 5) : ℂ) ≠ 0 := by
  have := (OscInt.d4ConstantPos.2.2 (1, 2, 3, 5) (by norm_num [disc])).2
  exact_mod_cast this.ne'

theorem W.ampBound_ball {r δ : ℝ} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 (Metric.closedBall 0 r) A M) (hr : r ≤ δ) :
    AmpBound 2 (Metric.closedBall 0 δ) A M :=
  ⟨h.1, h.2.1.trans (Metric.closedBall_subset_closedBall hr), h.2.2⟩

theorem W.true_limit : ∃ δ2 > 0, ∀ (r : ℝ) (hr : 0 < r), r ≤ δ2 →
    Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * oscInt2 (fun κ => t * genCubic 1 2 3 5 κ) (bumpC r hr))
      atTop (𝓝 (d4Constant (1, 2, 3, 5) : ℂ)) := by
  obtain ⟨h1, h2, h3, h4, h5⟩ := phase_generic
  obtain ⟨δ2, hδ2, hL⟩ := OscInt.d4Limit _ h1 h2 h3 h4 h5
  refine ⟨δ2, hδ2, fun r hr hrd => ?_⟩
  obtain ⟨M, hM⟩ := bumpC_ampBound r hr
  have := hL _ M (W.ampBound_ball hM hrd)
  rwa [coeffs_generic, bumpC_zero, mul_one] at this

theorem W.singleton_tb {K : Set (ℝ × ℝ)} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 K A M) :
    AmpTotallyBounded K {A} := by
  refine ⟨fun B hB => ⟨M, by rw [Set.mem_singleton_iff.mp hB]; exact h⟩,
    fun η hη => ⟨{A}, Set.finite_singleton A, le_rfl, fun B hB => ⟨A, rfl, ?_⟩⟩⟩
  rw [Set.mem_singleton_iff.mp hB, sub_self]
  exact ⟨contDiff_const, by simp, fun j _ k => by simp [hη.le]⟩

theorem W.bump_one_zero {r : ℝ} (hr : 0 < r) (hr2 : r ≤ 1 / 2) : bumpC r hr (1, 0) = 0 := by
  apply image_eq_zero_of_notMem_tsupport
  intro hm
  have := Metric.mem_closedBall.mp (bumpC_tsupport r hr hm)
  rw [dist_zero_right, Prod.norm_def] at this
  simp at this
  linarith

theorem W.lin_zero (t : ℝ) (φ : ℝ × ℝ → ℝ) : (fun κ => t * φ κ + lin 0 κ) = fun κ => t * φ κ := by
  funext κ; simp [lin]

theorem W.rate_contradiction {f : ℝ → ℂ} {c : ℂ}
    (h23 : Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t) atTop (𝓝 c))
    (h12 : Tendsto (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) atTop (𝓝 c)) : c = 0 := by
  have hneg : Tendsto (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ)) atTop (𝓝 0) := by
    exact (Complex.continuous_ofReal.tendsto 0).comp (tendsto_rpow_neg_atTop (by norm_num : (0 : ℝ) < 1 / 6))
  have hprod := hneg.mul h23
  rw [zero_mul] at hprod
  have heq : (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ) * (((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t)) =ᶠ[atTop]
      (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) := by
    filter_upwards [eventually_gt_atTop (0 : ℝ)] with t ht
    rw [← mul_assoc, ← Complex.ofReal_mul, ← Real.rpow_add ht]
    norm_num
  exact tendsto_nhds_unique h12 (hprod.congr' heq)

theorem W.scale_infinite (B : ℝ × ℝ → ℂ) (hB0 : B 0 = 1) : (scaleFamily B).Infinite := by
  have : scaleFamily B = (fun c : ℝ => (c : ℂ) • B) '' Set.Icc 1 2 := by
    ext A
    simp only [scaleFamily, Set.mem_ofPred_eq, Set.mem_image]
    constructor <;> rintro ⟨c, hc, h⟩ <;> exact ⟨c, hc, h.symm⟩
  rw [this, Set.infinite_image_iff]
  · exact Set.Icc_infinite (by norm_num)
  · intro c _ d _ hcd
    have := congrFun hcd 0
    simpa [hB0] using this
"""


def make(d, thms, mut, planted, witness=None):
    t = block(STMT, 'def', d)
    if planted:
        assert t.count(mut[0]) == 1, (d, mut[0])
        t = t.replace(mut[0], mut[1])
    parts = [t]
    if witness is None:
        parts += [block(SAN, 'theorem', th) for th in thms]
    else:
        parts.append(f'theorem witness_not_{d} : {witness[0]} :=\n  {witness[1]}')
    text = '\n\n'.join(parts)
    for n in sorted([d] + thms, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    text = text.replace('P_P_', 'P_')
    return HEADER + '\n' + text + '\n\nend OscInt.API.D4UniformSanity\n'


def run(tag, text):
    src = os.path.join(OUT, tag + '.lean')
    open(src, 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs


summary = []
ONLY = sys.argv[3:]
for pid, d, thms, mut, want, witness in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    jobs = [('ok-' + pid, make(d, thms, mut, False), 'accepted'),
            ('plant-' + pid, make(d, thms, mut, True), want)]
    if witness:
        jobs.append(('witness-' + pid, make(d, thms, mut, True, witness), 'accepted'))
    for tag, text, w in jobs:
        verdict, errs = run(tag, text)
        line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
        if errs:
            line += '\n    first error: ' + errs[0].split('/')[-1][:160]
        print(line, flush=True)
        summary.append(line)
open(os.path.join(OUT, 'SUMMARY.txt'), 'w').write('\n'.join(summary) + '\n')
