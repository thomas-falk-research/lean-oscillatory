theorem W.ampBound_ball {r δ : ℝ} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 (Metric.closedBall 0 r) A M) (hr : r ≤ δ) :
    AmpBound 2 (Metric.closedBall 0 δ) A M :=
  ⟨h.1, h.2.1.trans (Metric.closedBall_subset_closedBall hr), h.2.2⟩

theorem W.true_limit : ∃ δ2 > 0, ∀ (r : ℝ) (hr : 0 < r), r ≤ δ2 →
    Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * oscInt2 (fun κ => t * genCubic 1 2 3 5 κ) (bumpC r hr))
      atTop (𝓝 (d4Constant (1, 2, 3, 5) : ℂ)) := by
  obtain ⟨h1, h2, h3, h4, h5⟩ := phase_generic
  obtain ⟨δ2, hδ2, hL⟩ := OscInt.d4Limit _ h1 h2 h3 h4 h5
  refine ⟨δ2, hδ2, fun r hr hrd => ?_⟩
  obtain ⟨M, hM⟩ := bumpC_ampBound r hr
  have := hL _ M (W.ampBound_ball hM hrd)
  rwa [coeffs_generic, bumpC_zero, mul_one] at this

theorem W.singleton_tb {K : Set (ℝ × ℝ)} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 K A M) :
    AmpTotallyBounded K {A} := by
  refine ⟨fun B hB => ⟨M, by rw [Set.mem_singleton_iff.mp hB]; exact h⟩,
    fun η hη => ⟨{A}, Set.finite_singleton A, le_rfl, fun B hB => ⟨A, rfl, ?_⟩⟩⟩
  rw [Set.mem_singleton_iff.mp hB, sub_self]
  exact ⟨contDiff_const, by simp, fun j _ k => by simp [hη.le]⟩

theorem W.bump_one_zero {r : ℝ} (hr : 0 < r) (hr2 : r ≤ 1 / 2) : bumpC r hr (1, 0) = 0 := by
  apply image_eq_zero_of_notMem_tsupport
  intro hm
  have := Metric.mem_closedBall.mp (bumpC_tsupport r hr hm)
  rw [dist_zero_right, Prod.norm_def] at this
  simp at this
  linarith

theorem W.lin_zero (t : ℝ) (φ : ℝ × ℝ → ℝ) : (fun κ => t * φ κ + lin 0 κ) = fun κ => t * φ κ := by
  funext κ; simp [lin]

theorem W.rate_contradiction {f : ℝ → ℂ} {c : ℂ}
    (h23 : Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t) atTop (𝓝 c))
    (h12 : Tendsto (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) atTop (𝓝 c)) : c = 0 := by
  have hneg : Tendsto (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ)) atTop (𝓝 0) := by
    exact (Complex.continuous_ofReal.tendsto 0).comp (tendsto_rpow_neg_atTop (by norm_num : (0 : ℝ) < 1 / 6))
  have hprod := hneg.mul h23
  rw [zero_mul] at hprod
  have heq : (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ) * (((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t)) =ᶠ[atTop]
      (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) := by
    filter_upwards [eventually_gt_atTop (0 : ℝ)] with t ht
    rw [← mul_assoc, ← Complex.ofReal_mul, ← Real.rpow_add ht]
    norm_num
  exact tendsto_nhds_unique h12 (hprod.congr' heq)

theorem W.oscInt2_smul (Q : ℝ × ℝ → ℝ) (c : ℂ) (A : ℝ × ℝ → ℂ) : oscInt2 Q (c • A) = c * oscInt2 Q A := by
  unfold oscInt2
  rw [← MeasureTheory.integral_const_mul]
  congr 1
  funext k
  simp only [Pi.smul_apply, smul_eq_mul]
  ring

theorem W.lin_smul (t : ℝ) (φ : ℝ × ℝ → ℝ) (v : ℝ × ℝ) :
    (fun κ => t * φ κ + lin (t • v) κ) = fun κ => t * (φ κ + lin v κ) := by
  funext κ; simp only [lin, Prod.smul_fst, Prod.smul_snd, smul_eq_mul]; ring

theorem W.lin_mul (t : ℝ) (φ : ℝ × ℝ → ℝ) (v : ℝ × ℝ) :
    (fun κ => t * φ κ + t * lin v κ) = fun κ => t * (φ κ + lin v κ) := by
  funext κ; ring

theorem W.grad_lower (k : ℝ × ℝ) : GradLower (genCubic 1 2 3 5) ((100 : ℝ), (0 : ℝ)) k 1 := by
  unfold GradLower
  rw [fd_genCubic, fd_genCubic]
  simp only
  have ha : 0 ≤ 3 * k.1 ^ 2 + 4 * k.1 * k.2 + 3 * k.2 ^ 2 := by nlinarith [sq_nonneg (3 * k.1 + 2 * k.2), sq_nonneg k.2]
  nlinarith [sq_nonneg (2 * k.1 ^ 2 + 6 * k.1 * k.2 + 15 * k.2 ^ 2)]

theorem W.nonstat_zero : ∀ (r : ℝ) (hr : 0 < r), r ≤ 1 →
    Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) *
      oscInt2 (fun κ => t * (genCubic 1 2 3 5 κ + lin ((100 : ℝ), (0 : ℝ)) κ)) (bumpC r hr)) atTop (𝓝 0) := by
  intro r hr hr1
  obtain ⟨C, hC⟩ := OscInt.nonStationary Unit 1 (fun _ => genCubic 1 2 3 5) (Metric.closedBall 0 1) 1
    ⟨fun _ => (genCubic_contDiff 1 2 3 5).of_le (by norm_num), fun j hj =>
      ((genCubic_contDiff 1 2 3 5).continuous_iteratedFDeriv (by exact_mod_cast (hj.trans (by norm_num)))).comp
        continuous_snd⟩
    (isCompact_closedBall 0 1) one_pos
  obtain ⟨M, hM⟩ := OscInt.API.ampBoundOfContDiff 1 (Metric.closedBall 0 1) (bumpC r hr) (isCompact_closedBall 0 1)
    ((bumpC_contDiff r hr).of_le (by norm_num)) ((bumpC_tsupport r hr).trans (Metric.closedBall_subset_closedBall hr1))
  have hb : ∀ t : ℝ, 1 ≤ t → ‖((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) *
      oscInt2 (fun κ => t * (genCubic 1 2 3 5 κ + lin ((100 : ℝ), (0 : ℝ)) κ)) (bumpC r hr)‖ ≤
      C * M * t ^ (-((1 : ℝ) / 3)) := by
    intro t ht
    have h1 := hC (bumpC r hr) M () ((100 : ℝ), (0 : ℝ)) t hM (fun k _ => W.grad_lower k) ht
    have ht0 : 0 < t := by linarith
    rw [norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_of_nonneg (by positivity)]
    calc _ ≤ t ^ ((2 : ℝ) / 3) * (C * M / t ^ 1) := mul_le_mul_of_nonneg_left h1 (by positivity)
      _ = C * M * t ^ (-((1 : ℝ) / 3)) := by
          rw [pow_one, show (-((1 : ℝ) / 3)) = (2 : ℝ) / 3 - 1 by norm_num, Real.rpow_sub_one ht0.ne']
          ring
  apply squeeze_zero_norm' (eventually_atTop.2 ⟨1, hb⟩)
  have := (tendsto_rpow_neg_atTop (by norm_num : (0 : ℝ) < 1 / 3)).const_mul (C * M)
  simpa using this

theorem W.scale_infinite (B : ℝ × ℝ → ℂ) (hB0 : B 0 = 1) : (scaleFamily B).Infinite := by
  have : scaleFamily B = (fun c : ℝ => (c : ℂ) • B) '' Set.Icc 1 2 := by
    ext A
    simp only [scaleFamily, Set.mem_ofPred_eq, Set.mem_image]
    constructor <;> rintro ⟨c, hc, h⟩ <;> exact ⟨c, hc, h.symm⟩
  rw [this, Set.infinite_image_iff]
  · exact Set.Icc_infinite (by norm_num)
  · intro c _ d _ hcd
    have := congrFun hcd 0
    simpa [hB0] using this

def P_AmpTotallyBounded (K : Set (ℝ × ℝ)) (S : Set (ℝ × ℝ → ℂ)) : Prop :=
  (∀ A ∈ S, ∃ M : ℝ, AmpBound 2 K A M) ∧
    ∀ η : ℝ, 0 < η → ∃ B : Set (ℝ × ℝ → ℂ), B.Finite ∧ S ⊆ B ∧ ∀ A ∈ S, ∃ b ∈ B, AmpBound 2 K (A - b) η

theorem witness_not_AmpTotallyBounded : ¬ P_AmpTotallyBounded (Metric.closedBall 0 1) (scaleFamily (bumpC 1 one_pos)) :=
  by
  rintro ⟨-, h⟩
  obtain ⟨F, hF, hSF, -⟩ := h 1 one_pos
  exact W.scale_infinite _ (bumpC_zero 1 one_pos) (hF.subset hSF)
