#!/usr/bin/env python3
"""Generate and compile planted copies of the batch-D4U statements against copies of their sanity theorems.

As the batch-LS generator: for each plant, copy the named def (from OscInt/Statements/D4Uniform.lean) and sanity
theorems (from OscInt/Front/D4UniformSanity.lean) under the prefix P_, apply exactly one textual mutation to the copied
def, compile with `lake env lean -DautoImplicit=false`. The unplanted copy must compile. Witness files (lessons 44):
a theorem about the planted def, proved from the proved S4.2 (`OscInt.d4Limit`) and S4.3 (`OscInt.d4ConstantPos`)
or from the definitions, showing that the planted def is false (or, for a planted hypothesis, that it holds where the
sanity layer's bite says the original fails); it must compile. A plant without a witness that the sanity layer
rejects is recorded as "rejected (recorded)": the sanity layer sees the change, but no semantic evidence is given.
Usage: plants_d4u.py REPO OUTDIR [plant ids]
"""

def block(lines, kind, name):
    pat = re.compile(r'^(noncomputable )?' + kind + r' ' + re.escape(name) + r'\b')
    for i, l in enumerate(lines):
        if pat.match(l):
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {kind} {name}')

theorem W.c_ne : (d4Constant (1, 2, 3, 5) : ℂ) ≠ 0 := by

theorem W.ampBound_ball {r δ : ℝ} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 (Metric.closedBall 0 r) A M) (hr : r ≤ δ) :
    AmpBound 2 (Metric.closedBall 0 δ) A M :=

theorem W.true_limit : ∃ δ2 > 0, ∀ (r : ℝ) (hr : 0 < r), r ≤ δ2 →
    Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * oscInt2 (fun κ => t * genCubic 1 2 3 5 κ) (bumpC r hr))

theorem W.singleton_tb {K : Set (ℝ × ℝ)} {A : ℝ × ℝ → ℂ} {M : ℝ} (h : AmpBound 2 K A M) :
    AmpTotallyBounded K {A} := by
  refine ⟨fun B hB => ⟨M, by rw [Set.mem_singleton_iff.mp hB]; exact h⟩,
    fun η hη => ⟨{A}, Set.finite_singleton A, le_rfl, fun B hB => ⟨A, rfl, ?_⟩⟩⟩

theorem W.bump_one_zero {r : ℝ} (hr : 0 < r) (hr2 : r ≤ 1 / 2) : bumpC r hr (1, 0) = 0 := by
  apply image_eq_zero_of_notMem_tsupport
  intro hm

  simp at this
  linarith

theorem W.lin_zero (t : ℝ) (φ : ℝ × ℝ → ℝ) : (fun κ => t * φ κ + lin 0 κ) = fun κ => t * φ κ := by
  funext κ; simp [lin]

theorem W.rate_contradiction {f : ℝ → ℂ} {c : ℂ}
    (h23 : Tendsto (fun t : ℝ => ((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t) atTop (𝓝 c))
    (h12 : Tendsto (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) atTop (𝓝 c)) : c = 0 := by
  have hneg : Tendsto (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ)) atTop (𝓝 0) := by
    exact (Complex.continuous_ofReal.tendsto 0).comp (tendsto_rpow_neg_atTop (by norm_num : (0 : ℝ) < 1 / 6))
  have hprod := hneg.mul h23
  rw [zero_mul] at hprod
  have heq : (fun t : ℝ => ((t ^ (-((1 : ℝ) / 6)) : ℝ) : ℂ) * (((t ^ ((2 : ℝ) / 3) : ℝ) : ℂ) * f t)) =ᶠ[atTop]
      (fun t : ℝ => ((t ^ ((1 : ℝ) / 2) : ℝ) : ℂ) * f t) := by
    filter_upwards [eventually_gt_atTop (0 : ℝ)] with t ht
    rw [← mul_assoc, ← Complex.ofReal_mul, ← Real.rpow_add ht]
    norm_num
  exact tendsto_nhds_unique h12 (hprod.congr' heq)

theorem W.oscInt2_smul (Q : ℝ × ℝ → ℝ) (c : ℂ) (A : ℝ × ℝ → ℂ) : oscInt2 Q (c • A) = c * oscInt2 Q A := by
  unfold oscInt2
  rw [← MeasureTheory.integral_const_mul]
  congr 1
  funext k
  simp only [Pi.smul_apply, smul_eq_mul]
  ring

theorem W.lin_smul (t : ℝ) (φ : ℝ × ℝ → ℝ) (v : ℝ × ℝ) :
    (fun κ => t * φ κ + lin (t • v) κ) = fun κ => t * (φ κ + lin v κ) := by
  funext κ; simp only [lin, Prod.smul_fst, Prod.smul_snd, smul_eq_mul]; ring

    (fun κ => t * φ κ + t * lin v κ) = fun κ => t * (φ κ + lin v κ) := by

theorem W.scale_infinite (B : ℝ × ℝ → ℂ) (hB0 : B 0 = 1) : (scaleFamily B).Infinite := by
  have : scaleFamily B = (fun c : ℝ => (c : ℂ) • B) '' Set.Icc 1 2 := by

def make(d, thms, mut, planted, witness=None):
    t = block(STMT, 'def', d)
    if planted:
        assert t.count(mut[0]) == 1, (d, mut[0])
        t = t.replace(mut[0], mut[1])
    parts = [t]
    if witness is None:
        parts += [block(SAN, 'theorem', th) for th in thms]
    else:
        parts.append(f'theorem witness_not_{d} : {witness[0]} :=\n  {witness[1]}')
    text = '\n\n'.join(parts)
    for n in sorted([d] + thms, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    text = text.replace('P_P_', 'P_')
    return HEADER + '\n' + text + '\n\nend OscInt.API.D4UniformSanity\n'


def run(tag, text):
    src = os.path.join(OUT, tag + '.lean')
    open(src, 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs


summary = []
ONLY = sys.argv[3:]
for pid, d, thms, mut, want, witness in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    jobs = [('ok-' + pid, make(d, thms, mut, False), 'accepted'),
            ('plant-' + pid, make(d, thms, mut, True), want)]
    if witness:
        jobs.append(('witness-' + pid, make(d, thms, mut, True, witness), 'accepted'))
    for tag, text, w in jobs:
        verdict, errs = run(tag, text)
        line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
        if errs:
            line += '\n    first error: ' + errs[0].split('/')[-1][:160]
        print(line, flush=True)
        summary.append(line)
open(os.path.join(OUT, 'SUMMARY-second-run.txt'), 'w').write('\n'.join(summary) + '\n')
