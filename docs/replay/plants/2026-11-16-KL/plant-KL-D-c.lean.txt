import OscInt.Front.KusminLandauSanity
import Mathlib.Analysis.Real.Pi.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Deriv

namespace OscInt.API.KusminLandauSanity

open OscInt.Statements Set

def P_DiffsMonotoneIn (c : ℕ → ℝ) (N : ℕ) (k : ℤ) (lam : ℝ) : Prop :=
  ∀ n < N, (k : ℝ) + lam ≤ c (n + 1) - c n ∧ c (n + 1) - c n ≤ (k : ℝ) + 1 - lam

def P_KusminLandauStatement : Prop :=
  ∀ (c : ℕ → ℝ) (N : ℕ) (k : ℤ) (lam : ℝ), 0 < lam → lam ≤ 1 / 2 → P_DiffsMonotoneIn c N k lam →
    ‖expSum c N‖ ≤ Real.cos (Real.pi * lam / 2) / Real.sin (Real.pi * lam / 2)

theorem P_diffs_cThird : P_DiffsMonotoneIn cThird 2 0 (1 / 3) := by
  refine ⟨Or.inl fun n hn => ?_, fun n hn => ?_⟩
  · obtain rfl : n = 0 := by omega
    norm_num [cThird]
  · interval_cases n <;> norm_num [cThird]

theorem P_diffs_cAnti : P_DiffsMonotoneIn cAnti 2 0 (1 / 3) := by
  refine ⟨Or.inr fun n hn => ?_, fun n hn => ?_⟩
  · obtain rfl : n = 0 := by omega
    norm_num [cAnti]
  · interval_cases n <;> norm_num [cAnti]

theorem P_diffs_cShift : P_DiffsMonotoneIn cShift 2 1 (1 / 3) := by
  refine ⟨Or.inl fun n hn => ?_, fun n hn => ?_⟩
  · obtain rfl : n = 0 := by omega
    norm_num [cShift]
  · interval_cases n <;> norm_num [cShift]

theorem P_diffs_cSix : P_DiffsMonotoneIn cSix 6 0 (1 / 6) := by
  refine ⟨Or.inl fun n hn => ?_, fun n hn => ?_⟩
  · have hn' : n < 5 := by omega
    interval_cases n <;> norm_num [cSix]
  · interval_cases n <;> norm_num [cSix]

theorem P_diffs_cOne (k : ℤ) (lam : ℝ) : P_DiffsMonotoneIn cOne 0 k lam :=
  ⟨Or.inl fun _ hn => absurd hn (Nat.not_lt_zero _), fun _ hn => absurd hn (Nat.not_lt_zero _)⟩

theorem P_not_diffs_cAlt : ¬ P_DiffsMonotoneIn cAlt 4 0 (1 / 4) := by
  rintro ⟨h | h, -⟩
  · have := h 1 (by norm_num)
    norm_num [cAlt] at this
  · have := h 0 (by norm_num)
    norm_num [cAlt] at this

theorem P_not_diffs_cQuarter_third : ¬ P_DiffsMonotoneIn cQuarter 1 0 (1 / 3) := by
  rintro ⟨-, h⟩
  have := (h 0 (by norm_num)).1
  norm_num [cQuarter] at this

theorem P_not_diffs_cInt : ¬ P_DiffsMonotoneIn cInt 2 0 (1 / 2) := by
  rintro ⟨-, h⟩
  have := (h 0 (by norm_num)).2
  rw [cInt_diff] at this
  norm_num at this

theorem P_kl_third (h : P_KusminLandauStatement) : ‖expSum cThird 2‖ ≤ Real.sqrt 3 := by
  have := h cThird 2 0 (1 / 3) (by norm_num) (by norm_num) P_diffs_cThird
  rwa [cot_third] at this

theorem P_kl_anti (h : P_KusminLandauStatement) : ‖expSum cAnti 2‖ ≤ Real.sqrt 3 := by
  have := h cAnti 2 0 (1 / 3) (by norm_num) (by norm_num) P_diffs_cAnti
  rwa [cot_third] at this

theorem P_kl_shift (h : P_KusminLandauStatement) : ‖expSum cShift 2‖ ≤ Real.sqrt 3 := by
  have := h cShift 2 1 (1 / 3) (by norm_num) (by norm_num) P_diffs_cShift
  rwa [cot_third] at this

theorem P_kl_six (h : P_KusminLandauStatement) :
    Real.sqrt 13 ≤ Real.cos (Real.pi * (1 / 6) / 2) / Real.sin (Real.pi * (1 / 6) / 2) := by
  have := h cSix 6 0 (1 / 6) (by norm_num) (by norm_num) P_diffs_cSix
  rwa [norm_expSum_cSix] at this

theorem P_kl_one_half (h : P_KusminLandauStatement) :
    (1 : ℝ) ≤ Real.cos (Real.pi * (1 / 2) / 2) / Real.sin (Real.pi * (1 / 2) / 2) := by
  have := h cOne 0 0 (1 / 2) (by norm_num) le_rfl (P_diffs_cOne 0 (1 / 2))
  rwa [expSum_cOne, norm_one] at this

end OscInt.API.KusminLandauSanity
