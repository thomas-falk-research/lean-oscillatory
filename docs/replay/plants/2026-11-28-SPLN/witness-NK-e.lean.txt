/-- PLANTED copy of `StationaryLeadingSharpStatement`: the power λ^{−3/2} ↦ λ^{−5/2}. -/
def P_StationaryLeadingSharpStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ‖splRemainder g lam‖ ≤ (360987 / 500000 : ℝ) * (∫ t, ‖g₃ t‖) * lam ^ (-(5 : ℝ) / 2)

/-- Witness: at `G_1`, `λ = 100` the planted bound is `(360987/500000)(2 + 8e^{−3/2}) · 10⁻⁵ ≤ 2.8·10⁻⁵`, but
`|R| ≥ 1.24·10⁻³` (`splRemainder_gauss_one_hundred_lower`). -/
theorem witness_NK_e : ¬ P_StationaryLeadingSharpStatement := by
  intro h
  have h1 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  rw [integral_norm_gauss3 1 one_pos, spln_rpow_hundred_five_halves, one_pow, one_mul] at h1
  have h2 := splRemainder_gauss_one_hundred_lower
  have h3 := normThree_le
  nlinarith

#print axioms witness_NK_e
