open OscInt.Statements OscInt.API.KusminLandauSanity Set

def P_KusminLandauDerivStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b : ℝ) (k : ℤ) (lam : ℝ), 0 < lam → lam ≤ 1 / 2 → HasDerivOn f f' a b →
    (MonotoneOn f' (Ioo a b) ∨ AntitoneOn f' (Ioo a b)) →
    (∀ x ∈ Ioo a b, (k : ℝ) + lam ≤ f' x ∧ f' x ≤ (k : ℝ) + 1 - lam) →
    ‖∑ n ∈ Finset.Icc ⌊a⌋ ⌈b⌉, Complex.exp (↑(2 * Real.pi * f (n : ℝ)) * Complex.I)‖ ≤
      Real.cos (Real.pi * lam / 2) / Real.sin (Real.pi * lam / 2)

theorem witness_not_P_KusminLandauDerivStatement : ¬ P_KusminLandauDerivStatement := by
  intro h
  have h1 := h (fun _ => 0) (fun _ => 0) (1 / 2) (1 / 2) 0 (1 / 2) (by norm_num) le_rfl
    ⟨continuousOn_const, fun x _ => hasDerivAt_const x 0⟩ (Or.inl monotoneOn_const)
    (fun x hx => absurd (hx.1.trans hx.2) (lt_irrefl _))
  have hI : Finset.Icc ⌊(1 / 2 : ℝ)⌋ ⌈(1 / 2 : ℝ)⌉ = {0, 1} := by
    rw [show ⌊(1 / 2 : ℝ)⌋ = 0 from Int.floor_eq_iff.mpr ⟨by norm_num, by norm_num⟩,
      show ⌈(1 / 2 : ℝ)⌉ = 1 from Int.ceil_eq_iff.mpr ⟨by norm_num, by norm_num⟩]
    decide
  have hs : ∑ n ∈ Finset.Icc ⌊(1 / 2 : ℝ)⌋ ⌈(1 / 2 : ℝ)⌉, eTerm 0 = 2 := by
    rw [hI, Finset.sum_pair (by norm_num)]
    norm_num [eTerm_zero]
  have h2 : ‖(2 : ℂ)‖ ≤ _ := hs ▸ h1
  rw [cot_half] at h2
  norm_num at h2

end OscInt.API.KusminLandauSanity2
