def block(lines, kind, name):
    pat = re.compile(r'^(noncomputable )?' + kind + r' ' + re.escape(name) + r'\b')
    hits = [i for i, l in enumerate(lines) if pat.match(l)]
    if len(hits) != 1:
        raise SystemExit(f'{len(hits)} matches: {kind} {name}')
    i = j = hits[0]
    while j < len(lines) and lines[j].strip() != '':
        j += 1
    return '\n'.join(lines[i:j])

HEADER = """import OscInt.Front.KusminLandauSanity2
import Mathlib.Analysis.Real.Pi.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Deriv

namespace OscInt.API.KusminLandauSanity2

open OscInt.Statements OscInt.API.KusminLandauSanity Set
"""


def make(defs, thms, mut, planted, witness=None):
    parts = []
    for d in defs:
        t = block(STMT, 'def', d)
        if planted and d == mut[0]:
            assert t.count(mut[1]) == 1, (d, mut[1], t.count(mut[1]))
            t = t.replace(mut[1], mut[2])
        parts.append(t)
    if witness is None:
        parts += [block(SAN, 'theorem', th) for th in thms]
    text = '\n\n'.join(parts)
    names = defs + (thms if witness is None else [])
    for n in sorted(names, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    if witness is not None:
        text += '\n\n' + witness
    return HEADER + '\n' + text + '\n\nend OscInt.API.KusminLandauSanity2\n'


def run(tag, text):
    src = os.path.join(OUT, tag + '.lean')
    open(src, 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs


summary = []
ONLY = sys.argv[3:]
for pid, defs, thms, mut, note in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    jobs = [('ok-' + pid, make(defs, thms, mut, False), 'accepted'),
            ('plant-' + pid, make(defs, thms, mut, True), None)]
    if pid in WITNESS:
        jobs.append(('witness-' + pid, make(defs, thms, mut, True, WITNESS[pid]), 'accepted'))
    res = {}
    for tag, text, w in jobs:
        verdict, errs = run(tag, text)
        res[tag.split('-')[0]] = verdict
        line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
        if errs:
            line += '\n    first error: ' + errs[0].split('/')[-1][:160]
        print(line, flush=True)
        summary.append(line)
    if res.get('witness') == 'accepted':
        v = 'DETECTED (compiled witness of the negation of the planted statement)'
    elif pid in TRUE_WEAKENING:
        v = TRUE_WEAKENING[pid]
    else:
        v = 'NOT DETECTED' + ('' if res['plant'] == 'accepted' else ' (script-only rejection)')
    if pid in NOTE:
        v += '. ' + NOTE[pid]
    line = textwrap.fill(f"verdict {pid} [{note}]: {v}", width=116, initial_indent='    ',
                         subsequent_indent='      ')
    print(line, flush=True)
    summary.append(line)
if not ONLY:
    open(os.path.join(OUT, 'SUMMARY.txt'), 'w').write('\n'.join(summary) + '\n')
