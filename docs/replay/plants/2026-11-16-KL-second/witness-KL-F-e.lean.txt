import Mathlib.Analysis.Real.Pi.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Deriv

namespace OscInt.API.KusminLandauSanity2

open OscInt.Statements OscInt.API.KusminLandauSanity Set

def P_KusminLandauDerivStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b : ℝ) (k : ℤ) (lam : ℝ), 0 < lam → lam ≤ 1 / 2 →
    (MonotoneOn f' (Ioo a b) ∨ AntitoneOn f' (Ioo a b)) →
    (∀ x ∈ Ioo a b, (k : ℝ) + lam ≤ f' x ∧ f' x ≤ (k : ℝ) + 1 - lam) →
    ‖∑ n ∈ Finset.Icc ⌈a⌉ ⌊b⌋, Complex.exp (↑(2 * Real.pi * f (n : ℝ)) * Complex.I)‖ ≤
      Real.cos (Real.pi * lam / 2) / Real.sin (Real.pi * lam / 2)

theorem witness_not_P_KusminLandauDerivStatement : ¬ P_KusminLandauDerivStatement := by
  intro h
  have h1 := h (fun _ => 0) (fun _ => 1 / 2) 0 2 0 (1 / 2) (by norm_num) le_rfl
    (Or.inl monotoneOn_const) (fun x _ => by norm_num)
  have hs : ∑ n ∈ Finset.Icc ⌈(0 : ℝ)⌉ ⌊(2 : ℝ)⌋, eTerm 0 = 3 := by
    rw [Icc_ceil_floor_zero_two]
    norm_num [eTerm_zero]
  have h2 : ‖(3 : ℂ)‖ ≤ _ := hs ▸ h1
  rw [cot_half] at h2
  norm_num at h2

end OscInt.API.KusminLandauSanity2
