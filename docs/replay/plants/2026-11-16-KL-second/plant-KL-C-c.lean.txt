import Mathlib.Analysis.Real.Pi.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Bounds
import Mathlib.Analysis.SpecialFunctions.Trigonometric.Deriv

namespace OscInt.API.KusminLandauSanity2

open OscInt.Statements OscInt.API.KusminLandauSanity Set

def P_KusminLandauTwoOverPiStatement : Prop :=
  ∀ (c : ℕ → ℝ) (N : ℕ) (k : ℤ) (lam : ℝ), 0 ≤ lam → lam ≤ 1 / 2 → DiffsMonotoneIn c N k lam →
    ‖expSum c N‖ ≤ 2 / (Real.pi * lam)

theorem P_klc_six (h : P_KusminLandauTwoOverPiStatement) : Real.sqrt 13 ≤ 2 / (Real.pi * (1 / 6)) := by
  have := h cSix 6 0 (1 / 6) (by norm_num) (by norm_num) diffs_cSix
  rwa [norm_expSum_cSix] at this

theorem P_klc_third (h : P_KusminLandauTwoOverPiStatement) : Real.sqrt 3 ≤ 2 / (Real.pi * (1 / 3)) := by
  have := h cThird 2 0 (1 / 3) (by norm_num) (by norm_num) diffs_cThird
  rwa [norm_expSum_cThird] at this

theorem P_klc_one_half (h : P_KusminLandauTwoOverPiStatement) : (1 : ℝ) ≤ 2 / (Real.pi * (1 / 2)) := by
  have := h cOne 0 0 (1 / 2) (by norm_num) le_rfl (diffs_cOne 0 (1 / 2))
  rwa [expSum_cOne, norm_one] at this

end OscInt.API.KusminLandauSanity2
