/-- PLANTED copy of `StationaryLeadingStatement`: hypothesis 0 < λ dropped. -/
def P_StationaryLeadingStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ →
    ∃ τ : ℝ, 0 ≤ τ ∧ ‖splRemainder g lam‖ ≤ ‖splKernel τ‖ * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingStatement : ¬ P_StationaryLeadingStatement := by
  intro h
  obtain ⟨τ, -, h1⟩ := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 0 (gauss_splAmplitude 1 one_pos)
  rw [Real.zero_rpow (by norm_num : (-(3 : ℝ) / 2) ≠ 0), mul_zero, norm_splRemainder_gauss_zero] at h1
  linarith [sqrt_two_pi_lower]
