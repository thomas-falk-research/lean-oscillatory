/-- PLANTED copy of `StationaryLeadingStatement`: "0 ≤ τ" dropped. -/
def P_StationaryLeadingStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ∃ τ : ℝ, ‖splRemainder g lam‖ ≤ ‖splKernel τ‖ * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem planted_iff_StationaryLeadingStatement : P_StationaryLeadingStatement ↔ StationaryLeadingStatement := by
  constructor
  · intro h g g₁ g₂ g₃ lam hamp hlam
    obtain ⟨τ, h1⟩ := h g g₁ g₂ g₃ lam hamp hlam
    refine ⟨|τ|, abs_nonneg τ, ?_⟩
    rw [norm_splKernel_abs]
    exact h1
  · intro h g g₁ g₂ g₃ lam hamp hlam
    obtain ⟨τ, -, h1⟩ := h g g₁ g₂ g₃ lam hamp hlam
    exact ⟨τ, h1⟩

end OscInt.API.StationaryLeadingSanity
