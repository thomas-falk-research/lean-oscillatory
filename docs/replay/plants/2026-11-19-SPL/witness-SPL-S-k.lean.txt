/-- PLANTED copy of `StationaryLeadingStatement`: "∃ τ, 0 ≤ τ ∧ …" ↦ "∀ τ, 0 ≤ τ → …". -/
def P_StationaryLeadingStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ∀ τ : ℝ, 0 ≤ τ → ‖splRemainder g lam‖ ≤ ‖splKernel τ‖ * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingStatement : ¬ P_StationaryLeadingStatement := by
  intro h
  have h1 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num) 0 le_rfl
  rw [splKernel_zero, norm_zero, zero_mul, zero_mul] at h1
  linarith [splRemainder_gauss_one_hundred_lower]

end OscInt.API.StationaryLeadingSanity
