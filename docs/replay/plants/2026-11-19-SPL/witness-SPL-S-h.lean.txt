/-- PLANTED copy of `SplAmplitude`: Integrable g dropped from SplAmplitude (planted copy of the definition). -/
def P_SplAmplitude (g g₁ g₂ g₃ : ℝ → ℂ) : Prop :=
  (∀ x, HasDerivAt g (g₁ x) x) ∧ (∀ x, HasDerivAt g₁ (g₂ x) x) ∧ (∀ x, HasDerivAt g₂ (g₃ x) x) ∧
    Integrable g₁ ∧ Integrable g₂ ∧ Integrable g₃

/-- Copy of `StationaryLeadingStatement` over the copied definition. -/
def P_StationaryLeadingStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), P_SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ∃ τ : ℝ, 0 ≤ τ ∧ ‖splRemainder g lam‖ ≤ ‖splKernel τ‖ * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingStatement : ¬ P_StationaryLeadingStatement := by

  have hamp : P_SplAmplitude (fun _ => 1) (fun _ => 0) (fun _ => 0) (fun _ => 0) :=

      integrable_zero _ _ _, integrable_zero _ _ _, integrable_zero _ _ _⟩
  obtain ⟨τ, -, h1⟩ := h (fun _ => 1) (fun _ => 0) (fun _ => 0) (fun _ => 0) 1 hamp one_pos
  rw [norm_splRemainder_one] at h1
  simp only [norm_zero, integral_zero, mul_zero, zero_mul] at h1
