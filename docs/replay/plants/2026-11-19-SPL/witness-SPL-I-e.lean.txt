namespace OscInt.API.StationaryLeadingSanity

open OscInt.Statements MeasureTheory Set Filter
open scoped ContDiff Topology

/-- PLANTED copy of `splKernel`: the kernel extended as an even function (planted copy of splKernel). -/
noncomputable def P_splKernel (τ : ℝ) : ℂ := if 0 ≤ τ then splKernelPos τ else splKernelPos (-τ)

/-- Copy of `StationaryLeadingIdentityStatement` over the copied definition. -/
def P_StationaryLeadingIdentityStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    splRemainder g lam =
      -Complex.I * ((lam ^ (-(3 : ℝ) / 2) : ℝ) : ℂ) * ∫ t, g₃ t * P_splKernel (Real.sqrt lam * t)

/-- The planted kernel is even. -/
theorem P_splKernel_neg (τ : ℝ) : P_splKernel (-τ) = P_splKernel τ := by
  unfold P_splKernel
  rcases lt_trichotomy τ 0 with h | h | h
  · rw [ite_eq_left (by linarith), ite_eq_right (by linarith)]
  · subst h
    simp
  · rw [ite_eq_right (by linarith), ite_eq_left (by linarith), neg_neg]

/-- With an even kernel the integrand `G_1‴(t) κ(10t)` is odd, so the planted identity says `R(G_1, 100) = 0`. -/
theorem witness_not_P_StationaryLeadingIdentityStatement : ¬ P_StationaryLeadingIdentityStatement := by
  intro h
  have h1 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  have h2 : ∫ t, gauss3 1 t * P_splKernel (Real.sqrt 100 * t) = 0 := by
    refine integral_eq_zero_of_odd fun t => ?_
    show gauss3 1 (-t) * P_splKernel (Real.sqrt 100 * -t) = -(gauss3 1 t * P_splKernel (Real.sqrt 100 * t))
    rw [gauss3_neg, mul_neg, P_splKernel_neg]
    ring
  rw [h2, mul_zero] at h1
  have h3 := splRemainder_gauss_one_hundred_lower
  rw [h1, norm_zero] at h3
  norm_num at h3

end OscInt.API.StationaryLeadingSanity
