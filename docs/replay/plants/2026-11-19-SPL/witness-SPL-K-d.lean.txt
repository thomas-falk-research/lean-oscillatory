/-- PLANTED copy of `splRemainder`: √(2π/λ) ↦ √(π/λ) in the leading term (planted copy of splRemainder). -/
noncomputable def P_splRemainder (g : ℝ → ℂ) (lam : ℝ) : ℂ :=
  (∫ u, g u * Complex.exp (((lam * u ^ 2 / 2 : ℝ) : ℂ) * Complex.I)) -
    g 0 * ((Real.sqrt (Real.pi / lam) : ℝ) : ℂ) * Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I)

/-- Copy of `StationaryLeadingSevenEighthsStatement` over the copied definition. -/
def P_StationaryLeadingSevenEighthsStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ‖P_splRemainder g lam‖ ≤ (7 / 8 : ℝ) * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingSevenEighthsStatement : ¬ P_StationaryLeadingSevenEighthsStatement := by
  intro h
  have e : P_splRemainder (gauss 1) 100 = gaussRoot 1 100 - ((Real.sqrt (Real.pi / 100) : ℝ) : ℂ) *
      Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I) := by
    unfold P_splRemainder
    rw [integral_gauss_mul_exp 1 one_pos, gauss_zero, one_mul]
  have hl : ‖((Real.sqrt (Real.pi / 100) : ℝ) : ℂ) * Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I)‖ ≤
      9 / 50 := by
    rw [norm_mul, Complex.norm_exp_ofReal_mul_I, mul_one, Complex.norm_real, Real.norm_eq_abs,
      abs_of_nonneg (Real.sqrt_nonneg _)]
    exact (Real.sqrt_le_left (by norm_num)).2 (by nlinarith [Real.pi_lt_d2])
  have hlow : 7 / 100 ≤ ‖P_splRemainder (gauss 1) 100‖ := by
    rw [e]
    have h2 := norm_sub_norm_le (gaussRoot 1 100)
      (((Real.sqrt (Real.pi / 100) : ℝ) : ℂ) * Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I))
    linarith [norm_gaussRoot_one_hundred_bounds.1]
  have h1 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  rw [integral_norm_gauss3 1 one_pos, rpow_hundred] at h1
  linarith [normThree_le]
