#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-SPL statements against copies of their sanity theorems.

As the batch-VC2 generators (docs/replay/plants/2026-11-16-VC2/plants_vc2.py.txt and the second run). The statement
module (OscInt/Statements/StationaryLeading.lean, a draft, not frozen) and the sanity layer
(OscInt/Front/StationaryLeadingSanity.lean) are compiled modules, so every generated file imports
OscInt.Front.StationaryLeadingSanity only and contains, in the namespace of the sanity layer and under the prefix P_:
a copy of ONE statement constant; where the mutation is in a definition the statement uses (splRemainder,
SplAmplitude, splKernel), a copy of that definition, of every data lemma of the sanity layer that mentions it
(directly or through another copied lemma), and of the partner statement of the consistency theorem; then copies of
the scripts that go through the statement: every sanity theorem that takes the statement constant as a hypothesis.
Exactly one textual mutation is applied to one copied definition. The unplanted copy (ok-) must compile.
Witness files (docs/lessons.md, rule 9) contain the planted definitions and one compiled theorem, of one of these
kinds (the verdict says which):
  not        : ¬ planted statement, from unconditional data of the sanity layer            -> DETECTED
  cond       : (∀ τ, ‖splKernel τ‖ ≤ 2) → ¬ planted statement (Lemma 2 of docs/proof-SPL.md,
               not proved in Lean when this was run)                                 -> DETECTED CONDITIONALLY
  rel        : true statement → ¬ planted statement                                  -> DETECTED RELATIVE TO THE TRUE
                                                                                        STATEMENT
  iff        : planted statement ↔ draft statement                                   -> EQUIVALENT
  true, weak : the planted statement proved outright; draft statement → planted      -> TRUE WEAKENING
No file proves, or uses a proof of, a statement of the batch. A rejected script alone is never counted as a
detection. The compiler output is recorded with the scratch directory removed from the paths.
The first 28 plants are those of the lead's brief (with two verdicts stronger than the brief expected: SPL-I-d and
SPL-I-e); the last 14 are the mutations proposed by an external referee of the draft statements and two controls
that must not distinguish (SPL-I-h, SPL-K-l), added on the lead's instruction. Witness-only lemmas (evenness of the
second derivative of the Gaussian, a scaling of its third derivative, evenness of a planted kernel) are stated in
the witness files that use them; none is about the kernel of the batch beyond its oddness.
Usage: plants_spl.py REPO OUTDIR SCRATCHDIR [plant ids]
"""
import os, re, subprocess, sys, textwrap

REPO, OUT, SCR = sys.argv[1:4]
OUT, SCR = os.path.abspath(OUT), os.path.abspath(SCR)
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT = open(os.path.join(REPO, 'OscInt/Statements/StationaryLeading.lean')).read().split('\n')
SAN_ALL = open(os.path.join(REPO, 'OscInt/Front/StationaryLeadingSanity.lean')).read()
SAN = SAN_ALL[SAN_ALL.index('namespace OscInt.API.StationaryLeadingSanity'):].split('\n')
I, S, K, A = ('StationaryLeadingIdentityStatement', 'StationaryLeadingStatement',
              'StationaryLeadingSevenEighthsStatement', 'StationaryLeadingAttainedStatement')
STATEMENTS = [I, S, K, A]
REM, AMP, KER = 'splRemainder', 'SplAmplitude', 'splKernel'
STMT_DEFS = ['fresnelPrim', 'fresnelLimit', 'splKernelPos', KER, AMP, REM] + STATEMENTS
DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9']+)")
WORD = re.compile(r"[A-Za-z_][A-Za-z_0-9']*")


def block(lines, name):
    """The declaration `name` without its docstring, up to the next blank line."""
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    """All declarations of the sanity file with docstrings, in file order: name -> (index, text)."""
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()


def words(t):
    """The identifiers of a declaration, its docstring excluded."""
    ls = t.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    return set(WORD.findall('\n'.join(ls[k:])))


# Scripts: the sanity theorems that take a statement constant as a hypothesis, per statement, in file order.
THROUGH = {st: [n for n, (_, t) in sorted(DATA.items(), key=lambda kv: kv[1][0]) if st in words(t)]
           for st in STATEMENTS}
ALL_THROUGH = {n for ns in THROUGH.values() for n in ns}
for st in STATEMENTS:
    assert THROUGH[st], st


def mention_closure(name):
    """Data declarations (not scripts) of the sanity file that mention `name`, directly or through one another."""
    got, changed = set(), True
    while changed:
        changed = False
        for n, (_, t) in DATA.items():
            if n in got or n in ALL_THROUGH:
                continue
            w = words(t)
            if name in w or (w & got):
                got.add(n)
                changed = True
    return sorted(got, key=lambda n: DATA[n][0])


OPEN = """import OscInt.Front.StationaryLeadingSanity

namespace OscInt.API.StationaryLeadingSanity

open OscInt.Statements MeasureTheory Set Filter
open scoped ContDiff Topology
"""
CLOSE = '\n\nend OscInt.API.StationaryLeadingSanity\n'


def wrap(s):
    """A docstring within 120 columns."""
    return '\n'.join(textwrap.wrap('/-- ' + s + ' -/', 118))


def rename(text, names):
    """Prefix the given names with P_ in a declaration; its docstring is left as it is."""
    ls = text.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    doc, decl = '\n'.join(ls[:k]), '\n'.join(ls[k:])
    for n in sorted(names, key=len, reverse=True):
        decl = re.sub(r"(?<![\w.'])" + re.escape(n) + r"(?![\w'])", 'P_' + n, decl)
    return (doc + '\n' if doc else '') + decl


def make(stmt, mut, planted, note, witness=None):
    """The text of a generated file. `mut` = (definition to mutate, old, new, number of occurrences)."""
    target, old, new, count = mut
    in_def = target != stmt
    scripts = THROUGH[stmt]
    defs = [target, stmt] if in_def else [stmt]
    data = mention_closure(target) if in_def else []
    if in_def and witness is None:
        # the partner statement of a script with two statement hypotheses is copied over the planted definition
        for th in scripts:
            for st in STATEMENTS:
                if st in words(DATA[th][1]) and st not in defs:
                    defs.append(st)
    defs.sort(key=STMT_DEFS.index)
    parts = []
    for d in defs:
        t = block(STMT, d)
        if planted and d == target:
            assert t.count(old) == count, (d, old, t.count(old))
            t = t.replace(old, new)
        if d == target:
            doc = (f'PLANTED copy of `{d}`: {note.split(";")[0]}.' if planted else f'Unplanted copy of `{d}`.')
        else:
            doc = f'Copy of `{d}` over the copied definition' + ('s.' if len(defs) > 2 else '.')
        parts.append(wrap(doc) + '\n' + t)
    renamed = list(defs)
    if witness is None:
        parts += [DATA[n][1] for n in data] + [DATA[n][1] for n in scripts]
        renamed += data + scripts
    text = '\n\n'.join(rename(part, renamed) for part in parts)
    if witness is not None:
        text += '\n\n' + witness.strip('\n')
    return OPEN + '\n' + text + CLOSE


def compile_text(tag, text):
    """Compile one generated file (one Lean process at a time, 15 minutes at most). The result is kept in the scratch
    directory and reused when the same text is asked for again, so that an interrupted run can be resumed."""
    src, res = os.path.join(SCR, tag + '.lean'), os.path.join(SCR, tag + '.result')
    if os.path.exists(src) and os.path.exists(res) and open(src).read() == text:
        rc, out = open(res).read().split('\n', 1)
        return int(rc), out
    open(src, 'w').write(text)
    env = {k: v for k, v in os.environ.items() if k not in ('QISKIT_IBM_TOKEN', 'QISKIT_IBM_INSTANCE', 'CTQW_GUARD')}
    try:
        r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', src],
                           cwd=REPO, capture_output=True, text=True, env=env, timeout=900)
        rc, out = r.returncode, (r.stdout + r.stderr).replace(SCR + os.sep, '')
    except subprocess.TimeoutExpired:
        rc, out = 124, f'{tag}.lean:1:0: error: no result after 900 s (stopped by the generator)\n'
    open(res, 'w').write(f'{rc}\n{out}')
    return rc, out
