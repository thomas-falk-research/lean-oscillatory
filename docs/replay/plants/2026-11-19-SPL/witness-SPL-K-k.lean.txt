/-- PLANTED copy of `SplAmplitude`: third clause of SplAmplitude: HasDerivAt g₂ (g₃ x) x ↦ HasDerivAt g₁ (g₃ x) x
(planted copy of the definition). -/
def P_SplAmplitude (g g₁ g₂ g₃ : ℝ → ℂ) : Prop :=
  (∀ x, HasDerivAt g (g₁ x) x) ∧ (∀ x, HasDerivAt g₁ (g₂ x) x) ∧ (∀ x, HasDerivAt g₁ (g₃ x) x) ∧
    Integrable g ∧ Integrable g₁ ∧ Integrable g₂ ∧ Integrable g₃

/-- Copy of `StationaryLeadingSevenEighthsStatement` over the copied definition. -/
def P_StationaryLeadingSevenEighthsStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), P_SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ‖splRemainder g lam‖ ≤ (7 / 8 : ℝ) * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)
