/-- PLANTED copy of `StationaryLeadingIdentityStatement`: exponent −3/2 ↦ −1/2. -/
def P_StationaryLeadingIdentityStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    splRemainder g lam =
      -Complex.I * ((lam ^ (-(1 : ℝ) / 2) : ℝ) : ℂ) * ∫ t, g₃ t * splKernel (Real.sqrt lam * t)

theorem witness_StationaryLeadingIdentityStatement_not_P (hS : StationaryLeadingIdentityStatement) : ¬ P_StationaryLeadingIdentityStatement := by
  intro h
  have h1 := hS (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  have h2 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  have hne : splRemainder (gauss 1) 100 ≠ 0 := by
    intro h0
    have h3 := splRemainder_gauss_one_hundred_lower
    rw [h0, norm_zero] at h3
    norm_num at h3
  have hlt : (100 : ℝ) ^ (-(3 : ℝ) / 2) < (100 : ℝ) ^ (-(1 : ℝ) / 2) :=
    Real.rpow_lt_rpow_of_exponent_lt (by norm_num) (by norm_num)
  have h3 : ((((100 : ℝ) ^ (-(1 : ℝ) / 2) : ℝ) : ℂ) - (((100 : ℝ) ^ (-(3 : ℝ) / 2) : ℝ) : ℂ)) *
      splRemainder (gauss 1) 100 = 0 := by
    linear_combination (((100 : ℝ) ^ (-(1 : ℝ) / 2) : ℝ) : ℂ) * h1 - (((100 : ℝ) ^ (-(3 : ℝ) / 2) : ℝ) : ℂ) * h2
  rcases mul_eq_zero.1 h3 with h4 | h4
  · rw [sub_eq_zero] at h4
    have h5 := Complex.ofReal_injective h4
    linarith
  · exact hne h4
