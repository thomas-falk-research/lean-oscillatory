import OscInt.Front.StationaryLeadingSanity

namespace OscInt.API.StationaryLeadingSanity

open OscInt.Statements MeasureTheory Set Filter
open scoped ContDiff Topology

/-- PLANTED copy of `StationaryLeadingAttainedStatement`: ContDiff ℝ ∞ g ↦ ContDiff ℝ 3 g. -/
def P_StationaryLeadingAttainedStatement : Prop :=
  ∀ τ ε : ℝ, 0 ≤ τ → 0 < ε →
    ∃ g g₁ g₂ g₃ : ℝ → ℂ, SplAmplitude g g₁ g₂ g₃ ∧ ContDiff ℝ 3 g ∧ HasCompactSupport g ∧ (∃ x, g x ≠ 0) ∧
      (‖splKernel τ‖ - ε) * (∫ t, ‖g₃ t‖) ≤ ‖splRemainder g 1‖

theorem planted_StationaryLeadingAttainedStatement_of_true (h : StationaryLeadingAttainedStatement) : P_StationaryLeadingAttainedStatement := by
  intro τ ε hτ hε
  obtain ⟨g, g₁, g₂, g₃, h1, h2, h3, h4, h5⟩ := h τ ε hτ hε
  exact ⟨g, g₁, g₂, g₃, h1, h2.of_le (by norm_num), h3, h4, h5⟩

end OscInt.API.StationaryLeadingSanity
