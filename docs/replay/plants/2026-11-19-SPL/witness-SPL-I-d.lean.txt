/-- PLANTED copy of `StationaryLeadingIdentityStatement`: √λ t ↦ λ t in the kernel argument. -/
def P_StationaryLeadingIdentityStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    splRemainder g lam =
      -Complex.I * ((lam ^ (-(3 : ℝ) / 2) : ℝ) : ℂ) * ∫ t, g₃ t * splKernel (lam * t)

/-- `G_10‴(t) = 1000 G_1‴(10t)`. -/
theorem gauss3_ten (t : ℝ) : gauss3 10 t = 1000 * gauss3 1 (10 * t) := by
  unfold gauss3 gaussR
  rw [← Complex.ofReal_ofNat, ← Complex.ofReal_mul]
  congr 1
  have e : -(1 * (10 * t)) ^ 2 / 2 = -(10 * t) ^ 2 / 2 := by ring
  rw [e]
  ring

/-- `√(1/100) = 1/10`. -/
theorem sqrt_hundredth : Real.sqrt (1 / 100) = 1 / 10 := by
  rw [show (1 / 100 : ℝ) = (1 / 10) ^ 2 by norm_num, Real.sqrt_sq (by norm_num)]

/-- The true identity for `G_10` and the planted one for `G_1`, both at `λ = 1/100`, are about the same integral
after the substitution `y = 10t`, and give `R(G_10, 1/100) = 100 R(G_1, 1/100)`; the certified bounds say
`100 |R(G_1, 1/100)| ≥ 250 √(2π)` and `|R(G_10, 1/100)| ≤ 10 √(2π)`. -/
theorem witness_StationaryLeadingIdentityStatement_not_P (hS : StationaryLeadingIdentityStatement) : ¬ P_StationaryLeadingIdentityStatement := by
  intro h
  have h1 := hS (gauss 10) (gauss1 10) (gauss2 10) (gauss3 10) (1 / 100) (gauss_splAmplitude 10 (by norm_num))
    (by norm_num)
  have h2 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) (1 / 100) (gauss_splAmplitude 1 one_pos) (by norm_num)
  rw [sqrt_hundredth] at h1
  have hsub : ∫ t, gauss3 10 t * splKernel (1 / 10 * t) = 100 * ∫ y, gauss3 1 y * splKernel (1 / 100 * y) := by
    have h3 : ∫ t : ℝ, 1000 * (gauss3 1 (10 * t) * splKernel (1 / 100 * (10 * t))) =
        |(10 : ℝ)⁻¹| • ∫ y : ℝ, 1000 * (gauss3 1 y * splKernel (1 / 100 * y)) :=
      Measure.integral_comp_mul_left (fun y : ℝ => 1000 * (gauss3 1 y * splKernel (1 / 100 * y))) 10
    have e : ∀ t : ℝ, gauss3 10 t * splKernel (1 / 10 * t) =
        1000 * (gauss3 1 (10 * t) * splKernel (1 / 100 * (10 * t))) := by
      intro t
      have e1 : (1 / 100 : ℝ) * (10 * t) = 1 / 10 * t := by ring
      rw [gauss3_ten, e1]
      ring
    simp_rw [e]
    rw [h3, integral_const_mul, abs_of_pos (by norm_num : (0 : ℝ) < 10⁻¹), Complex.real_smul]
    push_cast
    ring
  rw [hsub] at h1
  have h4 : splRemainder (gauss 10) (1 / 100) = 100 * splRemainder (gauss 1) (1 / 100) := by
    rw [h1, h2]
    ring
  have h5 := (splRemainder_gauss_bounds 10 (1 / 100) (by norm_num) (by norm_num)).2
  have h6 := (splRemainder_gauss_bounds 1 (1 / 100) one_pos (by norm_num)).1
  rw [h4, norm_mul, Complex.norm_ofNat, sqrt_hundredth] at h5
  rw [sqrt_hundredth] at h6
  have ha : 0 < Real.sqrt (2 * Real.pi) := Real.sqrt_pos.2 (by have := Real.pi_pos; positivity)
  have hq1 : (100 : ℝ) ≤ Real.sqrt (10 ^ 4 + (1 / 100) ^ 2) := Real.le_sqrt_of_sq_le (by norm_num)
  have hq2 : Real.sqrt (1 ^ 4 + (1 / 100) ^ 2) ≤ 2 := (Real.sqrt_le_left (by norm_num)).2 (by norm_num)
  have hq3 : 0 < Real.sqrt (1 ^ 4 + (1 / 100) ^ 2) := Real.sqrt_pos.2 (by norm_num)
  have h7 : Real.sqrt (2 * Real.pi) * 10 ^ 2 / (1 / 10 * Real.sqrt (10 ^ 4 + (1 / 100) ^ 2)) ≤
      Real.sqrt (2 * Real.pi) * 10 := by
    have hq4 : 0 < Real.sqrt (10 ^ 4 + (1 / 100) ^ 2) := by linarith
    rw [div_le_iff₀ (by positivity)]
    nlinarith [mul_le_mul_of_nonneg_left hq1 ha.le]
  have h8 : Real.sqrt (2 * Real.pi) * (5 / 2) ≤
      Real.sqrt (2 * Real.pi) * 1 ^ 2 / (2 * (1 / 10) * Real.sqrt (1 ^ 4 + (1 / 100) ^ 2)) := by
    rw [le_div_iff₀ (by positivity)]
    nlinarith [mul_le_mul_of_nonneg_left hq2 ha.le]
  linarith

end OscInt.API.StationaryLeadingSanity
