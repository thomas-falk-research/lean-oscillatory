/-- PLANTED copy of `StationaryLeadingSevenEighthsStatement`: ‖g₃‖₁ = ∫‖g₃‖ ↦ ‖∫ g₃‖. -/
def P_StationaryLeadingSevenEighthsStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ‖splRemainder g lam‖ ≤ (7 / 8 : ℝ) * ‖∫ t, g₃ t‖ * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingSevenEighthsStatement : ¬ P_StationaryLeadingSevenEighthsStatement := by
  intro h
  have h1 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  have h2 : ∫ t, gauss3 1 t = 0 := integral_eq_zero_of_odd (gauss3_neg 1)
  rw [h2, norm_zero, mul_zero, zero_mul] at h1
  linarith [splRemainder_gauss_one_hundred_lower]
