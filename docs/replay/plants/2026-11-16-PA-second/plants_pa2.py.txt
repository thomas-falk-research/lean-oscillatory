OscInt/Statements/PeriodicAnalytic.lean) and every sanity theorem that goes through the statement constant (from
OscInt/Front/PeriodicAnalyticSanity.lean and OscInt/Front/PeriodicAnalyticSanity2.lean) under the prefix P_, apply one
mutation to one copied def, and compile with `lake env lean -DautoImplicit=false`. The unplanted copy (ok-) must
compile. When the mutated def is `trapezoidalSum`, the four value lemmas `trapezoidalSum_*` are copied with it (they
are the pins of that def). A mutation is one textual replacement, except the two type changes (ℤ ↦ ℝ), which replace
the binder and delete the now redundant coercions of the same variable.
Witness files (docs/lessons.md, rule 9): a compiled proof of the negation of the planted statement from unconditional
data lemmas of the two sanity files. No file proves, or uses a proof of, a statement of the batch.
Verdicts: DETECTED (witness compiled), NOT DETECTED (no witness), TRUE WEAKENING (implied by the frozen statement).

def block(lines, kind, name):
    pat = re.compile(r'^(noncomputable )?' + kind + r' ' + re.escape(name) + r'\b')
    for i, l in enumerate(lines):
        if pat.match(l):
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {kind} {name}')

def make(stmt, target, muts, planted, witness=None):
    defs = [target] if target == stmt else [target, stmt]
    parts = []
    for d in defs:
        t = block(STMT, 'def', d)
        if planted and d == target:
            for old, new, cnt in muts:
                assert t.count(old) == cnt, (d, old, t.count(old))
                t = t.replace(old, new)
        parts.append(t)
    pins = TS_PINS if target == TS and witness is None else []
    thms = THMS[stmt] if witness is None else []
    if witness is None:
        parts += [block(SAN, 'theorem', th) for th in pins + thms]
    else:
        parts.append(f'theorem witness_not_{stmt} : ¬ {stmt} :=\n  {witness}')
    text = '\n\n'.join(parts)
    for n in sorted(defs + pins + thms, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    return HEADER + '\n' + text + '\n\nend OscInt.API.PeriodicAnalyticSanity2\n'


def run(tag, text):
    src = os.path.join(OUT, tag + '.lean')
    open(src, 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs


ONLY = sys.argv[3:]
jobs = []
for pid, stmt, target, muts, witness, note in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    jobs.append((pid, 'ok-' + pid, make(stmt, target, muts, False), 'accepted'))
    jobs.append((pid, 'plant-' + pid, make(stmt, target, muts, True), None))
    if witness:
        jobs.append((pid, 'witness-' + pid, make(stmt, target, muts, True, witness), 'accepted'))
with ThreadPoolExecutor(max_workers=int(os.environ.get('PLANT_JOBS', '1'))) as ex:
    results = list(ex.map(lambda j: run(j[1], j[2]), jobs))

summary, res = [], {}
for (pid, tag, _, w), (verdict, errs) in zip(jobs, results):
    res[tag] = verdict
    line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
    if errs:
        line += '\n    first error: ' + errs[0].split('/')[-1][:160]
    summary.append((pid, line))
lines = []
for pid, stmt, target, muts, witness, note in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    lines += [l for p, l in summary if p == pid]
    if pid in TRUE_WEAKENING:
        v = 'TRUE WEAKENING: ' + TRUE_WEAKENING[pid]
    elif res.get('witness-' + pid) == 'accepted':
        v = 'DETECTED (witness compiled: ¬ planted statement)'
    else:
        v = 'NOT DETECTED (no compiled witness): ' + NOT_DETECTED.get(pid, '')
    lines.append(f"  {pid} [{note}]: {v}")
print('\n'.join(lines), flush=True)
open(os.path.join(OUT, 'SUMMARY.txt'), 'w').write('\n'.join(lines) + '\n')
