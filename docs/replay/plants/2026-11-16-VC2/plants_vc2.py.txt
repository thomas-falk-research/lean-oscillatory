#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-VC2 statements against copies of their sanity theorems.

As the batch-FR generator (docs/replay/plants/2026-11-16-FR/plants_fr.py.txt). The batch is not yet a module of the
library, so every generated file is self-contained: imports, the statement text copied verbatim from
SecondDerivSharp-statements.lean, the unconditional data lemmas of SecondDerivSharpSanity.lean that the file needs
(copied verbatim, selected by name closure), then, under the prefix P_, a copy of one statement (and of
DerivIncreasesAtRate with its data lemmas where the mutation is in that definition) and the sanity theorems that go
through the statement constant. Exactly one textual mutation is applied to one copied def. The unplanted copy (ok-)
must compile. Witness files (docs/lessons.md, rule 9): a compiled proof of the negation of the planted statement from
unconditional data lemmas. Four witnesses (S-a, S-c, S-d, S-f) also use the bound A(w) <= 2 sqrt(2w), which holds for
every w: it is proved in those witness files only (WITNESS_ONLY below), where it refutes a planted statement; it is
not part of the sanity file. No file proves, or uses a proof of, a statement of the batch.
Verdicts: DETECTED (witness compiled), NOT DETECTED (no witness), TRUE WEAKENING (implied by the draft statement).
Usage: plants_vc2.py REPO SANITY_DIR OUTDIR SCRATCHDIR [plant ids]
"""
import os, re, subprocess, sys, textwrap

REPO, SDIR, OUT, SCR = sys.argv[1:5]
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT_TEXT = open(os.path.join(SDIR, 'SecondDerivSharp-statements.lean')).read()

STMT_TEXT = STMT_TEXT[i0:i1] + 'end OscInt.Statements\n'
STMT = STMT_TEXT.split('\n')
SAN_ALL = open(os.path.join(SDIR, 'SecondDerivSharpSanity.lean')).read()
assert STMT_TEXT in SAN_ALL, 'the sanity file does not carry the statement text verbatim'
SAN_BODY = SAN_ALL[SAN_ALL.index('namespace OscInt.API.SecondDerivSharpSanity'):]
SAN = SAN_BODY.split('\n')
S, A, N, K = ('SecondDerivSharpStatement', 'SecondDerivAttainedStatement', 'Vdc2ProfileBoundStatement',
              'SecondDerivSevenHalvesStatement')
THMS = {S: ['sharp_quadratic', 'sharp_scaled', 'sharp_degenerate', 'sharp_profile_large'],
        A: ['attained_at', 'attained_zero'], N: ['profile_le', 'profile_within'],
        K: ['sevenHalves_quadratic', 'sevenHalves_linear', 'sevenHalves_neg', 'sevenHalves_scaled',
            'sevenHalves_ramp']}
RATE = 'DerivIncreasesAtRate'
RATE_DATA = ['scaled_increases', 'quad_increases']
DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9']+)")


def block(lines, name):
    """The declaration `name` without its docstring, up to the next blank line."""
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    """All declarations of the sanity file with docstrings, in file order: name -> (index, text)."""
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()
THROUGH = {t for ts in THMS.values() for t in ts}


def closure(text, skip=()):
    """Unconditional data declarations of the sanity file used by `text`, transitively, in file order."""
    need, todo = set(), [text]
    while todo:
        t = todo.pop()
        for w in set(re.findall(r"[A-Za-z_][A-Za-z_0-9']*", t)):
            if w in DATA and w not in need and w not in THROUGH and w not in skip:
                need.add(w)
                todo.append(DATA[w][1])
    return [DATA[n][1] for n in sorted(need, key=lambda n: DATA[n][0])]

def wrap(s):
    """A docstring within 120 columns."""
    return '\n'.join(textwrap.wrap('/-- ' + s + ' -/', 118))


def doc(name, planted, note):
    return wrap(f'PLANTED copy of `{name}`: {note.split(";")[0]}.' if planted else f'Unplanted copy of `{name}`.')


def make(stmt, mut, planted, note, witness=None, wonly=False):
    target, old, new, count = mut
    in_def = target == RATE
    defs = [target] if target == stmt else [target, stmt]
    parts = []
    for d in defs:
        t = block(STMT, d)
        if planted and d == target:
            assert t.count(old) == count, (d, old, t.count(old))
            t = t.replace(old, new)
        parts.append(doc(d, planted and d == target, note) + '\n' + t)
    renamed = list(defs)
    if in_def:
        parts += [DATA[th][1] for th in RATE_DATA]
        renamed += RATE_DATA
    if witness is None:
        parts += [DATA[th][1] for th in THMS[stmt]]
        renamed += THMS[stmt]
    else:
        parts.append(wrap(f'Witness: the planted copy is false ({note.split("witness: ")[-1]}).') + '\n' +
                     f'theorem witness_not_{stmt} : ¬ {stmt} := {witness}')
    text = '\n\n'.join(parts)
    for n in sorted(renamed, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r'(?![\w])', 'P_' + n, text)
    text = text.replace('copy of `P_', 'copy of `')
    extra = [WITNESS_ONLY] if wonly else []
    data = closure(text + '\n'.join(extra), skip=RATE_DATA if in_def else ())
    return (HEADER + STMT_TEXT + OPEN + '\n' + '\n\n'.join(data + extra + [text]) +
            '\n\nend OscInt.API.SecondDerivSharpSanity\n')


def run(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs

ONLY = sys.argv[5:]
jobs = []
for pid, stmt, mut, witness, wonly, note in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    jobs.append((pid, 'ok-' + pid, make(stmt, mut, False, note), 'accepted'))
    jobs.append((pid, 'plant-' + pid, make(stmt, mut, True, note), None))
    if witness:
        jobs.append((pid, 'witness-' + pid, make(stmt, mut, True, note, witness, wonly), 'accepted'))
results = [run(j[1], j[2]) for j in jobs]

by_pid = {}
for (pid, tag, _, w), (verdict, errs) in zip(jobs, results):
    line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
    if errs:
        line += '\n    first error: ' + errs[0].split('/')[-1][:160]
    by_pid.setdefault(pid, {})[tag.split('-')[0]] = (verdict, line)
lines = []
for pid, stmt, mut, witness, wonly, note in PLANTS:
    if pid not in by_pid:
        continue
    r = by_pid[pid]
    lines += [r[k][1] for k in ('ok', 'plant', 'witness') if k in r]
    if witness:
        v = ('DETECTED (witness compiled: ¬ planted statement)' if r['witness'][0] == 'accepted'
             else 'WITNESS FAILED  <-- MISMATCH')
    elif pid in TRUE_WEAKENING:
        v = 'TRUE WEAKENING: ' + TRUE_WEAKENING[pid]
    else:
        v = 'NOT DETECTED: ' + NOT_DETECTED[pid]
    lines.append(f'  {pid} [{note}]: {v}')
open(os.path.join(OUT, 'SUMMARY.txt' if not ONLY else 'SUMMARY-partial.txt'), 'w').write('\n'.join(lines) + '\n')
