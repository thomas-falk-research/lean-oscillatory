namespace OscInt.Statements

open Set

/-- `A(w) = 2 ∫_0^{√(2w)} sin(w − u²/2) du`: the largest real part, over rotations, that the quadratic phase can reach
when its positive arc ends at phase `w`. Its maximum over `[0, π]` is van der Corput's constant. -/
noncomputable def vdc2Profile (w : ℝ) : ℝ :=
  2 * ∫ u in (0 : ℝ)..Real.sqrt (2 * w), Real.sin (w - u ^ 2 / 2)

/-- `f′` increases at rate at least `lam` on the open interval `(a, b)`. -/
def DerivIncreasesAtRate (f' : ℝ → ℝ) (a b lam : ℝ) : Prop :=
  ∀ x ∈ Ioo a b, ∀ y ∈ Ioo a b, x ≤ y → lam * (y - x) ≤ f' y - f' x

/-- **VC2-S** The second-derivative van der Corput bound with its sharp constant: if `f′` increases at rate at least
`λ > 0`, then `|∫_a^b e^{if}| ≤ A(w)/√λ` for some `w ∈ [0, π]`. -/
def SecondDerivSharpStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b lam : ℝ), a ≤ b → 0 < lam → HasDerivOn f f' a b → DerivIncreasesAtRate f' a b lam →
    ∃ w ∈ Icc (0 : ℝ) Real.pi,
      ‖∫ x in a..b, Complex.exp (((f x : ℝ) : ℂ) * Complex.I)‖ ≤ vdc2Profile w / Real.sqrt lam

theorem statement_pin_SecondDerivSharp : SecondDerivSharpStatement = SecondDerivSharpStatement := rfl

/-- **VC2-A** The bound of VC2-S is attained by the quadratic phase: `A(w) ≤ |∫_{−√(2w)}^{√(2w)} e^{ix²/2} dx|`. -/
def SecondDerivAttainedStatement : Prop :=
  ∀ w : ℝ, 0 ≤ w →
    vdc2Profile w ≤
      ‖∫ x in (-Real.sqrt (2 * w))..(Real.sqrt (2 * w)), Complex.exp (((x ^ 2 / 2 : ℝ) : ℂ) * Complex.I)‖

theorem statement_pin_SecondDerivAttained : SecondDerivAttainedStatement = SecondDerivAttainedStatement := rfl

/-- **VC2-N** An explicit bound for the sharp constant: `A(w) ≤ 7/2` for `0 ≤ w ≤ π`. -/
def Vdc2ProfileBoundStatement : Prop :=
  ∀ w ∈ Icc (0 : ℝ) Real.pi, vdc2Profile w ≤ 7 / 2

theorem statement_pin_Vdc2ProfileBound : Vdc2ProfileBoundStatement = Vdc2ProfileBoundStatement := rfl

/-- **VC2-K** The second-derivative test with amplitude and a linear term, constant `7/2`:
`|∫_a^b e^{i(φ(x) + px)} ψ(x) dx| ≤ (7/2) μ^{−1/2} (|ψ(b)| + ∫_a^b |ψ′|)` when `|φ″| ≥ μ > 0` on `(a, b)`. -/
def SecondDerivSevenHalvesStatement : Prop :=
  ∀ (φ φ' φ'' : ℝ → ℝ) (ψ ψ' : ℝ → ℂ) (a b μ p : ℝ), a ≤ b → 0 < μ →
    HasDerivOn φ φ' a b → HasDerivOn φ' φ'' a b → (∀ x ∈ Ioo a b, μ ≤ |φ'' x|) → AmplitudeOn ψ ψ' a b →
    ‖oscInt (fun x => φ x + p * x) ψ a b‖ ≤ (7 / 2 : ℝ) * μ ^ (-(1 : ℝ) / 2) * ampNorm ψ ψ' a b

theorem statement_pin_SecondDerivSevenHalves :
    SecondDerivSevenHalvesStatement = SecondDerivSevenHalvesStatement := rfl

end OscInt.Statements

/-- The real part of the interval integral of a continuous complex function. -/
theorem re_intervalIntegral {F : ℝ → ℂ} (hF : Continuous F) (a b : ℝ) :
    (∫ x in a..b, F x).re = ∫ x in a..b, (F x).re :=
  (Complex.reCLM.intervalIntegral_comp_comm (hF.intervalIntegrable a b)).symm

/-- `7/8 ≤ cos (1/2)`. -/
theorem cos_half_lower : 7 / 8 ≤ Real.cos (1 / 2) := by
  have h := Real.one_sub_sq_div_two_le_cos (x := 1 / 2)
  norm_num at h
  linarith

/-- The scaled quadratic datum `x ↦ λ x²/2` has derivative `x ↦ λ x` on every interval. -/
theorem scaled_hasDerivOn (lam a b : ℝ) : HasDerivOn (fun x => lam * x ^ 2 / 2) (fun x => lam * x) a b := by
  have hd : ∀ x : ℝ, HasDerivAt (fun x : ℝ => lam * x ^ 2 / 2) (lam * x) x := fun x => by
    have h := ((hasDerivAt_pow 2 x).const_mul lam).div_const 2
    convert h using 1
    norm_num
    ring
  exact ⟨fun x _ => (hd x).continuousAt.continuousWithinAt, fun x _ => hd x⟩

/-- `x ↦ λ x` increases at rate (exactly) `λ` on every interval. -/
theorem scaled_increases (lam a b : ℝ) : DerivIncreasesAtRate (fun x => lam * x) a b lam := by
  intro x _ y _ _
  exact le_of_eq (by ring)

/-- The scaled datum: `(2/10) cos(1/2) ≤ |∫_{−1/10}^{1/10} e^{i·100x²/2} dx|` (true value `0.19778…`). -/
theorem scaled_tenth_lower :
    2 / 10 * Real.cos (1 / 2) ≤
      ‖∫ x in (-(1 / 10 : ℝ))..(1 / 10), Complex.exp (((100 * x ^ 2 / 2 : ℝ) : ℂ) * Complex.I)‖ := by
  refine le_trans ?_ (Complex.re_le_norm _)
  rw [re_intervalIntegral (by fun_prop)]
  simp only [Complex.exp_ofReal_mul_I_re]
  have h := intervalIntegral.integral_mono_on (μ := MeasureTheory.volume) (f := fun _ => Real.cos (1 / 2))
    (g := fun x => Real.cos (100 * x ^ 2 / 2)) (a := -(1 / 10)) (b := 1 / 10) (by norm_num)
    (continuous_const.intervalIntegrable _ _)
    ((by fun_prop : Continuous fun x : ℝ => Real.cos (100 * x ^ 2 / 2)).intervalIntegrable _ _)
    (fun x hx => Real.cos_le_cos_of_nonneg_of_le_pi (by positivity) (by linarith [Real.pi_gt_three])
      (by nlinarith [hx.1, hx.2]))
  rw [intervalIntegral.integral_const] at h
  norm_num at h
  linarith

/-- (Witness file only: it refutes a planted statement and proves nothing of the batch.) `A(w) ≤ 2B` when
`2w ≤ B²`, `B ≥ 0`: the trivial bound `A(w) ≤ 2 √(2w)` from `|sin| ≤ 1`. -/
theorem vdc2Profile_le_of_le (w B : ℝ) (hB : 0 ≤ B) (hw : 2 * w ≤ B ^ 2) : vdc2Profile w ≤ 2 * B := by
  have h := intervalIntegral.norm_integral_le_of_norm_le_const (a := 0) (b := Real.sqrt (2 * w)) (C := 1)
    (f := fun u : ℝ => Real.sin (w - u ^ 2 / 2)) (fun u _ => by simpa using Real.abs_sin_le_one _)
  rw [sub_zero, abs_of_nonneg (Real.sqrt_nonneg _), one_mul, Real.norm_eq_abs] at h
  have h1 := le_abs_self (∫ u in (0 : ℝ)..Real.sqrt (2 * w), Real.sin (w - u ^ 2 / 2))
  have h2 : Real.sqrt (2 * w) ≤ B := (Real.sqrt_le_left hB).2 hw
  unfold vdc2Profile
  linarith

/-- (Witness file only.) `A(w) ≤ 26/5` for `w ≤ π` (`2π < 6.3 ≤ (13/5)²`). -/
theorem vdc2Profile_le_pi (w : ℝ) (hw : w ≤ Real.pi) : vdc2Profile w ≤ 26 / 5 := by
  have h := vdc2Profile_le_of_le w (13 / 5) (by norm_num) (by nlinarith [Real.pi_lt_d2])
  linarith

/-- PLANTED copy of `SecondDerivSharpStatement`: / √λ ↦ / λ. -/
def P_SecondDerivSharpStatement : Prop :=
  ∀ (f f' : ℝ → ℝ) (a b lam : ℝ), a ≤ b → 0 < lam → HasDerivOn f f' a b → DerivIncreasesAtRate f' a b lam →
    ∃ w ∈ Icc (0 : ℝ) Real.pi,
      ‖∫ x in a..b, Complex.exp (((f x : ℝ) : ℂ) * Complex.I)‖ ≤ vdc2Profile w / lam

/-- Witness: the planted copy is false (λ = 100, f = 100x²/2 on [−1/10, 1/10]: left side ≥ (2/10)·(7/8) = 0.175,
planted bound A(w)/100 ≤ 0.052). -/
theorem witness_not_SecondDerivSharpStatement : ¬ P_SecondDerivSharpStatement := by
  intro h
  obtain ⟨w, hw, hle⟩ := h (fun x => 100 * x ^ 2 / 2) (fun x => 100 * x) (-(1 / 10)) (1 / 10) 100 (by norm_num)
    (by norm_num) (scaled_hasDerivOn _ _ _) (scaled_increases _ _ _)
  linarith [scaled_tenth_lower, cos_half_lower, vdc2Profile_le_pi w hw.2]
