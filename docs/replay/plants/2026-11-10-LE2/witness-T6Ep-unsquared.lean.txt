def P_LatticeLaplacianCubeRootStatement : Prop :=
  ∀ (t : ℝ) (x : ℤ × ℤ), 0 < t →
    ‖latticeKernel t x‖ ≤ (8 * (1 + Real.sqrt 7) / 3) * t ^ (-(2 : ℝ) / 3)

  fun h => laplacianCubeRoot_unsquared_false (h 1 (0, 0) one_pos)
