#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-HW statements against copies of their sanity theorems.

As the batch-FR generator (docs/replay/plants/2026-11-16-FR/plants_fr.py.txt). The batch is not yet a module of the
library, so every generated file is self-contained: imports, the statement text copied verbatim from
HadamardWalk-statements.lean, the unconditional data lemmas of HadamardWalkSanity.lean that the file needs (copied
verbatim, selected by name closure), then, under the prefix P_, a copy of one statement, of every definition of the
statement file between the mutated one and the statement (hadamardStep, hadamardWalk, hadamardDispersion,
hadamardKernel), of the data lemmas and checks of the sanity file that mention a copied name, and of the sanity
theorems that go through the statement constant. Exactly one mutation (one textual pattern, replaced at each of its
recorded occurrences, or one exchange of two patterns) is applied to one copied def. The unplanted copy (ok-) must
compile. Witness files (docs/lessons.md, rule 9): a compiled proof of the negation of the planted statement from
unconditional data lemmas and from data at t <= 2 of the planted definitions, proved inside the witness.
No file proves, or uses a proof of, a statement of the batch.
Verdicts: DETECTED (witness compiled), NOT DETECTED (no witness), TRUE WEAKENING (cannot be refuted).
Usage: plants_hw.py REPO SANITY_DIR OUTDIR SCRATCHDIR [plant ids]
"""
import os, re, subprocess, sys

REPO, SDIR, OUT, SCR = sys.argv[1:5]
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT_TEXT = open(os.path.join(SDIR, 'HadamardWalk-statements.lean')).read()
i0, i1 = STMT_TEXT.index('namespace OscInt.Statements'), STMT_TEXT.index('end OscInt.Statements')
STMT_TEXT = STMT_TEXT[i0:i1] + 'end OscInt.Statements\n'
STMT = STMT_TEXT.split('\n')
SAN_ALL = open(os.path.join(SDIR, 'HadamardWalkSanity.lean')).read()
assert STMT_TEXT in SAN_ALL, 'the sanity file does not carry the statement text verbatim'
SAN_BODY = SAN_ALL[SAN_ALL.index('namespace OscInt.API.HadamardWalkSanity'):]
SAN = SAN_BODY.split('\n')
R, K, D, L = ('HadamardKernelStatement', 'HadamardKernelBoundStatement', 'HadamardDecayStatement',
              'HadamardDispersiveStatement')
STEP, DISP, KER = 'hadamardStep', 'hadamardDispersion', 'hadamardKernel'
STMT_DEFS = [STEP, 'hadamardWalk', 'pointState', DISP, KER, R, K, D, L]
# Sanity theorems through each statement constant, and the unconditional checks of the same instances.
THMS = {R: ['hwR_zero', 'hwR_zero_table', 'hwR_one', 'hwR_one_table'],
        K: ['hwK_one', 'hwK_one_simplified', 'hwK_two', 'hwK_eight', 'hwK_eight_simplified'],
        D: ['hwD_one', 'hwD_two'], L: ['hwL_point', 'hwL_two_site', 'hwL_five']}
CHECKS = {R: ['hwR_zero_check', 'hwR_one_check'], K: ['hwK_check', 'hwK_zero_time_true'],
          D: ['hwD_one_check', 'hwD_two_check', 'hwD_zero_time_false'],
          L: ['hwL_two_site_check', 'hwL_const_false']}
DECL = re.compile(r'^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9\']+)')
WORD = re.compile(r"[A-Za-z_][A-Za-z_0-9']*")


def block(lines, name):
    """The declaration `name` without its docstring, up to the next blank line."""
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    """All declarations of the sanity file with docstrings, in file order: name -> (index, text)."""
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()
TEXT = {d: block(STMT, d) for d in STMT_DEFS}
TEXT.update({n: block(SAN, n) for n in DATA})
ORDER = STMT_DEFS + sorted(DATA, key=lambda n: DATA[n][0])


def words(text):
    return set(WORD.findall(text))


def dependents(target):
    """Every statement def and sanity declaration that mentions `target`, transitively (with `target`)."""
    res, grew = {target}, True
    while grew:
        grew = False
        for n in ORDER:
            if n not in res and words(TEXT[n]) & res:
                res.add(n)
                grew = True
    return res


def cone(names):
    """`names` and every statement def and sanity declaration they mention, transitively."""
    res, todo = set(), list(names)
    while todo:
        n = todo.pop()
        if n in res:
            continue
        res.add(n)
        todo += [w for w in words(TEXT[n]) if w in TEXT and w not in res]
    return res


def closure(text, exclude):
    """Unconditional declarations of the sanity file used by `text`, transitively, verbatim, in file order."""
    need, todo = set(), [text]
    while todo:
        t = todo.pop()
        for w in words(t):
            if w in DATA and w not in need:
                assert w not in exclude, f'the text uses {w}, which depends on the mutated definition'
                need.add(w)
                todo.append(DATA[w][1])
    return [DATA[n][1] for n in sorted(need, key=lambda n: DATA[n][0])]


HEADER = """import OscInt.Statements.Cusp
import OscInt.Front.GlobalSanity

"""
OPEN = """
namespace OscInt.API.HadamardWalkSanity

open OscInt.Statements OscInt.API.GlobalSanity
"""


def mutate(text, mut):
    """Apply the replacements of one mutation in order; each pattern must occur exactly `count` times."""
    for old, new, count in mut:
        assert text.count(old) == count, (old, text.count(old))
        text = text.replace(old, new)
    return text


def make(stmt, target, mut, planted, witness=None):
    dep = dependents(target)
    if witness is None:
        keep = cone(THMS[stmt] + (CHECKS[stmt] if target != stmt else [])) | {stmt}
    else:
        keep = cone([stmt])
    renamed = [n for n in ORDER if n in dep and n in keep]
    assert target in renamed and stmt in renamed
    parts = []
    for n in renamed:
        t = TEXT[n]
        if planted and n == target:
            t = mutate(t, mut)
        parts.append(t)
    if witness is not None:
        parts.append(f'theorem witness_not_{stmt} : ¬ {stmt} :=\n  {witness}')
    text = '\n\n'.join(parts)
    for n in sorted(renamed, key=len, reverse=True):
        text = re.sub(r'(?<![\w.])' + re.escape(n) + r"(?![\w'])", 'P_' + n, text)
    data = closure(text, dep)
    return (HEADER + STMT_TEXT + OPEN + '\n' + '\n\n'.join(data + [text]) + '\n\nend OscInt.API.HadamardWalkSanity\n')


def run(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
    r = subprocess.run(['nice', '-n', '10', 'lake', 'env', 'lean', '-DautoImplicit=false', os.path.abspath(src)],
                       cwd=REPO, capture_output=True, text=True)
    out = r.stdout + r.stderr
    open(os.path.join(OUT, tag + '.out'), 'w').write(out)
    errs = [l for l in out.split('\n') if ': error' in l]
    return ('accepted' if r.returncode == 0 and not errs else 'rejected'), errs

ONLY = sys.argv[5:]
jobs = []
for pid, stmt, target, mut, witness, note in PLANTS:
    if ONLY and pid not in ONLY:
        continue
    if not os.environ.get('HW_WITNESS_ONLY'):  # development switch; the recorded run does not set it
        jobs.append((pid, 'ok-' + pid, make(stmt, target, mut, False), 'accepted'))
        jobs.append((pid, 'plant-' + pid, make(stmt, target, mut, True), None))
    if witness:
        jobs.append((pid, 'witness-' + pid, make(stmt, target, mut, True, witness), 'accepted'))
results = [run(j[1], j[2]) for j in jobs]

by_pid = {}
for (pid, tag, _, w), (verdict, errs) in zip(jobs, results):
    line = f"{tag}: {verdict}" + (f" (want {w}){'' if verdict == w else '  <-- MISMATCH'}" if w else ' (recorded)')
    if errs:
        line += '\n    first error: ' + errs[0].split('/')[-1][:160]
    by_pid.setdefault(pid, {})[tag.split('-')[0]] = (verdict, line)
lines = []
for pid, stmt, target, mut, witness, note in PLANTS:
    if pid not in by_pid:
        continue
    r = by_pid[pid]
    lines += [r[k][1] for k in ('ok', 'plant', 'witness') if k in r]
    if witness:
        v = ('DETECTED (witness compiled: ¬ planted statement)' if r['witness'][0] == 'accepted'
             else 'WITNESS FAILED  <-- MISMATCH')
    elif pid in TRUE_WEAKENING:
        v = 'TRUE WEAKENING: ' + TRUE_WEAKENING[pid]
    else:
        v = 'NOT DETECTED: ' + NOT_DETECTED[pid]
    lines.append(f'  {pid} [{note}]: {v}')
open(os.path.join(OUT, 'SUMMARY.txt' if not ONLY else 'SUMMARY-partial.txt'), 'w').write('\n'.join(lines) + '\n')
print('\n'.join(lines))
