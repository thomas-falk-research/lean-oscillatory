/-- One step of the Hadamard walk on `ℤ`: the coin `[[1, 1], [1, −1]] / √2` at every site, then the first component
moves one site to the left and the second one site to the right. -/
noncomputable def hadamardStep (ψ : ℤ → ℂ × ℂ) (x : ℤ) : ℂ × ℂ :=
  (((ψ (x + 1)).1 + (ψ (x + 1)).2) / ((Real.sqrt 2 : ℝ) : ℂ),
    ((ψ (x - 1)).1 - (ψ (x - 1)).2) / ((Real.sqrt 2 : ℝ) : ℂ))

/-- `t` steps of the Hadamard walk from `ψ₀`. -/
noncomputable def hadamardWalk (ψ₀ : ℤ → ℂ × ℂ) (t : ℕ) : ℤ → ℂ × ℂ :=
  hadamardStep^[t] ψ₀

/-- The state `(a, b)` at the origin and `0` at every other site. -/
def pointState (a b : ℂ) (y : ℤ) : ℂ × ℂ :=
  if y = 0 then (a, b) else (0, 0)

/-- The dispersion relation of the Hadamard walk, `ω(k) = arcsin(sin k / √2)`. -/
noncomputable def hadamardDispersion (k : ℝ) : ℝ :=
  Real.arcsin (Real.sin k / Real.sqrt 2)

/-- The scalar kernel of the Hadamard walk,
`(2π)^{−1} ∫_{−π}^{π} (e^{−itω(k)} − (−1)^t e^{itω(k)}) / √(2 (1 + cos² k)) · e^{−ixk} dk`. -/
noncomputable def hadamardKernel (t : ℕ) (x : ℤ) : ℂ :=
  ((2 * Real.pi : ℝ) : ℂ)⁻¹ * ∫ k in (-Real.pi)..Real.pi,
    (Complex.exp (↑(-((t : ℝ) * hadamardDispersion k)) * Complex.I) -
        (-1 : ℂ) ^ t * Complex.exp (↑((t : ℝ) * hadamardDispersion k) * Complex.I)) /
      ((Real.sqrt (2 * (1 + Real.cos k ^ 2)) : ℝ) : ℂ) *
    Complex.exp (↑(-((x : ℝ) * k)) * Complex.I)

/-- **HW-R** The walk from a point state in terms of the scalar kernel: for every `t ≥ 0`,
`ψ_{t+1}(x) = ((a + b)/√2 · K(t+1, x+1) + a K(t, x), (a − b)/√2 · K(t+1, x−1) + b K(t, x))`. -/
def HadamardKernelStatement : Prop :=
  ∀ (a b : ℂ) (t : ℕ) (x : ℤ),
    hadamardWalk (pointState a b) (t + 1) x =
      ((a + b) / ((Real.sqrt 2 : ℝ) : ℂ) * hadamardKernel (t + 1) (x + 1) + a * hadamardKernel t x,
        (a - b) / ((Real.sqrt 2 : ℝ) : ℂ) * hadamardKernel (t + 1) (x - 1) + b * hadamardKernel t x)

theorem statement_pin_HadamardKernel : HadamardKernelStatement = HadamardKernelStatement := rfl

/-- **HW-K** Explicit bound for the scalar kernel: `|K(t, x)| ≤ 7 t^{−1/2} + 12 t^{−1/3}` for `t ≥ 1` and every
site `x`. -/
def HadamardKernelBoundStatement : Prop :=
  ∀ (t : ℕ) (x : ℤ), 1 ≤ t →
    ‖hadamardKernel t x‖ ≤ 7 * (t : ℝ) ^ (-(1 : ℝ) / 2) + 12 * (t : ℝ) ^ (-(1 : ℝ) / 3)

theorem statement_pin_HadamardKernelBound : HadamardKernelBoundStatement = HadamardKernelBoundStatement := rfl

/-- **HW-D** Decay of the Hadamard walk from a point state: each component of `ψ_t(x)` is at most
`39 (|a| + |b|) t^{−1/3}`, for `t ≥ 1` and every site `x`. -/
def HadamardDecayStatement : Prop :=
  ∀ (a b : ℂ) (t : ℕ) (x : ℤ), 1 ≤ t →
    ‖(hadamardWalk (pointState a b) t x).1‖ ≤ 39 * (‖a‖ + ‖b‖) * (t : ℝ) ^ (-(1 : ℝ) / 3) ∧
    ‖(hadamardWalk (pointState a b) t x).2‖ ≤ 39 * (‖a‖ + ‖b‖) * (t : ℝ) ^ (-(1 : ℝ) / 3)

theorem statement_pin_HadamardDecay : HadamardDecayStatement = HadamardDecayStatement := rfl

/-- **HW-L** The dispersive estimate `ℓ¹ → ℓ^∞` for finitely supported initial states: each component of `ψ_t(x)` is
at most `39 t^{−1/3} ∑_y (|ψ₀(y)₁| + |ψ₀(y)₂|)`, for `t ≥ 1` and every site `x`. -/
def HadamardDispersiveStatement : Prop :=
  ∀ (ψ₀ : ℤ → ℂ × ℂ) (t : ℕ) (x : ℤ), {y : ℤ | ψ₀ y ≠ (0, 0)}.Finite → 1 ≤ t →
    ‖(hadamardWalk ψ₀ t x).1‖ ≤
      39 * (∑ᶠ y : ℤ, (‖(ψ₀ y).1‖ + ‖(ψ₀ y).2‖)) * (t : ℝ) ^ (-(1 : ℝ) / 3) ∧
    ‖(hadamardWalk ψ₀ t x).2‖ ≤
      39 * (∑ᶠ y : ℤ, (‖(ψ₀ y).1‖ + ‖(ψ₀ y).2‖)) * (t : ℝ) ^ (-(1 : ℝ) / 3)

theorem statement_pin_HadamardDispersive : HadamardDispersiveStatement = HadamardDispersiveStatement := rfl

end OscInt.Statements

namespace OscInt.API.HadamardWalkSanity

open OscInt.Statements OscInt.API.GlobalSanity

/-- `√2 ≠ 0` as a complex number. -/
theorem sqrt_two_ne_zero : ((Real.sqrt 2 : ℝ) : ℂ) ≠ 0 :=
  Complex.ofReal_ne_zero.mpr (Real.sqrt_pos.mpr (by norm_num)).ne'

/-- `√2 · √2 = 2` as complex numbers. -/
theorem sqrt_two_mul_self : ((Real.sqrt 2 : ℝ) : ℂ) * ((Real.sqrt 2 : ℝ) : ℂ) = 2 := by
  rw [← Complex.ofReal_mul, Real.mul_self_sqrt (by norm_num)]; norm_num

/-- **Datum: one step from a point state, all sites.** `((a + b)/√2, 0)` at `−1`, `(0, (a − b)/√2)` at `1`, and
`(0, 0)` elsewhere: the first component has moved to the left. -/
theorem hadamardWalk_point_one (a b : ℂ) (x : ℤ) :
    hadamardWalk (pointState a b) 1 x =
      if x = -1 then ((a + b) / ((Real.sqrt 2 : ℝ) : ℂ), 0)
      else if x = 1 then (0, (a - b) / ((Real.sqrt 2 : ℝ) : ℂ)) else (0, 0) := by
  have e : hadamardWalk (pointState a b) 1 x = hadamardStep (pointState a b) x := rfl
  rw [e, hadamardStep, pointState, pointState]
  by_cases h1 : x = -1
  · subst h1; simp
  · by_cases h2 : x = 1
    · subst h2; simp
    · rw [ite_eq_right (by omega), ite_eq_right (by omega), ite_eq_right h1, ite_eq_right h2]; simp

/-- `(sin k / √2)² = sin² k / 2`. -/
theorem sin_div_sqrt_two_sq (k : ℝ) : (Real.sin k / Real.sqrt 2) ^ 2 = Real.sin k ^ 2 / 2 := by
  rw [div_pow, Real.sq_sqrt (by norm_num)]

/-- **Datum: `cos ω(k) = √(1 − sin² k / 2)`.** -/
theorem cos_hadamardDispersion (k : ℝ) :
    Real.cos (hadamardDispersion k) = Real.sqrt (1 - Real.sin k ^ 2 / 2) := by
  rw [hadamardDispersion, Real.cos_arcsin, sin_div_sqrt_two_sq]

/-- **Datum: `2 cos ω(k) = √(2 (1 + cos² k))`**, the denominator of the kernel. -/
theorem two_cos_hadamardDispersion (k : ℝ) :
    2 * Real.cos (hadamardDispersion k) = Real.sqrt (2 * (1 + Real.cos k ^ 2)) := by
  have h : 2 * (1 + Real.cos k ^ 2) = 2 ^ 2 * (1 - Real.sin k ^ 2 / 2) := by
    linarith [Real.sin_sq_add_cos_sq k]
  rw [cos_hadamardDispersion, h, Real.sqrt_mul (by positivity), Real.sqrt_sq (by norm_num)]

/-- `e^{−iθ} + e^{iθ} = 2 cos θ`. -/
theorem exp_neg_add_exp (θ : ℝ) :
    Complex.exp (↑(-θ) * Complex.I) + Complex.exp (↑θ * Complex.I) = ((2 * Real.cos θ : ℝ) : ℂ) := by
  rw [Complex.exp_mul_I, Complex.exp_mul_I]
  push_cast
  rw [Complex.cos_neg, Complex.sin_neg]
  ring

/-- `√(2 (1 + cos² k)) = √2 · √(1 + cos² k)`. -/
theorem sqrt_two_mul_amplitude (k : ℝ) :
    Real.sqrt (2 * (1 + Real.cos k ^ 2)) = Real.sqrt 2 * Real.sqrt (1 + Real.cos k ^ 2) :=
  Real.sqrt_mul (by norm_num) _

/-- **Datum: the integral of the integrand of `K(1, 0)`**, `∫_{−π}^{π} e^{−i·0·k} dk = 2π`. -/
theorem integral_site_zero :
    ∫ k in (-Real.pi)..Real.pi, Complex.exp (↑(-(((0 : ℤ) : ℝ) * k)) * Complex.I) = ((2 * Real.pi : ℝ) : ℂ) := by
  simp only [Int.cast_zero, zero_mul, neg_zero, Complex.ofReal_zero, Complex.exp_zero,
    intervalIntegral.integral_const]
  rw [show Real.pi - -Real.pi = 2 * Real.pi by ring, Complex.real_smul, mul_one]

noncomputable def P_hadamardKernel (t : ℕ) (x : ℤ) : ℂ :=
  ((2 * Real.pi : ℝ) : ℂ)⁻¹ * ∫ k in (-Real.pi)..Real.pi,
    (Complex.exp (↑(-((t : ℝ) * hadamardDispersion k)) * Complex.I) -
        (-1 : ℂ) ^ t * Complex.exp (↑((t : ℝ) * hadamardDispersion k) * Complex.I)) /
      ((Real.sqrt (1 + Real.cos k ^ 2) : ℝ) : ℂ) *
    Complex.exp (↑(-((x : ℝ) * k)) * Complex.I)

def P_HadamardKernelStatement : Prop :=
  ∀ (a b : ℂ) (t : ℕ) (x : ℤ),
    hadamardWalk (pointState a b) (t + 1) x =
      ((a + b) / ((Real.sqrt 2 : ℝ) : ℂ) * P_hadamardKernel (t + 1) (x + 1) + a * P_hadamardKernel t x,
        (a - b) / ((Real.sqrt 2 : ℝ) : ℂ) * P_hadamardKernel (t + 1) (x - 1) + b * P_hadamardKernel t x)

theorem witness_not_HadamardKernelStatement : ¬ P_HadamardKernelStatement :=
  by
  intro h
  have hpi : ((2 * Real.pi : ℝ) : ℂ) ≠ 0 := by exact_mod_cast (by positivity : (2 * Real.pi : ℝ) ≠ 0)
  have q : ∀ k : ℝ, (Complex.exp (↑(-(((1 : ℕ) : ℝ) * hadamardDispersion k)) * Complex.I) -
      (-1 : ℂ) ^ 1 * Complex.exp (↑(((1 : ℕ) : ℝ) * hadamardDispersion k) * Complex.I)) /
        ((Real.sqrt (1 + Real.cos k ^ 2) : ℝ) : ℂ) = ((Real.sqrt 2 : ℝ) : ℂ) := by
    intro k
    have hd : ((Real.sqrt (1 + Real.cos k ^ 2) : ℝ) : ℂ) ≠ 0 :=
      Complex.ofReal_ne_zero.mpr (Real.sqrt_pos.mpr (by positivity)).ne'
    rw [pow_one, neg_one_mul, sub_neg_eq_add, exp_neg_add_exp, Nat.cast_one, one_mul, two_cos_hadamardDispersion,
      sqrt_two_mul_amplitude, Complex.ofReal_mul, mul_div_assoc, div_self hd, mul_one]
  have k0 : P_hadamardKernel 0 (-1) = 0 := by simp [P_hadamardKernel]
  have k1 : P_hadamardKernel 1 0 = ((Real.sqrt 2 : ℝ) : ℂ) := by
    rw [P_hadamardKernel]
    simp only [q]
    rw [intervalIntegral.integral_const_mul, integral_site_zero]
    field_simp
  have h1 := congrArg Prod.fst (h 1 0 0 (-1))
  rw [zero_add, neg_add_cancel, hadamardWalk_point_one, ite_eq_left rfl, k0, k1] at h1
  have h2 : (1 : ℂ) / ((Real.sqrt 2 : ℝ) : ℂ) * 1 = 1 / ((Real.sqrt 2 : ℝ) : ℂ) * ((Real.sqrt 2 : ℝ) : ℂ) := by
    linear_combination h1
  have h3 := mul_left_cancel₀ (div_ne_zero one_ne_zero sqrt_two_ne_zero) h2
  have h4 := sqrt_two_mul_self
  rw [← h3] at h4
  norm_num at h4

end OscInt.API.HadamardWalkSanity
