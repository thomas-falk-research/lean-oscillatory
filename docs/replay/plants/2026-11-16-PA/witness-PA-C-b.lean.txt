import OscInt.Front.PeriodicAnalyticSanity

namespace OscInt.API.PeriodicAnalyticSanity

open OscInt.Statements OscInt.API.GlobalSanity OscInt.API.LatticeExplicitSanity OscInt.API.LatticeCubeRootSanity

def P_FourierCoeffStripStatement : Prop :=
  ∀ (v : ℂ → ℂ) (a M : ℝ) (k : ℤ), 0 < a → PeriodicAnalyticOn v a M →
    ‖∫ θ in (0 : ℝ)..(2 * Real.pi), v (θ : ℂ) * Complex.exp (↑(-((k : ℝ) * θ)) * Complex.I)‖ ≤
      2 * Real.pi * M * Real.exp (-(2 * a * |(k : ℝ)|))

theorem witness_not_FourierCoeffStripStatement : ¬ P_FourierCoeffStripStatement :=
  by
  intro h
  have h1 := h cos3 (1 / 2) (7 * Real.cosh (3 * (1 / 2))) 3 (by norm_num)
    (cos3_periodicAnalyticOn (1 / 2) (by norm_num))
  rw [integral_cos3_three, norm_mul_pi, Complex.norm_ofNat, Int.cast_ofNat, abs_of_pos (by norm_num),
    show (3 : ℝ) * (1 / 2) = 3 / 2 by norm_num,
    show -((2 : ℝ) * (1 / 2) * 3) = -(3 / 2) + -(3 / 2) by norm_num, Real.exp_add, Real.cosh_eq] at h1
  have hx : (5 / 2 : ℝ) ≤ Real.exp (3 / 2) := by linarith [Real.add_one_le_exp (3 / 2 : ℝ)]
  have hy : Real.exp (-(3 / 2)) * Real.exp (3 / 2) = 1 := by rw [← Real.exp_add]; norm_num
  have hy0 := Real.exp_pos (-(3 / 2))
  have hpi := Real.pi_pos
  have hy25 : Real.exp (-(3 / 2)) ≤ 2 / 5 := by nlinarith
  have h2 : 2 * Real.pi * (7 * ((Real.exp (3 / 2) + Real.exp (-(3 / 2))) / 2)) *
      (Real.exp (-(3 / 2)) * Real.exp (-(3 / 2))) =
      7 * Real.pi * (Real.exp (-(3 / 2)) + Real.exp (-(3 / 2)) ^ 3) := by
    linear_combination (7 * Real.pi * Real.exp (-(3 / 2))) * hy
  rw [h2] at h1
  have h3 : Real.exp (-(3 / 2)) ^ 3 ≤ (2 / 5) ^ 3 := pow_le_pow_left₀ hy0.le hy25 3
  have h4 : Real.exp (-(3 / 2)) + Real.exp (-(3 / 2)) ^ 3 ≤ 58 / 125 := by linarith
  nlinarith [mul_le_mul_of_nonneg_left h4 hpi.le]

end OscInt.API.PeriodicAnalyticSanity
