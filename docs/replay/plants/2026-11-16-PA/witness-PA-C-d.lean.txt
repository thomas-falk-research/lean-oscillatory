import OscInt.Front.PeriodicAnalyticSanity

namespace OscInt.API.PeriodicAnalyticSanity

open OscInt.Statements OscInt.API.GlobalSanity OscInt.API.LatticeExplicitSanity OscInt.API.LatticeCubeRootSanity

def P_PeriodicAnalyticOn (v : ℂ → ℂ) (a M : ℝ) : Prop :=
  DifferentiableOn ℂ v (openStrip a) ∧
    ∀ z ∈ openStrip a, ‖v z‖ ≤ M

def P_FourierCoeffStripStatement : Prop :=
  ∀ (v : ℂ → ℂ) (a M : ℝ) (k : ℤ), 0 < a → P_PeriodicAnalyticOn v a M →
    ‖∫ θ in (0 : ℝ)..(2 * Real.pi), v (θ : ℂ) * Complex.exp (↑(-((k : ℝ) * θ)) * Complex.I)‖ ≤
      2 * Real.pi * M * Real.exp (-(a * |(k : ℝ)|))

theorem witness_not_FourierCoeffStripStatement : ¬ P_FourierCoeffStripStatement :=
  by
  intro h
  have h1 := h halfExp 10 (Real.exp 5) 1 (by norm_num) ⟨halfExp_differentiable.differentiableOn, norm_halfExp_le⟩
  rw [integral_halfExp_one, norm_neg, norm_mul, Complex.norm_ofNat, Complex.norm_I, mul_one, Int.cast_one, abs_one,
    mul_one, mul_assoc, mul_assoc, ← Real.exp_add, show (5 : ℝ) + -10 = -5 by norm_num] at h1
  have hy : Real.exp (-5) * Real.exp 5 = 1 := by rw [← Real.exp_add]; norm_num
  have he : (6 : ℝ) ≤ Real.exp 5 := by linarith [Real.add_one_le_exp (5 : ℝ)]
  have hy0 := Real.exp_pos (-5)
  have hy6 : Real.exp (-5) ≤ 1 / 6 := by nlinarith
  have hpi := Real.pi_pos
  have hpi4 := Real.pi_le_four
  nlinarith [mul_le_mul_of_nonneg_left hy6 hpi.le]

end OscInt.API.PeriodicAnalyticSanity
