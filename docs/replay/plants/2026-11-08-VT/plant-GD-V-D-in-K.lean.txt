import OscInt.Front.VectorSanity

namespace OscInt.API.VectorSanity

open OscInt.Statements OscInt.API.GlobalSanity OscInt.API.ConvertSanity
open scoped ContDiff

def P_GlobalDecayVecStatement : Prop :=
  ∀ (Ψ : ℝ × ℝ → ℝ) (K U : Set (ℝ × ℝ)) (D : Finset (ℝ × ℝ)),
    IsCompact K → IsOpen U → K ⊆ U → (↑D : Set (ℝ × ℝ)) ⊆ K → ContDiffOn ℝ 4 Ψ U → Classified Ψ K D →
    ∃ C : ℝ, ∀ (E : Type) [NormedAddCommGroup E] [NormedSpace ℂ E] [CompleteSpace E]
      (A : ℝ × ℝ → E) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ K → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (Ψ k + L k)) * Complex.I) • A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3)

theorem P_globalDecay_of_vec (h : P_GlobalDecayVecStatement) : GlobalDecayStatement := by
  intro Ψ K U D hK hU hKU hDU hΨ hcl
  obtain ⟨C, hC⟩ := h Ψ K U D hK hU hKU hDU hΨ hcl
  exact ⟨C, fun A M L m hA hs hM hm => by simpa [smul_eq_mul] using hC ℂ A M L m hA hs hM hm⟩

end OscInt.API.VectorSanity
