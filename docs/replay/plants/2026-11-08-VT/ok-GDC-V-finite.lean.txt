import OscInt.Front.VectorSanity

namespace OscInt.API.VectorSanity

open OscInt.Statements OscInt.API.GlobalSanity OscInt.API.ConvertSanity
open scoped ContDiff

def P_GlobalDecayCritVecStatement : Prop :=
  ∀ (Ψ : ℝ × ℝ → ℝ) (K U : Set (ℝ × ℝ)) (K' : Set ((ℝ × ℝ) →L[ℝ] ℝ)) (D : Finset (ℝ × ℝ)),
    IsCompact K → IsOpen U → K ⊆ U → (↑D : Set (ℝ × ℝ)) ⊆ U → IsCompact K' → ContDiffOn ℝ 4 Ψ U →
    ClassifiedCrit Ψ K K' D →
    ∃ C : ℝ, ∀ (E : Type) [NormedAddCommGroup E] [NormedSpace ℂ E] [CompleteSpace E]
      (A : ℝ × ℝ → E) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ K → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → L ∈ K' → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (Ψ k + L k)) * Complex.I) • A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3)

theorem P_globalDecayCrit_of_vec (h : P_GlobalDecayCritVecStatement) : GlobalDecayCritStatement := by
  intro Ψ K U K' D hK hU hKU hDU hK' hΨ hcl
  obtain ⟨C, hC⟩ := h Ψ K U K' D hK hU hKU hDU hK' hΨ hcl
  exact ⟨C, fun A M L m hA hs hM hL hm => by simpa [smul_eq_mul] using hC ℂ A M L m hA hs hM hL hm⟩

end OscInt.API.VectorSanity
