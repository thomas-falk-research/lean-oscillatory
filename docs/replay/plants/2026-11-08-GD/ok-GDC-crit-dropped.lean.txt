open scoped ContDiff

def P_ClassifiedCrit (Ψ : ℝ × ℝ → ℝ) (K : Set (ℝ × ℝ)) (K' : Set ((ℝ × ℝ) →L[ℝ] ℝ)) (D : Finset (ℝ × ℝ)) : Prop :=
  (∀ L ∈ K', ∀ k ∈ K, k ∉ D → fderiv ℝ Ψ k + L = 0 → FoldOrNondegMatrix Ψ k) ∧ ∀ k₀ ∈ D, IsD4Point Ψ k₀

def P_GlobalDecayCritStatement : Prop :=
  ∀ (Ψ : ℝ × ℝ → ℝ) (K U : Set (ℝ × ℝ)) (K' : Set ((ℝ × ℝ) →L[ℝ] ℝ)) (D : Finset (ℝ × ℝ)),
    IsCompact K → IsOpen U → K ⊆ U → (↑D : Set (ℝ × ℝ)) ⊆ U → IsCompact K' → ContDiffOn ℝ 4 Ψ U →
    P_ClassifiedCrit Ψ K K' D →
    ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ K → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → L ∈ K' → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (Ψ k + L k)) * Complex.I) * A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3)

theorem P_globalDecayCrit_hyp_crit (h : P_GlobalDecayCritStatement) :
    ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ Metric.closedBall 0 2 → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) →
      L ∈ ({L₃} : Set ((ℝ × ℝ) →L[ℝ] ℝ)) → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (hypPhase k + L k)) * Complex.I) * A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3) :=
  h hypPhase (Metric.closedBall 0 2) Set.univ {L₃} ∅ (isCompact_closedBall _ _) isOpen_univ
    (Set.subset_univ _) (by simp) isCompact_singleton (genCubic_contDiff 1 0 0 1).contDiffOn
    ⟨fun L hL k _ _ hc => by
      rw [Set.mem_singleton_iff] at hL; subst hL; exact Or.inl (hyp_crit_nondeg k hc), by simp⟩

end OscInt.API.GlobalSanity
