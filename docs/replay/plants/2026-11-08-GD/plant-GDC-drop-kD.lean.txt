def P_ClassifiedCrit (Ψ : ℝ × ℝ → ℝ) (K : Set (ℝ × ℝ)) (K' : Set ((ℝ × ℝ) →L[ℝ] ℝ)) (D : Finset (ℝ × ℝ)) : Prop :=
  (∀ L ∈ K', ∀ k ∈ K, fderiv ℝ Ψ k + L = 0 → FoldOrNondegMatrix Ψ k) ∧ ∀ k₀ ∈ D, IsD4Point Ψ k₀

def P_GlobalDecayCritStatement : Prop :=
  ∀ (Ψ : ℝ × ℝ → ℝ) (K U : Set (ℝ × ℝ)) (K' : Set ((ℝ × ℝ) →L[ℝ] ℝ)) (D : Finset (ℝ × ℝ)),
    IsCompact K → IsOpen U → K ⊆ U → (↑D : Set (ℝ × ℝ)) ⊆ U → IsCompact K' → ContDiffOn ℝ 4 Ψ U →
    P_ClassifiedCrit Ψ K K' D →
    ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ K → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → L ∈ K' → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (Ψ k + L k)) * Complex.I) * A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3)

theorem P_globalDecayCrit_ell (h : P_GlobalDecayCritStatement) :
    ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (m : ℝ),
      ContDiff ℝ 2 A → tsupport A ⊆ Metric.closedBall 0 1 → (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) →
      L ∈ ({0, L₁} : Set ((ℝ × ℝ) →L[ℝ] ℝ)) → 1 ≤ m →
      ‖∫ k : ℝ × ℝ, Complex.exp (↑(m * (ellPhase k + L k)) * Complex.I) * A k‖ ≤ C * M * m ^ (-(2 : ℝ) / 3) :=
  h ellPhase (Metric.closedBall 0 1) Set.univ {0, L₁} {0} (isCompact_closedBall _ _) isOpen_univ
    (Set.subset_univ _) (Set.subset_univ _) ((Set.finite_singleton _).insert _).isCompact
    ellPhase_contDiff.contDiffOn
    ⟨fun _ _ k _ hk _ => Or.inl (ell_nondeg k (by simpa using hk)), ell_classified.2⟩

end OscInt.API.GlobalSanity
