/-- PLANTED copy of `splRemainder`: √(2π/λ) ↦ √(π/λ) in the leading term (planted copy of splRemainder). -/
noncomputable def P_splRemainder (g : ℝ → ℂ) (lam : ℝ) : ℂ :=
  (∫ u, g u * Complex.exp (((lam * u ^ 2 / 2 : ℝ) : ℂ) * Complex.I)) -
    g 0 * ((Real.sqrt (Real.pi / lam) : ℝ) : ℂ) * Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I)

/-- Copy of `StationaryLeadingIdentityStatement` over the copied definition. -/
def P_StationaryLeadingIdentityStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    P_splRemainder g lam =
      -Complex.I * ((lam ^ (-(3 : ℝ) / 2) : ℝ) : ℂ) * ∫ t, g₃ t * splKernel (Real.sqrt lam * t)

theorem witness_StationaryLeadingIdentityStatement_not_P (hS : StationaryLeadingIdentityStatement) : ¬ P_StationaryLeadingIdentityStatement := by
  intro h
  have h1 := hS (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  have h2 := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  rw [splRemainder_gauss 1 one_pos] at h1
  have hpos : 0 < Real.sqrt (2 * Real.pi / 100) := Real.sqrt_pos.2 (by have := Real.pi_pos; positivity)
  have e : P_splRemainder (gauss 1) 100 = gaussRoot 1 100 - ((Real.sqrt (Real.pi / 100) : ℝ) : ℂ) *
      Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I) := by
    unfold P_splRemainder
    rw [integral_gauss_mul_exp 1 one_pos, gauss_zero, one_mul]
  rw [e] at h2
  have h3 : splLead 100 = ((Real.sqrt (Real.pi / 100) : ℝ) : ℂ) *
      Complex.exp (((Real.pi / 4 : ℝ) : ℂ) * Complex.I) := by
    linear_combination h2 - h1
  have hn := norm_splLead 100
  rw [h3, norm_mul, Complex.norm_exp_ofReal_mul_I, mul_one, Complex.norm_real, Real.norm_eq_abs,
    abs_of_nonneg (Real.sqrt_nonneg _)] at hn
  have h4 := (Real.sqrt_inj (by have := Real.pi_pos; positivity) (by have := Real.pi_pos; positivity)).1 hn
  linarith [Real.pi_pos]

end OscInt.API.StationaryLeadingSanity
