/-- PLANTED copy of `StationaryLeadingAttainedStatement`: |κ(τ)| − ε ↦ |κ(τ)| + ε. -/
def P_StationaryLeadingAttainedStatement : Prop :=
  ∀ τ ε : ℝ, 0 ≤ τ → 0 < ε →
    ∃ g g₁ g₂ g₃ : ℝ → ℂ, SplAmplitude g g₁ g₂ g₃ ∧ ContDiff ℝ ∞ g ∧ HasCompactSupport g ∧ (∃ x, g x ≠ 0) ∧
      (‖splKernel τ‖ + ε) * (∫ t, ‖g₃ t‖) ≤ ‖splRemainder g 1‖

theorem witness_StationaryLeadingSevenEighthsStatement_not_P (hK : StationaryLeadingSevenEighthsStatement) : ¬ P_StationaryLeadingAttainedStatement := by
  intro h
  obtain ⟨g, g₁, g₂, g₃, hamp, -, -, hne, hle⟩ := h 0 1 le_rfl one_pos
  have hpos := integral_norm_pos_of_splAmplitude hamp hne
  have h1 := hK g g₁ g₂ g₃ 1 hamp one_pos
  rw [Real.one_rpow, mul_one] at h1
  rw [splKernel_zero, norm_zero] at hle
  linarith

end OscInt.API.StationaryLeadingSanity

theorem closed_now : ¬ OscInt.API.StationaryLeadingSanity.P_StationaryLeadingAttainedStatement :=
  OscInt.API.StationaryLeadingSanity.witness_StationaryLeadingSevenEighthsStatement_not_P OscInt.API.stationaryLeadingSevenEighths
