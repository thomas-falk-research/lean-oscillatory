/-- PLANTED copy of `splRemainder`: e^{iπ/4} ↦ 1 in the leading term (planted copy of splRemainder). -/
noncomputable def P_splRemainder (g : ℝ → ℂ) (lam : ℝ) : ℂ :=
  (∫ u, g u * Complex.exp (((lam * u ^ 2 / 2 : ℝ) : ℂ) * Complex.I)) -
    g 0 * ((Real.sqrt (2 * Real.pi / lam) : ℝ) : ℂ) * 1

/-- Copy of `StationaryLeadingStatement` over the copied definition. -/
def P_StationaryLeadingStatement : Prop :=
  ∀ (g g₁ g₂ g₃ : ℝ → ℂ) (lam : ℝ), SplAmplitude g g₁ g₂ g₃ → 0 < lam →
    ∃ τ : ℝ, 0 ≤ τ ∧ ‖P_splRemainder g lam‖ ≤ ‖splKernel τ‖ * (∫ t, ‖g₃ t‖) * lam ^ (-(3 : ℝ) / 2)

theorem witness_not_P_StationaryLeadingStatement (hκ : ∀ τ, ‖splKernel τ‖ ≤ 2) : ¬ P_StationaryLeadingStatement := by
  intro h
  have e : P_splRemainder (gauss 1) 100 = gaussRoot 1 100 - ((Real.sqrt (2 * Real.pi / 100) : ℝ) : ℂ) * 1 := by
    unfold P_splRemainder
    rw [integral_gauss_mul_exp 1 one_pos, gauss_zero, one_mul]
  have him : (P_splRemainder (gauss 1) 100).im = (gaussRoot 1 100).im := by
    rw [e, mul_one, Complex.sub_im, Complex.ofReal_im, sub_zero]
  have hlow : 1 / 10 ≤ ‖P_splRemainder (gauss 1) 100‖ := by
    have h2 := (le_abs_self _).trans (Complex.abs_im_le_norm (P_splRemainder (gauss 1) 100))
    rw [him] at h2
    exact gaussRoot_one_hundred_im_lower.trans h2
  obtain ⟨τ, -, h1⟩ := h (gauss 1) (gauss1 1) (gauss2 1) (gauss3 1) 100 (gauss_splAmplitude 1 one_pos) (by norm_num)
  rw [integral_norm_gauss3 1 one_pos, rpow_hundred, one_pow, one_mul] at h1
  have h2 : ‖splKernel τ‖ * (2 + 8 * Real.exp (-(3 / 2))) ≤ 2 * (379 / 100) :=
    mul_le_mul (hκ τ) normThree_le normThree_pos.le (by norm_num)
  linarith
