/-- PLANTED copy of `MonomialAsymptoticStatement`: the hypothesis 0 < t dropped. -/
def P_MonomialAsymptoticStatement : Prop :=
  ∀ (m : ℕ) (σ : ℝ), 2 ≤ m → (σ = 1 ∨ σ = -1) →
    ∃ C : ℂ, 1 / (12 * (m : ℝ)) ≤ C.re ∧
      Tendsto (fun R : ℝ => ∫ x in -R..R, Complex.exp (↑(σ * x ^ m) * Complex.I)) atTop (𝓝 C) ∧
      ∀ (b : ℝ → ℝ) (r t : ℝ), PlateauCutoff b r →
        ‖(∫ x : ℝ, (b x : ℂ) * Complex.exp (↑(t * σ * x ^ m) * Complex.I)) -
            ((t ^ (-(1 : ℝ) / (m : ℝ)) : ℝ) : ℂ) * C‖ ≤ 4 / ((m : ℝ) * t * r ^ (m - 1))

/-- Witness: at `t = 0` the planted remainder reads `‖∫ b‖ ≤ 0`, but `∫ plat 1 2 ≥ 2` (`ma_time_zero_bite`). -/
theorem witness_MA_c : ¬ P_MonomialAsymptoticStatement := by
  intro h
  obtain ⟨C, -, -, hb⟩ := h 2 1 le_rfl (Or.inl rfl)
  exact ma_time_zero_bite 2 le_rfl 1 C one_pos (by norm_num) (hb (plat 1 2) 1 0 (plat_plateau one_pos (by norm_num)))

#print axioms witness_MA_c
