#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-K2 statements against copies of their sanity theorems.

As the batch-TRI generator (docs/replay/plants/2026-11-27-TRI/plants_tri.py.txt). The draft statement module
(OscInt/Statements/TriangularDecay.lean, not frozen, already built in the repository) is read from a copy
(stmt/TriangularDecay.lean); the sanity layer (src/OscInt/Front/TriangularDecaySanity.lean, namespace
OscInt.API.TriangularDecaySanity) is compiled into a scratch library placed first on LEAN_PATH, so every generated
file imports OscInt.Front.TriangularDecaySanity (and, for one witness, the built OscInt.Front.Core) and contains,
under the prefix P_: a copy of ONE statement constant, with exactly one textual mutation in the planted copy (none in
the ok- copy), followed by copies of every sanity theorem that takes that statement constant as a hypothesis (renamed
P_..., so that they go through the copy). Witness files contain the planted copy and compiled theorems of one kind:
  not   : ¬ planted statement, from unconditional data of the sanity layer          -> DETECTED
  rel   : true statement → ¬ planted statement                                    -> DETECTED RELATIVE TO THE TRUE
                                                                                     STATEMENT
  weak  : draft statement → planted statement                                     -> TRUE WEAKENING
  char  : planted statement → (what a detection would have to refute)             -> NOT DETECTED (characterised)
No file proves, or uses a proof of, a statement of the batch. A rejected script alone is never counted as a
detection. Compiler output is recorded with the scratch directory removed from the paths.
Usage: plants_k2.py WORK OUTDIR SCRATCHDIR
"""
import os, re, subprocess, sys, textwrap, time

WORK, OUT, SCR = [os.path.abspath(a) for a in sys.argv[1:4]]
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT = open(os.path.join(WORK, 'stmt/TriangularDecay.lean')).read().split('\n')
SAN_ALL = open(os.path.join(WORK, 'src/OscInt/Front/TriangularDecaySanity.lean')).read()
SAN = SAN_ALL[SAN_ALL.index('namespace OscInt.API.TriangularDecaySanity'):].split('\n')
L, MA, ML, F, U, D = ('ReducedOrderBoundStatement', 'MonomialAsymptoticStatement', 'MonomialLowerStatement',
                      'TorusLocalizationStatement', 'TriDecayUpperStatement', 'TriDecayLowerStatement')
STATEMENTS = [L, MA, ML, F, U, D]
DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9']+)")
WORD = re.compile(r"[A-Za-z_][A-Za-z_0-9']*")


def block(lines, name):
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()


def words(t):
    ls = t.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    return set(WORD.findall('\n'.join(ls[k:])))


THROUGH = {st: [n for n, (_, t) in sorted(DATA.items(), key=lambda kv: kv[1][0]) if st in words(t)]
           for st in STATEMENTS}
for st in STATEMENTS:
    assert THROUGH[st], st

HEAD = """namespace OscInt.API.TriangularDecaySanity

open OscInt.Statements OscInt.API.Poly OscInt.API.CuspSanity OscInt.API.TriangularAnisoSanity MeasureTheory
open scoped ContDiff Topology
open Filter
"""
CLOSE = '\n\nend OscInt.API.TriangularDecaySanity\n'


def wrap(s):
    return '\n'.join(textwrap.wrap('/-- ' + s + ' -/', 118))


def rename(text, names):
    ls = text.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    doc, decl = '\n'.join(ls[:k]), '\n'.join(ls[k:])
    for n in sorted(names, key=len, reverse=True):
        decl = re.sub(r"(?<![\w.'])" + re.escape(n) + r"(?![\w'])", 'P_' + n, decl)
    return (doc + '\n' if doc else '') + decl


def make(stmt, mut, planted, note, witness=None, imports=()):
    old, new, count = mut
    t = block(STMT, stmt)
    if planted:
        assert t.count(old) == count, (stmt, old, t.count(old))
        t = t.replace(old, new)
    doc = f'PLANTED copy of `{stmt}`: {note}.' if planted else f'Unplanted copy of `{stmt}`.'
    parts = [wrap(doc) + '\n' + t]
    names = [stmt]
    if witness is None:
        parts += [DATA[n][1] for n in THROUGH[stmt]]
        names += THROUGH[stmt]
    text = '\n\n'.join(rename(p, names) for p in parts)
    if witness is not None:
        text += '\n\n' + witness.strip('\n')
        for n in re.findall(r'^theorem ([A-Za-z_0-9]+)', witness, re.M):
            text += f'\n\n#print axioms {n}'
    imp = ''.join(f'import {i}\n' for i in ('OscInt.Front.TriangularDecaySanity',) + tuple(imports))
    return imp + '\n' + HEAD + '\n' + text + CLOSE


def compile_text(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    t0 = time.time()
    r = subprocess.run([os.path.join(WORK, 'lean.sh'), src], capture_output=True, text=True, timeout=1800)
    out = (r.stdout + r.stderr).replace(SCR + os.sep, '').replace(WORK + os.sep, '')
    out = re.sub(r'elapsed [0-9.]+ s rc=\d+\n?', '', out)
    return r.returncode, out, time.time() - t0

    ('D-b', D, ('c * t ^ (-triRate α)', 'c * t ^ (-(3 / 4 : ℝ))', 1), 'triRate ↦ 3/4 everywhere', 'weak', r'''
/-- Witness: the draft implies the planted copy (`triRate α ≤ 3/4`, `t ≥ max T 1`). -/
theorem witness_D_b (h : TriDecayLowerStatement) : P_TriDecayLowerStatement := by
  intro α
  obtain ⟨c, hc, T, hT⟩ := h α
  refine ⟨c, hc, max T 1, fun t ht => ?_⟩
  obtain ⟨n, hn⟩ := hT t (le_of_max_le_left ht)
  refine ⟨n, le_trans ?_ hn⟩
  have h1 : t ^ (-(3 / 4 : ℝ)) ≤ t ^ (-triRate α) :=
    Real.rpow_le_rpow_of_exponent_le (le_of_max_le_right ht)
      (by rcases triRate_cases α with e | e <;> rw [e] <;> norm_num)
  exact mul_le_mul_of_nonneg_left h1 hc.le
''', ()),

VERDICT = {
    'not': 'DETECTED (witness compiled: ¬ planted statement)',
    'rel': 'DETECTED RELATIVE TO THE TRUE STATEMENT (witness compiled: true statement → ¬ planted statement); '
           'NOT DETECTED unconditionally',
    'weak': 'TRUE WEAKENING (compiled: draft statement → planted statement); no datum can refute it',
    'char': 'NOT DETECTED (compiled characterisation of what a detection would have to refute; no refutation)',
}

only = set(os.environ.get('ONLY', '').split(',')) - {''}
log = []
for pid, st, mut, note, kind, wit, imps in PLANTS:
    if only and pid not in only:
        continue
    for tag, planted, w in (('ok-' + pid, False, None), ('plant-' + pid, True, None), ('witness-' + pid, True, wit)):
        text = make(st, mut, planted, note, w, imps if w is not None else ())
        rc, out, dt = compile_text(tag, text)
        want = 'recorded' if tag.startswith('plant-') else 'accepted'
        got = 'accepted' if rc == 0 else 'rejected'
        if w is not None and rc == 0:
            ax = re.findall(r"depends on axioms: \[([^\]]*)\]", out)
            if not ax or any(a.replace(' ', '') != 'propext,Classical.choice,Quot.sound' for a in ax):
                got = 'rejected (axioms)'
        open(os.path.join(OUT, tag + '.lean.txt'), 'w').write(text)
        open(os.path.join(OUT, tag + '.out'), 'w').write(f'exit code {rc}\n' + out)
        line = f'{tag}: {got} (want {want}) [{dt:.1f} s]'
        if rc != 0:
            first = next((l for l in out.split('\n') if 'error' in l), '')
            line += '\n    first error: ' + first[:110]
        log.append(line)
        print(line, flush=True)
        if tag.startswith('witness-'):
            v = VERDICT[kind] if got == 'accepted' else 'NOT DETECTED (witness did not compile)'
            log.append(f'  {pid} [{note}]: {v}')
            print(log[-1], flush=True)
open(os.path.join(OUT, 'SUMMARY.txt' if not only else 'SUMMARY-partial.txt'), 'w').write('\n'.join(log) + '\n')
