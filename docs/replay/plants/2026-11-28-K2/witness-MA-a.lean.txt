/-- PLANTED copy of `MonomialAsymptoticStatement`: the bound Re C ≥ 1/(12m) ↦ Re C ≥ 12/m. -/
def P_MonomialAsymptoticStatement : Prop :=
  ∀ (m : ℕ) (σ : ℝ), 2 ≤ m → (σ = 1 ∨ σ = -1) →
    ∃ C : ℂ, 12 / (m : ℝ) ≤ C.re ∧
      Tendsto (fun R : ℝ => ∫ x in -R..R, Complex.exp (↑(σ * x ^ m) * Complex.I)) atTop (𝓝 C) ∧
      ∀ (b : ℝ → ℝ) (r t : ℝ), PlateauCutoff b r → 0 < t →
        ‖(∫ x : ℝ, (b x : ℂ) * Complex.exp (↑(t * σ * x ^ m) * Complex.I)) -
            ((t ^ (-(1 : ℝ) / (m : ℝ)) : ℝ) : ℂ) * C‖ ≤ 4 / ((m : ℝ) * t * r ^ (m - 1))

/-- Witness: the planted statement's own remainder at `m = 2`, `t = 1`, `b = plat 1 (5/4)` gives
`‖C‖ ≤ 5/2 + 2 < 6 = 12/m` (`norm_integral_plat_le`). -/
theorem witness_MA_a : ¬ P_MonomialAsymptoticStatement := by
  intro h
  obtain ⟨C, hC, -, hb⟩ := h 2 1 le_rfl (Or.inl rfl)
  have h1 := hb (plat 1 (5 / 4)) 1 1 (plat_plateau one_pos (by norm_num)) one_pos
  simp only [Real.one_rpow, Complex.ofReal_one, one_mul, mul_one, one_pow] at h1
  have h2 := norm_integral_plat_le (r := 1) (R := 5 / 4) (by norm_num) (by norm_num) (fun x => x ^ 2)
  beta_reduce at h2
  have h3 := norm_sub_norm_le C (∫ x : ℝ, (plat 1 (5 / 4) x : ℂ) * Complex.exp (↑(x ^ 2) * Complex.I))
  rw [norm_sub_rev] at h3
  have h4 := Complex.re_le_norm C
  have e1 : (12 : ℝ) / ((2 : ℕ) : ℝ) = 6 := by norm_num
  have e2 : (4 : ℝ) / ((2 : ℕ) : ℝ) = 2 := by norm_num
  rw [e1] at hC
  rw [e2] at h1
  linarith

#print axioms witness_MA_a
