/-- PLANTED copy of `TriDecayUpperStatement`: triRate ↦ 3/4 (exceptional branches dropped). -/
def P_TriDecayUpperStatement : Prop :=
  ∀ α : ℝ, ∃ C : ℝ, ∀ (t : ℝ) (n : ℤ × ℤ), 1 ≤ t → ‖triKernel α t n‖ ≤ C * t ^ (-(3 / 4 : ℝ))

/-- Witness (relative to K2-D): at `α = 0` K2-D gives `c t^{−2/3}` at some site, against `C t^{−3/4}`
(`rate_separation`). -/
theorem witness_U_a (hD : TriDecayLowerStatement) : ¬ P_TriDecayUpperStatement := by
  intro h
  obtain ⟨C, hC⟩ := h 0
  obtain ⟨c, hc, T, hT⟩ := hD 0
  rw [triRate_zero] at hT
  obtain ⟨t, hTt, ht1, hlt⟩ := rate_separation (2 / 3) (3 / 4) c C T (by norm_num) hc
  obtain ⟨n, hn⟩ := hT t hTt
  have := hC t n ht1
  linarith

#print axioms witness_U_a

end OscInt.API.TriangularDecaySanity
