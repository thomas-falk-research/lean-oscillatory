import OscInt.Front.Poly
import OscInt.Statements.TriangularAniso

/-! A4 datum for the mutation `Icc 1 5 ↦ Icc 1 4` of `IsA5Point` (finding A2 of the TRI blind review; suggested by an
external reviewer). `Φ(x, y) = y² + x⁵ + x⁶` at `0`, `v = (1, 0)`, `w = (0, 1)`, the reduced curve `h = 0`: the reduced
function is `u⁵ + u⁶`, so its derivatives of orders 1 to 4 vanish, the fifth is 120 and the sixth 720. -/

namespace A4Datum

open OscInt.Statements OscInt.API.Poly
open scoped ContDiff

def Φ (p : ℝ × ℝ) : ℝ := p.2 ^ 2 + p.1 ^ 5 + p.1 ^ 6

theorem Φ_smooth (n : ℕ) : ContDiff ℝ n Φ := by unfold Φ; fun_prop

theorem Φ_line (k v : ℝ × ℝ) (t : ℝ) :
    Φ (k + t • v) = ∑ i : Fin 7, ![Φ k, 2 * k.2 * v.2 + 5 * k.1 ^ 4 * v.1 + 6 * k.1 ^ 5 * v.1,
      v.2 ^ 2 + 10 * k.1 ^ 3 * v.1 ^ 2 + 15 * k.1 ^ 4 * v.1 ^ 2, 10 * k.1 ^ 2 * v.1 ^ 3 + 20 * k.1 ^ 3 * v.1 ^ 3,
      5 * k.1 * v.1 ^ 4 + 15 * k.1 ^ 2 * v.1 ^ 4, v.1 ^ 5 + 6 * k.1 * v.1 ^ 5, v.1 ^ 6] i * t ^ (i : ℕ) := by
  simp [Fin.sum_univ_succ, Φ]; ring

theorem d1 (k v : ℝ × ℝ) : fderiv ℝ Φ k v = 2 * k.2 * v.2 + 5 * k.1 ^ 4 * v.1 + 6 * k.1 ^ 5 * v.1 := by
  have := iteratedFDeriv_diag_of_poly (Φ_smooth 7) k v _ (Φ_line k v) 1 (by norm_num)
  rw [iteratedFDeriv_one_apply] at this
  simpa using this

theorem d2 (v : ℝ × ℝ) : iteratedFDeriv ℝ 2 Φ 0 ![v, v] = 2 * v.2 ^ 2 := by
  have := iteratedFDeriv_diag_of_poly (Φ_smooth 7) 0 v _ (Φ_line 0 v) 2 (by norm_num)
  rw [show (![v, v] : Fin 2 → ℝ × ℝ) = fun _ => v by funext i; fin_cases i <;> rfl, this]
  simp

theorem kernel (z : ℝ × ℝ) : iteratedFDeriv ℝ 2 Φ 0 ![((1 : ℝ), (0 : ℝ)), z] = 0 := by
  rw [iteratedFDeriv_two_polar ((Φ_smooth 2).contDiffAt), d2, d2, d2]
  simp

theorem hess_w : iteratedFDeriv ℝ 2 Φ 0 ![((0 : ℝ), (1 : ℝ)), ((0 : ℝ), (1 : ℝ))] = 2 := by rw [d2]; norm_num

theorem curve : ReducedCurve Φ 0 (1, 0) (0, 1) 1 (fun _ => 0) := by
  refine ⟨contDiff_const, rfl, fun u _ => ?_⟩
  rw [d1, d1]; simp

theorem reduced_eq : reducedFn Φ 0 (1, 0) (0, 1) (fun _ => 0) = fun u => ∑ i : Fin 7, ![0, 0, 0, 0, 0, 1, 1] i * u ^ (i : ℕ) := by
  funext u
  simp only [reducedFn, d1]
  simp [Fin.sum_univ_succ, Φ]

theorem jets : (∀ j ∈ Finset.Icc 1 4, iteratedDeriv j (reducedFn Φ 0 (1, 0) (0, 1) (fun _ => 0)) 0 = 0) ∧
    iteratedDeriv 5 (reducedFn Φ 0 (1, 0) (0, 1) (fun _ => 0)) 0 = 120 ∧
    iteratedDeriv 6 (reducedFn Φ 0 (1, 0) (0, 1) (fun _ => 0)) 0 = 720 := by
  have key := fun n : Fin 7 => iteratedDeriv_zero_of_poly ![(0 : ℝ), 0, 0, 0, 0, 1, 1] _ (fun u => congrFun reduced_eq u) n
  refine ⟨fun j hj => ?_, ?_, ?_⟩
  · simp only [Finset.mem_Icc] at hj
    obtain ⟨h1, h4⟩ := hj
    interval_cases j
    · simpa using key 1
    · simpa using key 2
    · simpa using key 3
    · simpa using key 4
  · simpa [Nat.factorial] using key 5
  · simpa [Nat.factorial] using key 6

/-- With the frozen definition, this witness is not an A5 witness: its fifth derivative is `120 ≠ 0`. -/
theorem frozen_rejects_witness :
    ¬ ∀ j ∈ Finset.Icc 1 5, iteratedDeriv j (reducedFn Φ 0 (1, 0) (0, 1) (fun _ => 0)) 0 = 0 := by
  intro h
  have := h 5 (by simp)
  rw [jets.2.1] at this
  norm_num at this


/-- The A4 point `0` of `Φ` counted as an A5 point. Compiles only if `IsA5Point` asks for orders 1 to 4 (mutation M2);
with the frozen orders 1 to 5 it must fail, because the fifth derivative is 120. -/
theorem accepted_as_A5 : IsA5Point Φ 0 :=
  ⟨(1, 0), (0, 1), by simp, kernel, by rw [hess_w]; norm_num, 1, one_pos, fun _ => 0, curve, jets.1,
    by rw [jets.2.2]; norm_num⟩

#print axioms accepted_as_A5

end A4Datum
