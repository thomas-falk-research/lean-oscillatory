#!/usr/bin/env python3
"""Generate and compile planted copies of the draft batch-TRI statements against copies of their sanity theorems.

As the batch-SPL generator (docs/replay/plants/2026-11-19-SPL/plants_spl.py.txt), simplified. The draft statement
module (a copy of OscInt/Statements/TriangularAniso.lean, not frozen) and the sanity layer
(TriangularAnisoSanity.lean, namespace OscInt.API.TriangularAnisoSanity) are compiled into a scratch library placed
first on LEAN_PATH, so every generated file imports OscInt.Front.TriangularAnisoSanity only and contains, under the
prefix P_: a copy of ONE statement constant, with exactly one textual mutation in the planted copy (none in the ok-
copy), followed by copies of every sanity theorem that takes that statement constant as a hypothesis (renamed
P_..., so that they go through the copy). Witness files contain the planted copy and compiled theorems of one kind:
  not   : ¬ planted statement, from unconditional data of the sanity layer          -> DETECTED
  rel   : true statement → ¬ planted statement                                    -> DETECTED RELATIVE TO THE TRUE
                                                                                     STATEMENT
  weak  : draft statement → planted statement                                     -> TRUE WEAKENING
  char  : draft ∧ planted → (what a detection would need does not exist)          -> NOT DETECTED (characterised)
No file proves, or uses a proof of, a statement of the batch. A rejected script alone is never counted as a
detection. Compiler output is recorded with the scratch directory removed from the paths.
Usage: plants_tri.py WORK OUTDIR SCRATCHDIR
"""
import os, re, subprocess, sys, textwrap, time

WORK, OUT, SCR = [os.path.abspath(a) for a in sys.argv[1:4]]
REPO = '/home/user/lean-oscillatory'
os.makedirs(OUT, exist_ok=True)
os.makedirs(SCR, exist_ok=True)
STMT = open(os.path.join(WORK, 'src/OscInt/Statements/TriangularAniso.lean')).read().split('\n')
SAN_ALL = open(os.path.join(WORK, 'src/OscInt/Front/TriangularAnisoSanity.lean')).read()
SAN = SAN_ALL[SAN_ALL.index('namespace OscInt.API.TriangularAnisoSanity'):].split('\n')
K, X, Wt, Q = ('TriClassificationStatement', 'TriA5LocusStatement', 'TriA5WitnessStatement', 'TriQuarticStatement')
STATEMENTS = [K, X, Wt, Q]
DECL = re.compile(r"^(?:noncomputable )?(theorem|def) ([A-Za-z_0-9']+)")
WORD = re.compile(r"[A-Za-z_][A-Za-z_0-9']*")


def block(lines, name):
    for i, l in enumerate(lines):
        m = DECL.match(l)
        if m and m.group(2) == name:
            j = i
            while j < len(lines) and lines[j].strip() != '':
                j += 1
            return '\n'.join(lines[i:j])
    raise SystemExit(f'not found: {name}')


def data_blocks():
    res, cur = {}, []
    for l in SAN[1:] + ['']:
        if l.strip() == '':
            for c in cur:
                m = DECL.match(c)
                if m:
                    res[m.group(2)] = (len(res), '\n'.join(cur))
                    break
            cur = []
        else:
            cur.append(l)
    return res


DATA = data_blocks()


def words(t):
    ls = t.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    return set(WORD.findall('\n'.join(ls[k:])))


THROUGH = {st: [n for n, (_, t) in sorted(DATA.items(), key=lambda kv: kv[1][0]) if st in words(t)]
           for st in STATEMENTS}
for st in STATEMENTS:
    assert THROUGH[st], st

OPEN = """import OscInt.Front.TriangularAnisoSanity

namespace OscInt.API.TriangularAnisoSanity

open OscInt.Statements OscInt.API.Poly OscInt.API.CuspSanity
open scoped ContDiff
"""
CLOSE = '\n\nend OscInt.API.TriangularAnisoSanity\n'


def wrap(s):
    return '\n'.join(textwrap.wrap('/-- ' + s + ' -/', 118))


def rename(text, names):
    ls = text.split('\n')
    k = next(i for i, l in enumerate(ls) if DECL.match(l))
    doc, decl = '\n'.join(ls[:k]), '\n'.join(ls[k:])
    for n in sorted(names, key=len, reverse=True):
        decl = re.sub(r"(?<![\w.'])" + re.escape(n) + r"(?![\w'])", 'P_' + n, decl)
    return (doc + '\n' if doc else '') + decl


def make(stmt, mut, planted, note, witness=None):
    old, new, count = mut
    t = block(STMT, stmt)
    if planted:
        assert t.count(old) == count, (stmt, old, t.count(old))
        t = t.replace(old, new)
    doc = f'PLANTED copy of `{stmt}`: {note}.' if planted else f'Unplanted copy of `{stmt}`.'
    parts = [wrap(doc) + '\n' + t]
    names = [stmt]
    if witness is None:
        parts += [DATA[n][1] for n in THROUGH[stmt]]
        names += THROUGH[stmt]
    text = '\n\n'.join(rename(p, names) for p in parts)
    if witness is not None:
        text += '\n\n' + witness.strip('\n')
        for n in re.findall(r'^theorem ([A-Za-z_0-9]+)', witness, re.M):
            text += f'\n\n#print axioms {n}'
    return OPEN + '\n' + text + CLOSE


def compile_text(tag, text):
    src = os.path.join(SCR, tag + '.lean')
    open(src, 'w').write(text)
    t0 = time.time()
    r = subprocess.run([os.path.join(WORK, 'lean.sh'), src], capture_output=True, text=True, timeout=900)
    out = (r.stdout + r.stderr).replace(SCR + os.sep, '')
    out = re.sub(r'elapsed [0-9.]+ s rc=\d+\n?', '', out)
    return r.returncode, out, time.time() - t0

VERDICT = {'not': 'DETECTED (witness compiled: ¬ planted statement)',
           'rel': 'DETECTED RELATIVE TO THE TRUE STATEMENT (witness compiled: true statement → ¬ planted statement); '
                  'NOT DETECTED unconditionally',
           'weak': 'TRUE WEAKENING (compiled: draft statement → planted statement); no datum can refute it',
           'char': 'NOT DETECTED (compiled: draft ∧ plant → no reduced curve exists; a detection needs a reduced '
                   'curve with its sixth derivative, not constructed)'}

summary = []
for pid, st, mut, note, kind, wit in PLANTS:
    res = {}
    for tag, planted, w in (('ok', False, None), ('plant', True, None), ('witness', True, wit)):
        name = f'{tag}-{pid}'
        text = make(st, mut, planted, note, w)
        rc, out, dt = compile_text(name, text)
        open(os.path.join(OUT, name + '.lean.txt'), 'w').write(text)
        open(os.path.join(OUT, name + '.out'), 'w').write(f'exit code {rc}\n' + out)
        errs = [l for l in out.split('\n') if ': error' in l]
        status = 'accepted' if rc == 0 and not errs else 'rejected'
        if tag == 'witness':
            ax = re.findall(r"depends on axioms: \[([^\]]*)\]", out)
            okax = all(set(x.strip() for x in a.split(',')) <= {'propext', 'Classical.choice', 'Quot.sound'}
                       for a in ax) and len(ax) == len(re.findall(r'^theorem ', w, re.M))
            if not okax:
                status = 'rejected (axioms)'
        res[tag] = status
        want = {'ok': 'accepted', 'plant': 'recorded', 'witness': 'accepted'}[tag]
        line = f'{name}: {status} (want {want}) [{dt:.1f} s]'
        if errs:
            line += '\n    first error: ' + errs[0][:160]
        summary.append(line)
        print(line, flush=True)
    good = res['ok'] == 'accepted' and res['witness'] == 'accepted'
    summary.append(f'  {pid} [{note}]: ' + (VERDICT[kind] if good else 'NOT DETECTED (ok- or witness file failed)'))
    print(summary[-1], flush=True)
open(os.path.join(OUT, 'SUMMARY.txt'), 'w').write('\n'.join(summary) + '\n')
