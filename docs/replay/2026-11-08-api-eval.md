## T1 (stationary phase, phase k₁² + 2k₂², L = 0)

Route: `stationaryPhaseFE` with K = closedBall 0 1 and K' = {0}. The Hessian is !![2,0;0,4] everywhere (det 8), from
`hessianMatrix_eq_of_diag` plus `iteratedFDeriv_diag_of_poly`.

```lean
import OscInt.API

open OscInt.API.Poly in
theorem T1 : ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M m : ℝ), ContDiff ℝ 2 A →
    tsupport A ⊆ Metric.closedBall 0 1 →
    (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → 1 ≤ m →
    ‖∫ k, Complex.exp (((m * (k.1 ^ 2 + 2 * k.2 ^ 2) : ℝ) : ℂ) * Complex.I) * A k‖ ≤ C * M / m := by
  set Ψ : ℝ × ℝ → ℝ := fun k => k.1 ^ 2 + 2 * k.2 ^ 2 with hΨ
  have hC : ContDiff ℝ 4 Ψ := by fun_prop
  have diag : ∀ k v : ℝ × ℝ, (iteratedFDeriv ℝ 2 Ψ k) ![v, v] = 2 * (v.1 ^ 2 + 2 * v.2 ^ 2) := by
    intro k v
    have h := iteratedFDeriv_diag_of_poly (N := 3) (hC.of_le (by norm_num)) k v
      ![Ψ k, 2 * k.1 * v.1 + 4 * k.2 * v.2, v.1 ^ 2 + 2 * v.2 ^ 2]
      (by intro t; simp [Fin.sum_univ_three, hΨ]; ring) 2 (by norm_num)
    have e : (![v, v] : Fin 2 → ℝ × ℝ) = fun _ => v := by funext i; fin_cases i <;> rfl
    rw [e, h]; simp
  have hH : ∀ k, (OscInt.Statements.hessianMatrix Ψ k).det ≠ 0 := by
    intro k
    rw [hessianMatrix_eq_of_diag (hC.contDiffAt.of_le (by norm_num)) (diag k (1, 0)) (diag k (0, 1))
      (diag k (1, 1))]
    simp [Matrix.det_fin_two]; norm_num
  obtain ⟨C, hC'⟩ := OscInt.API.stationaryPhaseFE Ψ (Metric.closedBall 0 1) {0} hC
    (isCompact_closedBall 0 1) isCompact_singleton (fun L _ k _ _ => hH k)
  refine ⟨C, fun A M m hA hs hM hm => ?_⟩
  have := hC' A M 0 m hA hs hM rfl hm
  simpa only [hΨ, zero_apply, add_zero] using this
```

## T2 (non-stationary, N = 3, phase k₁ + k₂²)

Route: `nonStationaryFE 3` with c = 1 and L = 0. The lower bound `1 ≤ ‖fderiv Ψ k‖` comes from `le_opNorm` at (1, 0);
fderiv comes from an explicit `HasFDerivAt` (fst + snd²).

```lean
import OscInt.API

theorem T2 : ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M m : ℝ), ContDiff ℝ 3 A →
    tsupport A ⊆ Metric.closedBall 0 (1/4) →
    (∀ j ≤ 3, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → 1 ≤ m →
    ‖∫ k, Complex.exp (((m * (k.1 + k.2 ^ 2) : ℝ) : ℂ) * Complex.I) * A k‖ ≤ C * M / m ^ 3 := by
  set Ψ : ℝ × ℝ → ℝ := fun k => k.1 + k.2 ^ 2 with hΨ
  have hC : ContDiff ℝ ((3 : ℕ) + 1) Ψ := by fun_prop
  have hgrad : ∀ k, 1 ≤ ‖fderiv ℝ Ψ k + 0‖ := by
    intro k
    have hd : HasFDerivAt Ψ (ContinuousLinearMap.fst ℝ ℝ ℝ +
        (2 * k.2) • ContinuousLinearMap.snd ℝ ℝ ℝ) k := by
      have h1 := (hasFDerivAt_fst (𝕜 := ℝ) (p := k)).add
        ((hasFDerivAt_snd (𝕜 := ℝ) (p := k)).pow 2)
      convert h1 using 1
      ext <;> simp
    rw [add_zero, hd.fderiv]
    have := ContinuousLinearMap.le_opNorm (ContinuousLinearMap.fst ℝ ℝ ℝ +
        (2 * k.2) • ContinuousLinearMap.snd ℝ ℝ ℝ) ((1 : ℝ), (0 : ℝ))
    simpa using this
  obtain ⟨C, hC'⟩ := OscInt.API.nonStationaryFE 3 Ψ (Metric.closedBall 0 (1/4)) 1 hC
    (isCompact_closedBall 0 _) one_pos
  refine ⟨C, fun A M m hA hs hM hm => ?_⟩
  have := hC' A M 0 m hA hs hM (fun k _ => hgrad k) hm
  simpa only [hΨ, zero_apply, add_zero] using this
```

## T3 (uniform D4, φ = k₁³ − 3k₁k₂²)

Route: `d4BoundFE φ 1 0 (-3) 0`. `D²φ(0) = 0` comes from `iteratedFDeriv_two_eq_zero_of_diag` plus
`iteratedFDeriv_diag_of_poly`. The third-derivative identity comes from `iteratedFDeriv_diag_of_poly` (n = 3).
`discr = 108 ≠ 0` comes from `simp [Cubic.discr]`.

```lean
import OscInt.API

open OscInt.API.Poly in
theorem T3 : ∃ δ > 0, ∃ C : ℝ, ∀ (A : ℝ × ℝ → ℂ) (M : ℝ) (L : (ℝ × ℝ) →L[ℝ] ℝ) (t : ℝ),
    ContDiff ℝ 2 A → tsupport A ⊆ Metric.closedBall 0 δ →
    (∀ j ≤ 2, ∀ k, ‖iteratedFDeriv ℝ j A k‖ ≤ M) → 1 ≤ t →
    ‖∫ κ, Complex.exp (((t * (κ.1 ^ 3 - 3 * κ.1 * κ.2 ^ 2) + L κ : ℝ) : ℂ) * Complex.I) * A κ‖
      ≤ C * M * t ^ (-(2:ℝ)/3) := by
  set φ : ℝ × ℝ → ℝ := fun k => k.1 ^ 3 - 3 * k.1 * k.2 ^ 2 with hφ
  have hC : ContDiff ℝ 4 φ := by fun_prop
  have poly : ∀ v : ℝ × ℝ, ∀ s : ℝ, φ (0 + s • v) =
      ∑ i, (![0, 0, 0, v.1 ^ 3 - 3 * v.1 * v.2 ^ 2] : Fin 4 → ℝ) i * s ^ (i : ℕ) := by
    intro v s; simp [Fin.sum_univ_four, hφ]; ring
  have d2 : iteratedFDeriv ℝ 2 φ 0 = 0 := by
    apply iteratedFDeriv_two_eq_zero_of_diag (hC.contDiffAt.of_le (by norm_num))
    intro v
    have e : (![v, v] : Fin 2 → ℝ × ℝ) = fun _ => v := by funext i; fin_cases i <;> rfl
    rw [e, iteratedFDeriv_diag_of_poly (hC.of_le (by norm_num)) 0 v _ (poly v) 2 (by norm_num)]
    simp
  have d3 : ∀ x y : ℝ, (iteratedFDeriv ℝ 3 φ 0) ![(x, y), (x, y), (x, y)] =
      6 * (1 * x ^ 3 + 0 * x ^ 2 * y + (-3) * x * y ^ 2 + 0 * y ^ 3) := by
    intro x y
    have e : (![(x, y), (x, y), (x, y)] : Fin 3 → ℝ × ℝ) = fun _ => (x, y) := by
      funext i; fin_cases i <;> rfl
    rw [e, iteratedFDeriv_diag_of_poly (hC.of_le (by norm_num)) 0 (x, y) _ (poly (x, y)) 3
      (by norm_num)]
    simp [Nat.factorial]; ring
  obtain ⟨δ, hδ, C, hC'⟩ := OscInt.API.d4BoundFE φ 1 0 (-3) 0 hC d2 d3
    (by simp [Cubic.discr])
  exact ⟨δ, hδ, C, fun A M L t hA hs hM ht => by simpa only [hφ] using hC' A M L t hA hs hM ht⟩
```
